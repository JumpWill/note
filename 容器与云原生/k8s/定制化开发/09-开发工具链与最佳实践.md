# K8s 定制化开发工具链与最佳实践

## 一、为什么需要工具链

### 1.1 痛点

```text
K8s 定制化开发的特点：
  - 多语言（Go、Python、TypeScript）
  - 多框架（Kubebuilder、Operator SDK、controller-runtime）
  - 多组件（CRD、Controller、AA、Webhook、Scheduler、CNI、CSI、DP）
  - 多工具链（kubectl、yq、jsonnet、helm、kustomize）
  - 测试复杂（envtest、kind、e2e、framework）

没有统一工具链：
  - 每个项目重复搭脚手架
  - 代码风格不一致
  - 测试方法不统一
  - 部署脚本各异
  - 升级困难
```

### 1.2 工具链价值

```text
1. 标准化
   - 统一项目结构
   - 统一代码生成
   - 统一测试方法
   - 统一部署流程

2. 效率
   - 自动生成样板代码
   - 一键测试
   - 一键构建/发布
   - 一键部署

3. 最佳实践
   - 复用社区标准
   - 集成安全策略
   - 内置监控和调试
```

### 3.2 核心 Makefile

```makefile
# 镜像配置
IMG ?= registry.example.com/my-operator:v1.0.0

# 环境
ENVTEST_K8S_VERSION ?= 1.30.0

# 获取项目信息
GOBIN  ?= $(shell pwd)/bin
PATH  := $(PATH):$(GOBIN)
SHELL := bash
PROJECT := my-operator

# go 相关
.PHONY: go-mod
go-mod:
	go mod tidy

.PHONY: go-test
go-test:
	go test ./... -coverprofile cover.out

# CRD 与 RBAC
.PHONY: manifests
manifests: controller-gen
	$(CONTROLLER_GEN) crd:trivialVersions=true \
	  rbac:roleName=manager-role webhook paths="./..." output:crd:artifacts:config

.PHONY: generate
generate: controller-gen
	$(CONTROLLER_GEN) object:headerFile="hack/boilerplate.go.txt" paths="./..."

# 构建
.PHONY: build
build: generate fmt vet
	go build -o bin/manager cmd/main.go

.PHONY: run
run: generate fmt vet manifests
	go run ./cmd/main.go

# 镜像
.PHONY: docker-build
docker-build: test
	docker build -t ${IMG} .

.PHONY: docker-push
docker-push:
	docker push ${IMG}

# 部署
.PHONY: deploy
deploy: manifests
	kustomize build config/default | kubectl apply -f -

# 测试
.PHONY: test
test: generate fmt vet envtest
	go test ./... -coverprofile cover.out

.PHONY: envtest
envtest:
	KUBEBUILDER_ASSETS="$(shell $(ENVTEST) use $(ENVTEST_K8S_VERSION) --bin-dir $(LOCALBIN) -p path)" \
	  go test ./... -coverprofile cover.out

# 格式化与检查
.PHONY: fmt
fmt:
	go fmt ./...

.PHONY: vet
vet:
	go vet ./...

.PHONY: lint
lint:
	golangci-lint run
```

### 3.3 完整开发流程

```bash
# 1. 初始化项目
mkdir my-operator && cd my-operator
go mod init github.com/example/my-operator
kubebuilder init --domain example.com --repo github.com/example/my-operator

# 2. 创建 API
kubebuilder create api --group myapp --version v1 --kind MyResource

# 3. 编写业务逻辑
$EDITOR internal/controller/myresource_controller.go

# 4. 生成 CRD 与 RBAC

# 5. 本地测试
make run

# 6. 单元测试
make test

# 7. 构建镜像
make docker-build IMG=registry.example.com/my-operator:v1.0.0
make docker-push IMG=registry.example.com/my-operator:v1.0.0

# 8. 部署
make deploy
```

### 4.2 Manager 启动完整示例

```go
// cmd/main.go
package main

import (
    "os"
    "sigs.k8s.io/controller-runtime/pkg/cache"
    "sigs.k8s.io/controller-runtime/pkg/log/zap"
    "sigs.k8s.io/controller-runtime/pkg/metrics/server"
    "sigs.k8s.io/controller-runtime/pkg/webhook"
    "sigs.k8s.io/controller-runtime/pkg/client"
    "sigs.k8s.io/controller-runtime/pkg/manager"
    
    "github.com/example/my-operator/api/v1"
    "github.com/example/my-operator/internal/controller"
    ctrl "sigs.k8s.io/controller-runtime"
)

var (
    setupLog = ctrl.Log.WithName("setup")
)

func main() {
    var metricsAddr         string
    var enableLeaderElection bool
    var probeAddr            string
    var webhookServer        bool
    
    ctrl.SetLogger(zap.New(zap.UseDevMode(true)))
    
    // 1. 创建 Manager
    mgr, err := ctrl.NewManager(ctrl.GetConfigOrDie(), ctrl.Options{
        Scheme:                 scheme,
        Metrics:                metricsserver.Options{BindAddress: metricsAddr},
        HealthProbeBindAddress: probeAddr,
        LeaderElection:         enableLeaderElection,
        LeaderElectionID:       "my-operator-leader",
        WebhookServer:          webhook.NewServer(webhook.Options{Port: 9443}),
    })
    if err != nil {
        setupLog.Error(err, "unable to start manager")
        os.Exit(1)
    }
    
    // 2. 设置 Controller
    if err = (&controller.MyResourceReconciler{
        Client: mgr.GetClient(),
        Scheme: mgr.GetScheme(),
    }).SetupWithManager(mgr); err != nil {
        setupLog.Error(err, "unable to create controller")
        os.Exit(1)
    }
    
    // 3. 设置 Webhook
    if webhookServer {
        if err = (&v1.MyResource{}).SetupWebhookWithManager(mgr); err != nil {
            setupLog.Error(err, "unable to create webhook")
            os.Exit(1)
        }
    }
    
    // 4. 启动 Manager
    setupLog.Info("starting manager")
    if err := mgr.Start(ctrl.SetupSignalHandler()); err != nil {
        setupLog.Error(err, "problem running manager")
        os.Exit(1)
    }
}
```

## 五、envtest 单元测试

### 5.1 envtest 简介

```text
envtest = testing K8s API server in a library

优点：
  - 不需要真实 K8s 集群
  - 启动快
  - 完整 K8s API
  - 支持 CRD
  - 单元测试友好

启动 envtest：
  - Kubebuilder 自动集成
  - 需要下载 etcd + kube-apiserver + kubectl 二进制
  - 存储在 /tmp/envtest-binaries/
```

### 5.2 完整测试示例

```go
// internal/controller/suite_test.go
package controller

import (
    "context"
    "path/filepath"
    "testing"
    "time"
    
    . "github.com/onsi/ginkgo/v2"
    . "github.com/onsi/gomega"
    "k8s.io/client-go/kubernetes/scheme"
    "k8s.io/client-go/rest"
    "sigs.k8s.io/controller-runtime/pkg/client"
    "sigs.k8s.io/controller-runtime/pkg/envtest"
    logf "sigs.k8s.io/controller-runtime/pkg/log"
    "sigs.k8s.io/controller-runtime/pkg/log/zap"
    "sigs.k8s.io/controller-runtime/pkg/manager"
    
    myappv1 "github.com/example/my-operator/api/v1"
    "github.com/example/my-operator/internal/controller"
    
    metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
    "k8s.io/apimachinery/pkg/types"
)

var (
    testEnv   *envtest.Environment
    k8sClient client.Client
)

func TestAPIs(t *testing.T) {
    RegisterFailHandler(Fail)
    RunSpecs(t, "Controller Suite")
}

var _ = BeforeSuite(func() {
    logf.SetLogger(zap.New(zap.WriteTo(GinkgoWriter), zap.UseDevMode(true)))
    
    By("bootstrapping test environment")
    testEnv = &envtest.Environment{
        CRDDirectoryPaths:     []string{filepath.Join("..", "..", "config", "crd", "bases")},
        ErrorIfCRDPathMissing: true,
        BinaryAssetsDirectory: filepath.Join("..", "..", "bin", "k8s", "1.30.0-linux-amd64"),
    }
    
    var err error
    cfg, err := testEnv.Start()
    Expect(err).NotTo(HaveOccurred())
    Expect(cfg).NotTo(BeNil())
    
    err = myappv1.AddToScheme(scheme.Scheme)
    Expect(err).NotTo(HaveOccurred())
    
    k8sClient, err = client.New(cfg, client.Options{Scheme: scheme.Scheme})
    Expect(err).NotTo(HaveOccurred())
    Expect(k8sClient).NotTo(BeNil())
    
    k8sManager, err := ctrl.NewManager(cfg, ctrl.Options{
        Scheme:             scheme.Scheme,
        MetricsBindAddress: "0",
    })
    Expect(err).ToNot(HaveOccurred())
    
    err = (&MyResourceReconciler{
        Client: k8sManager.GetClient(),
        Scheme: k8sManager.GetScheme(),
        Log:    logf.Log,
    }).SetupWithManager(k8sManager)
    Expect(err).ToNot(HaveOccurred())
    
    go func() {
        err = k8sManager.Start(ctx)
        Expect(err).ToNot(HaveOccurred(), "failed to run manager")
    }()
})

var _ = AfterSuite(func() {
    By("tearing down the test environment")
    err := testEnv.Stop()
    Expect(err).NotTo(HaveOccurred())
})

var _ = Describe("MyResource Controller", func() {
    Context("When creating a new MyResource", func() {
        It("Should create a Pod with correct spec", func() {
            mr := &myappv1.MyResource{
                Spec: myappv1.MyResourceSpec{
                    Image:    "nginx:1.25",
                    Replicas: 3,
                },
            }
            mr.Name = "test"
            mr.Namespace = "default"
            
            Expect(k8sClient.Create(ctx, mr)).Should(Succeed())
            
            // 等待 reconcile
            Eventually(func() bool {
                f := &appsv1.Deployment{}
                err := k8sClient.Get(ctx, types.NamespacedName{
                    Name: "test", Namespace: "default",
                }, f)
                return err == nil && *f.Spec.Replicas == int32(3)
            }, timeout, interval).Should(BeTrue())
        })
    })
})
```

### 7.1 GitHub Actions

```yaml
# .github/workflows/test.yml
name: Test

on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v4
    
    - name: Set up Go
      uses: actions/setup-go@v5
      with:
        go-version: 1.22
    
    - name: Install envtest
      run: |
        curl -L https://go.kubebuilder.io/test-tools/1.30.0/linux/amd64/setup-envtest -o setup-envtest
        chmod +x setup-envtest
        mkdir -p bin/k8s
        ./setup-envtest use 1.30.0 -p bin/k8s
    
    - name: Run tests
      run: make test
    
    - name: Lint
      uses: golangci/golangci-lint-action@v3
      with:
        version: latest
    
    - name: Build image
      run: make docker-build IMG=ghcr.io/${{ github.repository }}:latest
    
    - name: Push image
      if: github.event_name == 'push' && github.ref == 'refs/heads/main'
      run: |
        echo "${{ secrets.GITHUB_TOKEN }}" | docker login ghcr.io -u ${{ github.actor }} --password-stdin
        make docker-push IMG=ghcr.io/${{ github.repository }}:latest
```

## 八、Operator SDK 与 OLM

### 8.1 OLM（Operator Lifecycle Manager）

```text
OLM = Operator Lifecycle Manager（K8s 官方）

OLM 的作用：
  - 管理 Operator 的全生命周期
  - 自动升级 Operator
  - 依赖管理
  - CRD 升级
  - 集群权限

适用场景：
  - Operator 要分发到多个客户
  - 需要自动升级
  - 需要依赖管理
```

### 8.2 使用 OLM 部署 Operator

```bash
# 1. 安装 OLM
kubectl create -f https://github.com/operator-framework/operator-lifecycle-manager/releases/download/v0.27.0/crds.yaml
kubectl create -f https://github.com/operator-framework/operator-lifecycle-manager/releases/download/v0.27.0/olm.yaml

# 2. 创建 ClusterServiceVersion（CSV）
cat <<EOF | kubectl apply -f -
apiVersion: operators.coreos.com/v1alpha1
kind: ClusterServiceVersion
metadata:
  name: my-operator.v1.0.0
  annotations:
    capabilities: Full Lifecycle
spec:
  displayName: My Operator
  description: My custom operator
  version: 1.0.0
  install:
    strategy: deployment
    spec:
      deployments:
      - name: my-operator
        spec:
          template:
            spec:
              containers:
              - name: my-operator
                image: my-operator:v1.0.0
                ports:
                - containerPort: 9443
EOF

# 3. 创建 Subscription
cat <<EOF | kubectl apply -f -
apiVersion: operators.coreos.com/v1alpha1
kind: Subscription
metadata:
  name: my-operator
  namespace: olm
spec:
  channel: stable
  name: my-operator
  source:
    name: operatorhubio
    registryNamespace: olm
EOF

# 4. 验证安装
kubectl get csv -n olm
kubectl get installplan -n olm

# 5. 使用 CRD
kubectl create -f myresource.yaml
```

## 九、Helm 与 Kustomize

### 9.1 Helm（Operator 打包）

```text
Helm = K8s 包管理工具

对于 Operator 的价值：
  - 一键安装 Operator + CRD + RBAC
  - 一键升级
  - 参数化配置
  - 版本管理
```

```bash
# 创建 Helm chart for Operator
helm create my-operator-chart

# 包含 CRD
echo "templates/" >> my-operator-chart/.helmignore
cp config/crd/bases/* my-operator-chart/templates/

# 安装 Operator
helm install my-operator my-operator-chart/

# 升级
helm upgrade my-operator my-operator-chart/
```

### 9.2 Kustomize（清单管理）

```yaml
# kustomization.yaml
apiVersion: kustomize.config.k8s.io/v1beta1
kind: Kustomization
resources:
- manager.yaml
- crds.yaml
- service-account.yaml

# 镜像标签覆盖
images:
- name: my-operator
  newName: registry.example.com/my-operator
  newTag: v1.2.0

# 通用 label
labels:
- includeSelectors: true
  pairs:
    app.kubernetes.io/managed-by: kustomize
    app.kubernetes.io/version: v1.0.0
```

```bash
# 构建清单
kustomize build config/default > install.yaml

# 部署
kubectl apply -f install.yaml
```

### 10.2 Makefile 模板

```makefile
# 镜像
IMG ?= controller:latest
TAG ?= $(shell git rev-parse --short HEAD)

# 工具
CONTROLLER_GEN = $(shell pwd)/bin/controller-gen
ENVTEST = $(shell pwd)/bin/setup-envtest
KUSTOMIZE = $(shell pwd)/bin/kustomize

# 颜色输出
BOLD := \033[1m
RESET := \033[0m

.PHONY: all
all: help

.PHONY: help
help: ## Display this help
	@awk 'BEGIN {FS = ":.*##"; printf "Usage:\n  make \033[36m\033[0m\n"} /^[a-zA-Z_0-9-]+:.*?##/ { printf "  \033[36m%-20s\033[0m %s\n", $$1, $$2 } /^##@/ { printf "  \033[1m%s\033[0m\n", substr($$0, 5) }' $(MAKEFILE_LIST)

##@ Development

.PHONY: fmt
fmt: ## Run go fmt against code
	go fmt ./...

.PHONY: vet
vet: ## Run go vet against code
	go vet ./...

##@ Testing

.PHONY: test
test: fmt vet envtest ## Run tests
	KUBEBUILDER_ASSETS="$(shell $(ENVTEST) use $(ENVTEST_K8S_VERSION) --bin-dir $(LOCALBIN) -p path)" \
		go test ./... -coverprofile cover.out

##@ Build

.PHONY: build
build: fmt vet ## Build manager binary
	go build -o bin/manager cmd/main.go

##@ Deployment

.PHONY: install
install: manifests ## Install CRDs into the K8s cluster specified in ~/.kube/config
	kubectl apply -f config/crd/bases

.PHONY: deploy
deploy: manifests ## Deploy controller to the K8s cluster specified in ~/.kube/config
	kustomize build config/default | kubectl apply -f -

##@ Generate

.PHONY: manifests
manifests: controller-gen ## Generate WebhookConfiguration, ClusterRole and CustomResourceDefinition objects
	$(CONTROLLER_GEN) \
	  rbac:roleName=manager-role \
	  crd:trivialVersions=true \
	  webhook \
	  output:crd:artifacts:config
```
