## 17.5 实战：TensorFlow 2 MLP 收益预测

```python
"""完整 MLP 收益预测"""
import numpy as np
import pandas as pd
import yfinance as yf
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Dropout, BatchNormalization
from tensorflow.keras.callbacks import EarlyStopping, ReduceLROnPlateau
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import TimeSeriesSplit

# 1. 数据
prices = yf.download(['AAPL', 'MSFT'], start='2015-01-01')['Close']
returns = prices.pct_change().dropna()

# 2. 特征
features = pd.DataFrame(index=returns.index)
features['mom_5'] = returns.rolling(5).mean()
features['mom_20'] = returns.rolling(20).mean()
features['vol_20'] = returns.rolling(20).std()
features['mom_5_20_diff'] = features['mom_5'] - features['mom_20']
features['vol_ratio'] = returns.rolling(5).std() / returns.rolling(60).std()
features = features.dropna()

# 3. 目标
target = returns.loc[features.index].stack().droplevel(1)

# 4. 标准化
scaler = StandardScaler()
X_scaled = scaler.fit_transform(features)

# 5. 时序 CV
tscv = TimeSeriesSplit(n_splits=5, test_size=63)
predictions = []

for train_idx, test_idx in tscv.split(X_scaled):
    X_train, X_val = X_scaled[train_idx], X_scaled[test_idx]

    model = Sequential([
        Dense(128, activation='relu', input_shape=(X_train.shape[1],)),
        BatchNormalization(),
        Dropout(0.3),
        Dense(64, activation='relu'),
        BatchNormalization(),
        Dropout(0.3),
        Dense(1),
    ])
    model.compile(optimizer='adam', loss='mse')

    early_stop = EarlyStopping(monitor='val_loss', patience=10, restore_best_weights=True)
    reduce_lr = ReduceLROnPlateau(monitor='val_loss', factor=0.5, patience=5)

    model.fit(
        X_train, returns.iloc[train_idx],
        validation_data=(X_val, returns.iloc[test_idx]),
        epochs=50, batch_size=32,
        callbacks=[early_stop, reduce_lr], verbose=0,
    )

    pred = pd.Series(model.predict(X_val).flatten(),
                     index=returns.iloc[test_idx].index)
    predictions.append(pred)

predictions = pd.concat(predictions)
```

## 17.6 实战：CNN 序列预测

```python
"""CNN 序列预测"""
import numpy as np
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv1D, MaxPooling1D, Dropout, Dense

# 1. 创建序列数据
def make_sequences(prices, window=20):
    X, y = [], []
    for i in range(window, len(prices)):
        X.append(prices[i-window:i])
        y.append(prices[i])
    return np.array(X), np.array(y)

X, y = make_sequences(prices.values, window=20)

# 2. CNN
model = Sequential([
    Conv1D(filters=32, kernel_size=3, activation='relu',
           input_shape=(X.shape[1], 1)),
    MaxPooling1D(pool_size=2),
    Conv1D(filters=64, kernel_size=3, activation='relu'),
    MaxPooling1D(pool_size=2),
    Dropout(0.3),
    tf.keras.layers.Flatten(),
    Dense(64, activation='relu'),
    Dropout(0.3),
    Dense(1),
])

model.compile(optimizer='adam', loss='mse')
model.fit(X, y, epochs=50, batch_size=32, validation_split=0.2)
```

"""LSTM 多资产预测"""
import numpy as np
import tensorflow as tf
from tensorflow.keras.layers import LSTM, Dense, Dropout
from tensorflow.keras.models import Sequential

# 多资产序列
def make_multiseq(prices_dict, window=20):
    X, y = [], []
    stocks = list(prices_dict.keys())
    arrays = [prices_dict[s].values for s in stocks]

    n = len(arrays[0])
    for i in range(window, n):
        X.append(np.stack([a[i-window:i] for a in arrays], axis=1))
        y.append(np.array([a[i] for a in arrays]))
    return np.array(X), np.array(y)

X, y = make_multiseq(prices, window=20)

# LSTM

    Dropout(0.3),

    Dropout(0.3),
    Dense(len(prices.columns)),
])

model.fit(X, y, epochs=50, batch_size=32)

## 17.10 实战：完整深度学习流水线

```python
"""端到端深度学习交易"""
import numpy as np
import pandas as pd
import yfinance as yf
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Dropout
from sklearn.preprocessing import RobustScaler
from sklearn.model_selection import TimeSeriesSplit
import pyfolio as pf

# 1. 数据
prices = yf.download('^GSPC', start='2010-01-01')['Close']
returns = prices.pct_change().dropna()

# 2. 特征工程
features = pd.DataFrame(index=returns.index)
features['ret_1'] = returns
features['ret_5'] = prices.pct_change(5)
features['ret_20'] = prices.pct_change(20)
features['vol_5'] = returns.rolling(5).std()
features['vol_20'] = returns.rolling(20).std()
features['mom_diff'] = features['ret_5'] - features['ret_20']
features = features.dropna()

target = returns.loc[features.index]

# 3. 标准化
scaler = RobustScaler()
X_scaled = scaler.fit_transform(features)
y_arr = target.values

# 4. 序列窗口
def make_seq(X, y, window):
    Xs, ys = [], []
    for i in range(window, len(X)):
        Xs.append(X[i-window:i])
        ys.append(y[i])
    return np.array(Xs), np.array(ys)

window = 20
X_seq, y_seq = make_seq(X_scaled, y_arr, window)

# 5. 时序 CV
tscv = TimeSeriesSplit(n_splits=5, test_size=63)
predictions = []

for train_idx, test_idx in tscv.split(X_seq):
    X_train, X_val = X_seq[train_idx], X_seq[test_idx]
    y_train, y_val = y_seq[train_idx], y_seq[test_idx]

    model = Sequential([
        LSTM(32, input_shape=(window, X_seq.shape[2])),
        Dropout(0.3),
        Dense(1),
    ])
    model.compile(optimizer='adam', loss='mse')

    early_stop = tf.keras.callbacks.EarlyStopping(patience=10, restore_best_weights=True)
    model.fit(X_train, y_train,
              validation_data=(X_val, y_val),
              epochs=50, batch_size=32,
              callbacks=[early_stop], verbose=0)

    pred = model.predict(X_val).flatten()
    predictions.append(pd.Series(pred, index=target.index[window:][test_idx]))

predictions = pd.concat(predictions)

# 6. 评估
signal = predictions.reindex(returns.index).fillna(0)

# 简单策略
position = np.sign(signal).shift(1)
strategy_returns = returns * position

# Sharpe
sharpe = strategy_returns.mean() / strategy_returns.std() * np.sqrt(252)
print(f"Sharpe: {sharpe:.2f}")

# 评估
pf.create_simple_tear_sheet(strategy_returns.dropna())
```

## 17.12 关键 takeaway

- DL 适合复杂非线性关系
- 数据要 **足够大**
- 防止过拟合：Dropout / EarlyStop
- 时序 CV 必备

## 17.13 实际陷阱

### 数据量

- 金融数据相对少
- 用预训练 / 迁移学习
- 用更简单的模型

### 信噪比

- 金融数据信噪比低
- 期望收益要现实（Sharpe < 1）

### 非平稳

- 金融分布随时间变
- 用 walk-forward CV

### 交易成本

- 模型换手率要高 → 成本低 alpha
- 优化时考虑成本

## 17.14 经验法则

| 建议 | 说明 |
|------|------|
| 从简单模型开始 | MLP / RF / GBM |
| 再上复杂模型 | CNN / LSTM |
| 强正则化 | 金融易过拟合 |
| 滚动训练 | 适应分布变化 |
| 评估含成本 | 含 slippage / fee |
