## 11.1 随机森林基础

### Bagging 思想

- 多棵树投票
- 减少过拟合
- 天然并行

### 算法

```
1. Bootstrap 采样训练集
2. 每棵树：随机选 m 个特征
3. 训练决策树（不剪枝）
4. 集成预测
```

### 优势

- 鲁棒
- 不易过拟合
- 特征重要度

### 劣势

- 回归预测偏平均
- 大模型大
- 解释性弱

## 11.5 滚动训练

```python
def rolling_rf(X, y, retrain_freq=21, lookback=252, **rf_params):
    """滚动训练 RF。"""
    predictions = []
    for t in range(lookback, len(X)):
        if (t - lookback) % retrain_freq == 0:
            X_train = X.iloc[t - lookback:t]
            y_train = y.iloc[t - lookback:t]
            model = RandomForestRegressor(**rf_params, n_jobs=-1)
            model.fit(X_train, y_train)

        pred = model.predict(X.iloc[t:t+1])[0]
        predictions.append(pred)

    return pd.Series(predictions, index=X.index[lookback:])
```

## 11.6 案例：日本股票多空策略

```python
"""日本股票多空策略"""
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import TimeSeriesSplit

# 假设有日本股票数据
prices_japan = pd.read_csv('japan_stocks.csv', index_col=0, parse_dates=True)

# 1. 构建特征
def build_features(prices):
    """对每只股票计算特征。"""
    features = pd.DataFrame()
    returns = prices.pct_change()

    features['mom_1m'] = prices.pct_change(20)
    features['mom_3m'] = prices.pct_change(60)
    features['mom_12m'] = prices.pct_change(252)
    features['vol_20'] = returns.rolling(20).std()
    features['vol_60'] = returns.rolling(60).std()
    features['skew_60'] = returns.rolling(60).skew()
    features['rsi_14'] = compute_rsi(prices, 14)
    features['ma_50_ratio'] = prices / prices.rolling(50).mean()

    return features

features = {stock: build_features(prices_japan[stock]) for stock in prices_japan.columns}

# 2. 合并到 MultiIndex
X = pd.concat(features, axis=1).dropna()
y = prices_japan.pct_change(5).shift(-5).stack()

# 3. 横截面标准化
def cross_sectional_rank(df):
    """每个时间点横截面排名。"""
    return df.groupby(level=0).rank(pct=True)

# 4. 滚动训练
tscv = TimeSeriesSplit(n_splits=5)
predictions = []

for train_idx, test_idx in tscv.split(X):
    X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
    y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]

    scaler = StandardScaler()
    X_train_s = scaler.fit_transform(X_train)
    X_test_s = scaler.transform(X_test)

    model = RandomForestRegressor(
        n_estimators=500,
        max_depth=8,
        min_samples_leaf=20,
        max_features='sqrt',
        n_jobs=-1,
    )
    model.fit(X_train_s, y_train)
    pred = pd.Series(model.predict(X_test_s), index=y_test.index)
    predictions.append(pred)

predictions = pd.concat(predictions)

# 5. 多空组合
def long_short_strategy(predictions, top_pct=0.2, bottom_pct=0.2, horizon_days=5):
    daily_returns = prices_japan.pct_change(horizon_days).shift(-horizon_days)
    daily_returns = daily_returns.loc[predictions.index.get_level_values(0).unique()]

    strategy_returns = []
    for date in predictions.index.get_level_values(0).unique():
        pred_t = predictions.loc[date]
        if pred_t.dropna().empty:
            continue

        # 横截面 rank
        ranked = pred_t.rank(pct=True)

        # 多空
        longs = ranked[ranked > 1 - top_pct].index
        shorts = ranked[ranked < bottom_pct].index

        if date not in daily_returns.index:
            continue

        # 收益
        long_ret = daily_returns.loc[date, longs].mean() if len(longs) > 0 else 0
        short_ret = daily_returns.loc[date, shorts].mean() if len(shorts) > 0 else 0

        strategy_returns.append({
            'date': date,
            'long_return': long_ret,
            'short_return': short_ret,
            'strategy_return': long_ret - short_ret,
        })

    return pd.DataFrame(strategy_returns).set_index('date')

results = long_short_strategy(predictions)
print(f"Sharpe: {results['strategy_return'].mean() / results['strategy_return'].std() * np.sqrt(252):.2f}")
```

```python
# 用 target encoding（避免独热维度过高）
def target_encode(X_train, y_train, X_test, categorical_cols, smoothing=10):
    encodings = {}
    for col in categorical_cols:
        means = y_train.groupby(X_train[col]).mean()
        global_mean = y_train.mean()
        # 平滑
        smoothed = (means * X_train.groupby(col).count() + global_mean * smoothing) / \
                  (X_train.groupby(col).count() + smoothing)
        encodings[col] = smoothed

    # 应用
    for col in categorical_cols:
        X_train[col + '_enc'] = X_train[col].map(encodings[col]).fillna(global_mean)
        X_test[col + '_enc'] = X_test[col].map(encodings[col]).fillna(global_mean)

    return X_train, X_test
```
