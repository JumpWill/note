## 9.1 时序模型基础

### 核心思想

时序数据有自相关 + 季节性 + 残差：

$$
X_t = T_t + S_t + \epsilon_t
$$

- $T_t$：趋势
- $S_t$：季节性
- $\epsilon_t$：噪声

### 平稳性

- **平稳**：均值 / 方差不随时间变
- **不平稳**：有单位根（unit root）

```python
from statsmodels.tsa.stattools import adfuller

def test_stationarity(ts):
    result = adfuller(ts.dropna())
    print(f"ADF Statistic: {result[0]:.4f}")
    print(f"p-value: {result[1]:.4f}")
    if result[1] < 0.05:
        print("平稳")
    else:
        print("不平稳")
```

## 9.2 ARIMA

### 概念

ARIMA(p, d, q)：
- AR(p)：$X_t = \phi_1 X_{t-1} + \dots + \phi_p X_{t-p}$
- I(d)：差分次数
- MA(q)：$X_t = \epsilon_t + \theta_1 \epsilon_{t-1} + \dots$

### 拟合

```python
import statsmodels.api as sm

# 手动指定
model = sm.tsa.ARIMA(returns, order=(1, 0, 1))
result = model.fit()
print(result.summary())

# 自动选择

    returns,
    start_p=1, start_q=1,
    max_p=3, max_q=3,
    seasonal=False,
    stepwise=True,
    trace=True,
)
print(auto_model.summary())
```

### 预测

```python
# 多步预测

```

## 9.3 VAR（向量自回归）

### 思想

多变量同时建模，相互影响：

$$
Y_t = c + A_1 Y_{t-1} + \dots + A_p Y_{t-p} + \epsilon_t
$$

```python

# 多变量
data = pd.DataFrame({
```

## 9.4 GARCH（波动率）

### 思想

波动率有聚集性：

$$
\begin{aligned}
r_t &= \sigma_t \epsilon_t \\
\sigma_t^2 &= \omega + \alpha r_{t-1}^2 + \beta \sigma_{t-1}^2
\end{aligned}
$$

### 拟合

```python
from arch import arch_model

# GARCH(1, 1)
am = arch_model(returns * 100, vol='Garch', p=1, q=1, dist='t')
res = am.fit()
print(res.summary())

# 预测
forecasts = res.forecast(horizon=5)
vol_forecast = forecasts.variance[-1:] ** 0.5
```

### 扩展

- **EGARCH**：杠杆效应
- **GJR-GARCH**：非对称
- **DCC-GARCH**：动态相关性

## 9.5 协整（Cointegration）

### 概念

两个非平稳序列的线性组合是平稳的：

$$
Y_t = \alpha + \beta X_t + \epsilon_t, \quad \epsilon_t \sim \text{stationary}
$$

### 检验

```python
from statsmodels.tsa.stattools import coint

# 两支股票是否协整
score, pvalue, _ = coint(stock_a, stock_b)
print(f"Cointegration p-value: {pvalue:.4f}")
if pvalue < 0.05:
    print("协整")
```

## 9.6 配对交易

### 步骤

1. 找协整对
2. 计算 spread = $Y - \beta X$
3. Z-score = (spread - mean) / std
4. 当 |z| > 2：开仓
5. 当 |z| → 0：平仓

### 实施

```python
def pair_trading_strategy(s1, s2, lookback=10, entry_z=2, exit_z=0.5):
    # 协整回归（β）
    model = LinearRegression()
    model.fit(s2.values.reshape(-1, 1), s1.values)
    beta = model.coef_[0]

    # 计算 spread
    spread = s1 - beta * s2

    # Z-score
    rolling_mean = spread.rolling(lookback).mean()
    rolling_std = spread.rolling(lookback).std()
    z_score = (spread - rolling_mean) / rolling_std

    # 信号
    signal = pd.Series(0, index=z_score.index)
    signal[z_score > entry_z] = -1   # 多 s2 空 s1
    signal[z_score < -entry_z] = 1   # 多 s1 空 s2
    signal[(z_score.abs() < exit_z) & (signal != 0)] = 0  # 平仓

    return signal, spread, z_score

# 用法
signal, spread, z = pair_trading_strategy(stock_a, stock_b)
```

## 9.7 Kalman Filter 在金融

### 动态线性回归

```python
from pykalman import KalmanFilter

def dynamic_hedge_ratio(s1, s2):
    """用 Kalman 估计动态 hedge ratio。"""
    delta = 1e-5
    trans_cov = delta / (1 - delta) * np.eye(2)
    obs_mat = np.expand_dims(np.vstack([[s2.values], [np.ones(len(s2))]]).T, axis=1)
    kf = KalmanFilter(
        n_dim_obs=1,
        n_dim_state=2,
        initial_state_mean=np.zeros(2),
        initial_state_covariance=np.ones((2, 2)),
        transition_matrices=np.eye(2),
        observation_matrices=obs_mat,
        observation_covariance=1.0,
        transition_covariance=trans_cov,
    )

    state_means, _ = kf.filter(s1.values)
    hedge_ratios = -state_means[:, 0]
    intercept = state_means[:, 1]
    spread = s1.values + hedge_ratios * s2.values + intercept
    return pd.Series(spread, index=s1.index), pd.Series(hedge_ratios, index=s1.index)

spread, hedge = dynamic_hedge_ratio(stock_a, stock_b)
```

```python
from statsmodels.tsa.arima.model import ARIMA

def arima_residual_features(returns, n_lags=5):
    features = pd.DataFrame(index=returns.index)
    for lag in range(1, n_lags + 1):
        model = ARIMA(returns.shift(lag - 1).dropna(), order=(1, 0, 0)).fit()
```

```python
from scipy.stats import norm

def bs_call(S, K, T, r, sigma):
    d1 = (np.log(S/K) + (r + sigma**2/2)*T) / (sigma*np.sqrt(T))
    d2 = d1 - sigma*np.sqrt(T)
    return S*norm.cdf(d1) - K*np.exp(-r*T)*norm.cdf(d2)
```

## 9.12 实战：完整统计套利

```python
"""完整统计套利策略"""
import yfinance as yf
import pandas as pd
import numpy as np
from statsmodels.tsa.stattools import coint
import statsmodels.api as sm

# 1. 数据
stocks = yf.download(['KO', 'PEP', 'WMT', 'TGT', 'HD', 'LOW'],
                     start='2015-01-01')['Close']

# 2. 找协整对
pairs = []
for i in range(len(stocks.columns)):
    for j in range(i + 1, len(stocks.columns)):
        score, pvalue, _ = coint(stocks.iloc[:, i].dropna(),
                                 stocks.iloc[:, j].dropna())
        if pvalue < 0.05:
            pairs.append((stocks.columns[i], stocks.columns[j], pvalue))

pairs.sort(key=lambda x: x[2])
print(f"Found {len(pairs)} cointegrated pairs")

# 3. 用第一对构建策略
top_pair = pairs[0]
s1, s2 = stocks[top_pair[0]], stocks[top_pair[1]]

# 协整回归
X = sm.add_constant(s2)
model = sm.OLS(s1, X).fit()
beta = model.params[1]

# Spread
spread = s1 - beta * s2

# Z-score
z_score = (spread - spread.mean()) / spread.std()

# 信号
signal = pd.Series(0, index=z_score.index)
signal[z_score > 2] = -1   # 多 s2 空 s1
signal[z_score < -2] = 1   # 多 s1 空 s2

# 回测
position = signal.shift(1)
returns_pair = position * (s1.pct_change() - beta * s2.pct_change())
returns_pair = returns_pair.dropna()

# 评估
import pyfolio as pf
pf.create_returns_tear_sheet(returns_pair)
```
