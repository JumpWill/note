## 16.1 回测哲学

### 重要

回测 ≠ 实盘。回测充满陷阱。

### Lopez de Prado 三原则

1. **回测不能仅用历史数据** — 需要 out-of-sample
2. **回测要有真实假设** — 含成本
3. **回测要避免过拟合** — 多次检验修正

## 16.2 简单回测引擎

```python
"""简单事件驱动回测"""
import pandas as pd
import numpy as np

def backtest(prices, signals, initial_cash=100000, transaction_cost=0.001):
    """
    prices: 收盘价
    signals: 信号（1: 持有，-1: 卖空，0: 无）
    """
    cash = initial_cash
    position = 0  # 持仓数量
    portfolio_value = []

    for i in range(len(prices)):
        price = prices.iloc[i]
        signal = signals.iloc[i]

        # 交易
        if signal == 1 and position == 0:
            # 买入
            position = cash / price * (1 - transaction_cost)
            cash = 0
        elif signal == -1 and position > 0:
            # 卖出
            cash = position * price * (1 - transaction_cost)
            position = 0

        # 计算组合价值
        value = cash + position * price
        portfolio_value.append(value)

    return pd.Series(portfolio_value, index=prices.index)
```

## 16.3 向量化回测

```python
"""向量化回测"""
def vectorized_backtest(prices, signals, transaction_cost=0.001):
    """基于信号权重回测。"""
    # 收益
    returns = prices.pct_change()

    # 仓位（滞后 1 期避免前看）
    positions = signals.shift(1).fillna(0)

    # 策略收益
    strategy_returns = positions * returns

    # 交易成本
    trades = positions.diff().abs()
    cost = trades * transaction_cost

    strategy_returns = strategy_returns - cost

    return strategy_returns
```

"""backtrader 回测"""
import backtrader as bt

class MAStrategy(bt.Strategy):
    def __init__(self):
        self.fast_ma = bt.indicators.SMA(period=10)
        self.slow_ma = bt.indicators.SMA(period=20)

    def next(self):
        if not self.position:
            if self.fast_ma[0] > self.slow_ma[0]:
                self.buy()
        else:
            if self.fast_ma[0] < self.slow_ma[0]:
                self.sell()

## 16.7 防过拟合

### Deflated Sharpe Ratio

```python
"""Deflated Sharpe Ratio (DSR)"""
import numpy as np
from scipy.stats import norm

def deflated_sharpe_ratio(returns, n_trials, risk_free_rate=0):
    """Deflated Sharpe Ratio (López de Prado)."""
    sharpe = (returns.mean() - risk_free_rate) / returns.std()
    sr_annual = sharpe * np.sqrt(252)

    # 偏度 / 峰度调整
    skew = returns.skew()
    kurt = returns.kurtosis()

    # 期望最大 Sharpe（n_trials 次试验）
    e_max_sr = np.sqrt(2 * np.log(n_trials)) - \
               (np.log(np.log(n_trials)) + np.log(4 * np.pi)) / (2 * np.sqrt(2 * np.log(n_trials)))

    # DSR
    dsr = norm.cdf(
        (sr_annual - e_max_sr) * np.sqrt(252 - 1) / np.sqrt(1 - skew * sr_annual + ((kurt - 1) / 4) * sr_annual**2)
    )

    return dsr

# 用法
n_trials = 100  # 测试了多少个策略
dsr = deflated_sharpe_ratio(strategy_returns, n_trials)
print(f"DSR: {dsr:.4f}")
```

### CSCV（Combinatorially-Symmetric Cross-Validation）

```python
def cscv_test(returns, n_splits=16, n_trials=100):
    """CSCV（组合对称交叉验证）。"""
    T = len(returns)
    block_size = T // n_splits

    # 分块
    blocks = [returns.iloc[i*block_size:(i+1)*block_size] for i in range(n_splits)]

    # 构造所有可能的 S/N 配置
    performance = []

    # 用 trial selection 思路（占位）
    for _ in range(n_trials):
        # 随机选一些块为 in-sample
        n_in = n_splits // 2
        in_idx = np.random.choice(n_splits, n_in, replace=False)
        out_idx = list(set(range(n_splits)) - set(in_idx))

        in_sample = pd.concat([blocks[i] for i in in_idx])
        out_sample = pd.concat([blocks[i] for i in out_idx])

        in_sharpe = in_sample.mean() / in_sample.std() * np.sqrt(252)
        out_sharpe = out_sample.mean() / out_sample.std() * np.sqrt(252)

        performance.append((in_sharpe, out_sharpe))

    # 计算过拟合概率
    overfit_count = sum(1 for i, o in performance if i > 0 and o < 0)
    prob_overfit = overfit_count / n_trials

    return prob_overfit
```

## 16.8 White's Reality Check

```python
def whites_reality_check(strategy_returns, benchmark_returns, n_bootstrap=1000):
    """White's Reality Check。"""
    # Bootstrap
    superior_count = 0

    for _ in range(n_bootstrap):
        # 重采样
        idx = np.random.choice(len(strategy_returns), len(strategy_returns))
        boot_strategy = strategy_returns.iloc[idx]
        boot_benchmark = benchmark_returns.iloc[idx]

        # 比较
        if boot_strategy.mean() > boot_benchmark.mean():
            superior_count += 1

    p_value = superior_count / n_bootstrap
    return p_value
```

## 16.10 实战：完整回测框架

```python
"""完整回测框架"""
import pandas as pd
import numpy as np
import yfinance as yf

# 1. 数据
prices = yf.download('AAPL', start='2010-01-01')['Close']
returns = prices.pct_change()

# 2. 策略（动量）
mom_20 = prices.pct_change(20).shift(1)  # 滞后 1 期
signal = (mom_20 > 0).astype(int) - (mom_20 < 0).astype(int)

# 3. 回测（含成本）
def backtest_with_costs(prices, signal, cost=0.001):
    positions = signal.shift(1).fillna(0)
    returns = prices.pct_change()
    trades = positions.diff().abs()

    strategy = positions * returns - trades * cost
    return strategy

strategy_returns = backtest_with_costs(prices, signal)

# 4. 评估
total_return = (1 + strategy_returns).prod() - 1
annual_return = strategy_returns.mean() * 252
annual_vol = strategy_returns.std() * np.sqrt(252)
sharpe = annual_return / annual_vol

cum_returns = (1 + strategy_returns).cumprod()
peak = cum_returns.cummax()
drawdown = (cum_returns - peak) / peak
max_dd = drawdown.min()

print(f"Total Return: {total_return:.2%}")
print(f"Annual Return: {annual_return:.2%}")
print(f"Annual Vol: {annual_vol:.2%}")
print(f"Sharpe: {sharpe:.2f}")
print(f"Max DD: {max_dd:.2%}")

# 5. DSR
n_trials = 10  # 实际测试的策略数
dsr = deflated_sharpe_ratio(strategy_returns, n_trials)
print(f"DSR: {dsr:.4f}")
```

- 市价单 ≠ 期望价

### 过度拟合

- 同一数据多次测试
