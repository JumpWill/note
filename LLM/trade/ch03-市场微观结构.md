## 3.3 订单簿 (Limit Order Book)

```python
"""订单簿重建"""
from collections import defaultdict

class OrderBook:
    """限价订单簿。"""
    def __init__(self):
        self.bids = defaultdict(dict)  # price -> {order_id: size}
        self.asks = defaultdict(dict)

    def add_order(self, side, price, order_id, size):
        book = self.bids if side == 'B' else self.asks
        book[price][order_id] = size

    def cancel_order(self, side, order_id):
        book = self.bids if side == 'B' else self.asks
        for price, orders in book.items():
            if order_id in orders:
                del orders[order_id]
                if not orders:
                    del book[price]
                break

    def execute(self, side, price, size):
        """从最优价开始执行。"""
        book = self.asks if side == 'B' else self.bids

        while size > 0 and book:
            best_price = min(book.keys()) if side == 'B' else max(book.keys())

            if (side == 'B' and price > best_price) or \
               (side == 'S' and price < best_price):
                break  # 不能成交

            orders = book[best_price]
            for order_id, order_size in list(orders.items()):
                if order_size <= size:
                    size -= order_size
                    del orders[order_id]
                else:
                    orders[order_id] -= size
                    size = 0
                    break

            if not orders:
                del book[best_price]

    def best_bid_ask(self):
        """最优买卖价。"""
        bb = max(self.bids.keys()) if self.bids else None
        ba = min(self.asks.keys()) if self.asks else None
        return bb, ba

    def mid_price(self):
        bb, ba = self.best_bid_ask()
        return (bb + ba) / 2 if bb and ba else None

    def spread(self):
        bb, ba = self.best_bid_ask()
        return ba - bb if bb and ba else None

    def depth(self, levels=5):
        """深度：前 N 档累计量。"""
        bid_depth = sum([sum(o.values()) for o in list(self.bids.values())[-levels:]])
        ask_depth = sum([sum(o.values()) for o in list(self.asks.values())[:levels]])
        return bid_depth, ask_depth
```

"""订单流不平衡"""
def order_flow_imbalance(events):
    """events 是订单流（add/cancel/execute）。"""
    ofi = 0

    for event in events:
        side = event.side
        size = event.size

        if event.type == 'add':
            ofi += size if side == 'B' else -size
        elif event.type == 'cancel':
            ofi -= size if side == 'B' else +size
        elif event.type == 'execute':
            ofi -= size if side == 'B' else +size

    return ofi

### 交易方向 (Trade Sign)

```python
"""Lee-Ready 算法：判断交易方向"""
def trade_direction(trade, book):
    """判断主买 / 主卖。"""
    mid = book.mid_price()

    if trade.price > mid:
        return 'buy'  # 主买
    elif trade.price < mid:
        return 'sell'  # 主卖
    else:
        # 报价规则
        bid, ask = book.best_bid_ask()
        if trade.price == ask:
            return 'buy'
        elif trade.price == bid:
            return 'sell'
        return 'unknown'

def price_impact(trades):
    """价格冲击：交易后价格变化。"""
    impacts = []
    for i in range(1, len(trades)):
        price_change = trades[i].price - trades[i-1].price
        volume = trades[i].size
        impacts.append(price_change / np.log(1 + volume))
    return np.mean(impacts)
```

### Kyle's Lambda

```python
def kyles_lambda(trades, prices):
    """Kyle's Lambda：价格冲击系数。"""
    # 价格变化 vs 净订单流
    net_flow = np.cumsum([t.direction * t.size for t in trades])
    price_changes = np.diff(prices)

    # 回归
    from sklearn.linear_model import LinearRegression
    model = LinearRegression()
    model.fit(net_flow[:-1].reshape(-1, 1), price_changes)

    return model.coef_[0]
```

## 3.6 ITCH 数据解析

```python
"""NASDAQ ITCH 5.0 解析"""
import struct

def parse_itch_add_order(msg):
    """Parse Add Order message."""
    stock_locate = struct.unpack('<H', msg[0:2])[0]
    tracking_number = struct.unpack('<H', msg[2:4])[0]
    timestamp = struct.unpack('<Q', msg[4:12])[0] / 1e9
    order_ref = struct.unpack('<Q', msg[12:20])[0]
    side = msg[20:21].decode()
    shares = struct.unpack('<I', msg[21:25])[0]
    stock = msg[2:7].decode()
    price = struct.unpack('<I', msg[27:31])[0] / 10000

    return {
        'type': 'add',
        'timestamp': timestamp,
        'order_ref': order_ref,
        'side': side,
        'shares': shares,
        'stock': stock,
        'price': price,
    }

def parse_itch_order_execute(msg):
    """Parse Order Executed message."""
    stock_locate = struct.unpack('<H', msg[0:2])[0]
    tracking_number = struct.unpack('<H', msg[2:4])[0]
    timestamp = struct.unpack('<Q', msg[4:12])[0] / 1e9
    order_ref = struct.unpack('<Q', msg[12:20])[0]
    executed_shares = struct.unpack('<I', msg[20:24])[0]

    return {
        'type': 'execute',
        'timestamp': timestamp,
        'order_ref': order_ref,
        'shares': executed_shares,
    }
```

## 3.7 重建订单簿

```python
"""从 ITCH 重建订单簿"""
def reconstruct_order_book(itch_file):
    """从 ITCH 文件重建订单簿。"""
    book = OrderBook()
    trades = []

    with open(itch_file, 'rb') as f:
        while True:
            msg_len = f.read(1)
            if not msg_len:
                break

            msg_type = msg_len.decode()
            msg = msg_len + f.read(message_size(msg_type) - 1)

            if msg_type == 'A':  # Add Order
                parsed = parse_itch_add_order(msg)
                book.add_order(
                    parsed['side'],
                    parsed['price'],
                    parsed['order_ref'],
                    parsed['shares'],
                )
            elif msg_type == 'E':  # Execute
                parsed = parse_itch_order_execute(msg)
                book.execute_order(parsed)
                trades.append(parsed)

    return book, trades
```

## 3.8 跳价检测

### 价格跳跃

```python
"""检测价格跳跃"""
import numpy as np

def detect_jumps(prices, threshold_std=3):
    """检测价格跳跃。"""
    returns = np.diff(np.log(prices))
    mean_return = np.mean(returns)
    std_return = np.std(returns)

    jumps = []
    for i, r in enumerate(returns):
        if abs(r - mean_return) > threshold_std * std_return:
            jumps.append({
                'index': i,
                'return': r,
                'price_change': prices[i+1] - prices[i],
            })

    return jumps
```

### Realized Variance

```python
"""已实现波动率（含跳跃分离）"""
def realized_variance(prices):
    """高频数据的已实现方差。"""
    returns = np.diff(np.log(prices))
    return np.sum(returns**2)

def bipower_variation(prices):
    """双幂次变差（不含跳跃部分）。"""
    returns = np.diff(np.log(prices))
    return (np.pi / 2) * np.sum(np.abs(returns[:-1]) * np.abs(returns[1:]))

def jump_component(prices):
    """跳跃成分。"""
    rv = realized_variance(prices)
    bv = bipower_variation(prices)
    return max(rv - bv, 0)
```
