## 2.3 alphalens-reloaded

评估 alpha 因子的库（Quantopian 开源）。

```python
import alphalens

# 准备因子数据（多索引： (date, asset)）
factor_data = alphalens.utils.get_clean_factor_and_forward_returns(
    factor=df['alpha'],          # 因子值
    prices=df['prices'],         # 资产价格
    quantiles=5,
    periods=(1, 5, 10),         # 持有期
)

# 因子 IC
ic = alphalens.performance.factor_information_coefficient(factor_data)
mean_ic = ic.mean()
print(f"Mean IC: {mean_ic:.4f}")

# 因子收益
returns, std = alphalens.performance.mean_return_by_quantile(factor_data)

# 因子换手率
turnover = alphalens.performance.quantile_turnover(factor_data['factor_quantile'], 1)

# Tear Sheet
alphalens.tears.create_returns_tear_sheet(factor_data)
```

### 关键指标

- **IC (Information Coefficient)**：因子值与未来收益的相关系数
- **Mean IC**：IC 平均值
- **IC Decay**：IC 衰减速度
- **Quantile Returns**：分层组合收益
- **Turnover**：换手率
- **Factor Returns**：因子解释的收益

# 策略示例
from zipline.api import (
    order_target_percent, symbol, set_commission, set_slippage,
    schedule_function, date_rules, time_rules,
)

def initialize(context):
    context.asset = symbol('AAPL')
    schedule_function(rebalance, date_rules.every_day(), time_rules.market_open())

def rebalance(context, data):
    hist = data.history(context.asset, 'price', 20, '1d')
    sma = hist.mean()
    price = data.current(context.asset, 'price')

    if price > sma:
        order_target_percent(context.asset, 1.0)
    else:
        order_target_percent(context.asset, 0.0)

def analyze(context, perf):
    fig = plt.figure()
    ax = fig.add_subplot(111)
    ax.plot(perf.index, perf['portfolio_value'])
    plt.show()

## 2.6 backtrader

另一个流行的事件驱动框架。

```python
import backtrader as bt

class MyStrategy(bt.Strategy):
    def __init__(self):
        self.sma = bt.indicators.SMA(period=20)
        self.rsi = bt.indicators.RSI()

    def next(self):
        if not self.position and self.rsi < 30:
            self.buy()
        elif self.position and self.rsi > 70:
            self.sell()

cerebro = bt.Cerebro()
cerebro.addstrategy(MyStrategy)
data = bt.feeds.YahooFinanceData(dataname='AAPL', fromdate=datetime(2010, 1, 1))
cerebro.adddata(data)
cerebro.run()
cerebro.plot()
```

## 2.18 ML4T 流水线示例

```python
"""端到端 ML4T 流水线"""
import pandas as pd
import numpy as np
import yfinance as yf
import talib
from sklearn.ensemble import RandomForestRegressor
from sklearn.model_selection import TimeSeriesSplit
import alphalens
import pyfolio as pf

# 1. 数据
data = yf.download(['AAPL', 'MSFT', 'GOOGL'], start='2015-01-01')

# 2. 因子
def compute_factors(df):
    df = df.copy()
    close = df['Close']
    df['momentum'] = close.pct_change(20)
    df['mean_rev'] = -close.pct_change(5)
    df['vol'] = close.pct_change().rolling(20).std()
    return df

factors = data.groupby(level=1, group_keys=False).apply(compute_factors)

# 3. 标签（次日收益）
df['return_1d'] = df['Close'].groupby(level=1).pct_change().shift(-1)

# 4. 模型
features = ['momentum', 'mean_rev', 'vol']
X = factors[features]
y = factors['return_1d']

tscv = TimeSeriesSplit(n_splits=5)
model = RandomForestRegressor(n_estimators=100)

for train_idx, test_idx in tscv.split(X):
    X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
    y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]
    model.fit(X_train, y_train)
    pred = model.predict(X_test)
    # 评估...

# 5. alphalens 评估
factor_data = alphalens.utils.get_clean_factor_and_forward_returns(
    pred, data['Close'], quantiles=5, periods=(1, 5, 10)
)
alphalens.tears.create_returns_tear_sheet(factor_data)

# 6. 简单回测（每日 top/bottom）
signals = pred.groupby(level=0).apply(lambda x: x.rank(pct=True))
top_assets = signals[signals > 0.8].index.get_level_values(1)
returns = (signals > 0.8).astype(int) * y

# 7. pyfolio tear sheet
pf.create_returns_tear_sheet(returns)
```
