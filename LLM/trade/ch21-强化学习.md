## 22.1 强化学习基础

### 核心概念

- **Agent**：决策者
- **Environment**：世界
- **State (s)**：状态
- **Action (a)**：动作
- **Reward (r)**：奖励
- **Policy (π)**：策略
- **Value (V/Q)**：价值

### MDP

$$
(s_t, a_t, r_t, s_{t+1})
$$

### 目标

最大化累计奖励：

$$
G_t = \sum_{k=0}^∞ \gamma^k r_{t+k+1}
$$

## 22.5 Q-Learning

### 公式

$$
Q(s, a) \leftarrow Q(s, a) + \alpha [r + \gamma \max_{a'} Q(s', a') - Q(s, a)]
$$

- 表格 Q-learning：状态空间小时
- 状态空间大 → DQN

## 22.6 DQN

### 思想

用神经网络近似 Q 函数：

$$
Q(s, a; \theta) \approx Q^*(s, a)
$$

### 关键技术

- **Experience replay**
- **Target network**
- **Reward clipping**

```python
import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np
import random
from collections import deque

class DQN(nn.Module):
    def __init__(self, state_dim, action_dim):
        super().__init__()
        self.fc1 = nn.Linear(state_dim, 128)
        self.fc2 = nn.Linear(128, 64)
        self.fc3 = nn.Linear(64, action_dim)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = torch.relu(self.fc2(x))
        return self.fc3(x)

class ReplayBuffer:
    def __init__(self, capacity=10000):
        self.buffer = deque(maxlen=capacity)

    def push(self, transition):
        self.buffer.append(transition)

    def sample(self, batch_size):
        return random.sample(self.buffer, batch_size)

    def __len__(self):
        return len(self.buffer)

class DQNAgent:
    def __init__(self, state_dim, action_dim, lr=1e-3, gamma=0.99, epsilon=0.1):
        self.state_dim = state_dim
        self.action_dim = action_dim
        self.gamma = gamma
        self.epsilon = epsilon

        self.policy_net = DQN(state_dim, action_dim)
        self.target_net = DQN(state_dim, action_dim)
        self.target_net.load_state_dict(self.policy_net.state_dict())

        self.optimizer = optim.Adam(self.policy_net.parameters(), lr=lr)
        self.replay_buffer = ReplayBuffer()

    def act(self, state):
        if random.random() < self.epsilon:
            return random.randint(0, self.action_dim - 1)
        with torch.no_grad():
            q_values = self.policy_net(torch.tensor(state, dtype=torch.float32))
            return q_values.argmax().item()

    def learn(self, batch_size=64):
        if len(self.replay_buffer) < batch_size:
            return

        batch = self.replay_buffer.sample(batch_size)
        states, actions, rewards, next_states, dones = zip(*batch)

        states = torch.tensor(np.array(states), dtype=torch.float32)
        actions = torch.tensor(actions, dtype=torch.long)
        rewards = torch.tensor(rewards, dtype=torch.float32)
        next_states = torch.tensor(np.array(next_states), dtype=torch.float32)
        dones = torch.tensor(dones, dtype=torch.float32)

        q_values = self.policy_net(states).gather(1, actions.unsqueeze(1))

        with torch.no_grad():
            next_q_values = self.target_net(next_states).max(dim=1)[0]
            target = rewards + (1 - dones) * self.gamma * next_q_values

        loss = nn.functional.mse_loss(q_values.squeeze(), target)
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
```

## 22.7 DDPG（连续动作）

### 思想

- Actor-Critic
- 连续动作空间

```python
class Actor(nn.Module):
    def __init__(self, state_dim, action_dim):
        super().__init__()
        self.fc1 = nn.Linear(state_dim, 128)
        self.fc2 = nn.Linear(128, 64)
        self.fc3 = nn.Linear(64, action_dim)

    def forward(self, state):
        x = torch.relu(self.fc1(state))
        x = torch.relu(self.fc2(x))
        return torch.tanh(self.fc3(x))  # -1 to 1

class Critic(nn.Module):
    def __init__(self, state_dim, action_dim):
        super().__init__()
        self.fc1 = nn.Linear(state_dim + action_dim, 128)
        self.fc2 = nn.Linear(128, 64)
        self.fc3 = nn.Linear(64, 1)

    def forward(self, state, action):
        x = torch.cat([state, action], dim=1)
        x = torch.relu(self.fc1(x))
        x = torch.relu(self.fc2(x))
        return self.fc3(x)
```

## 22.8 PPO

### 思想

- 策略梯度
- 限制更新步长

```python
class PPOAgent:
    def __init__(self, state_dim, action_dim, lr=3e-4, gamma=0.99, eps_clip=0.2):
        self.gamma = gamma
        self.eps_clip = eps_clip

        self.actor = Actor(state_dim, action_dim)
        self.critic = Critic(state_dim, action_dim)

        self.optimizer_actor = optim.Adam(self.actor.parameters(), lr=lr)
        self.optimizer_critic = optim.Adam(self.critic.parameters(), lr=lr)

    def update(self, states, actions, rewards, old_logprobs):
        # 计算优势
        advantages = self.compute_advantages(states, rewards)
        returns = self.compute_returns(states, rewards)

        # Actor loss (clipped)
        new_probs = self.actor(states)
        new_logprobs = torch.log(new_probs.gather(1, actions))

        ratio = torch.exp(new_logprobs - old_logprobs)
        surr1 = ratio * advantages
        surr2 = torch.clamp(ratio, 1 - self.eps_clip, 1 + self.eps_clip) * advantages
        actor_loss = -torch.min(surr1, surr2).mean()

        # Critic loss
        value_pred = self.critic(states, actions)
        critic_loss = nn.functional.mse_loss(value_pred, returns)

        # 更新
        self.optimizer_actor.zero_grad()
        actor_loss.backward()
        self.optimizer_actor.step()

        self.optimizer_critic.zero_grad()
        critic_loss.backward()
        self.optimizer_critic.step()
```

## 22.9 交易环境

### OpenAI Gym 风格

```python
import gym
from gym import spaces

class TradingEnv(gym.Env):
    def __init__(self, prices, features, initial_cash=100000, transaction_cost=0.001):
        super().__init__()
        self.prices = prices
        self.features = features
        self.initial_cash = initial_cash
        self.transaction_cost = transaction_cost

        self.action_space = spaces.Box(low=-1, high=1, shape=(1,))  # 仓位
        self.observation_space = spaces.Box(low=-np.inf, high=np.inf, shape=(features.shape[1] + 2,))

        self.reset()

    def reset(self):
        self.t = 0
        self.cash = self.initial_cash
        self.position = 0
        self.portfolio_value = self.initial_cash
        return self._get_state()

    def _get_state(self):
        # 状态 = 特征 + 仓位 + 现金比例
        feature = self.features[self.t]
        position_value = self.position * self.prices[self.t]
        cash_ratio = self.cash / self.portfolio_value
        position_ratio = position_value / self.portfolio_value
        return np.concatenate([feature, [position_ratio, cash_ratio]])

    def step(self, action):
        # action: 目标仓位 (-1 to 1)
        target_position_value = action[0] * self.portfolio_value
        current_position_value = self.position * self.prices[self.t]

        # 交易
        trade_value = target_position_value - current_position_value
        cost = abs(trade_value) * self.transaction_cost

        self.position = target_position_value / self.prices[self.t]
        self.cash -= trade_value + cost

        # 推进时间
        self.t += 1
        if self.t >= len(self.prices) - 1:
            done = True
        else:
            done = False

        # 收益
        new_portfolio_value = self.cash + self.position * self.prices[self.t]
        reward = (new_portfolio_value - self.portfolio_value) / self.portfolio_value
        self.portfolio_value = new_portfolio_value

        return self._get_state(), reward, done, {}
```

## 22.10 实战：DQN 交易

```python
"""DQN 交易"""
import numpy as np
import pandas as pd
import yfinance as yf
import torch

# 1. 数据
prices = yf.download('AAPL', start='2010-01-01')['Close'].values
returns = pd.Series(prices).pct_change().fillna(0).values

# 2. 特征
def build_features(prices, window=20):
    features = []
    for i in range(window, len(prices)):
        feat = [
            prices[i] / prices[i-window:i].mean() - 1,  # 均值偏离
            prices[i] / prices[i] - 1,  # 收益
            (prices[i-window:i].std() / prices[i-window:i].mean()),  # 波动率
        ]
        features.append(feat)
    return np.array(features)

features = build_features(prices)

# 3. 简化环境
class SimpleTradingEnv:
    def __init__(self, prices, features):
        self.prices = prices
        self.features = features
        self.reset()

    def reset(self):
        self.t = 20
        self.position = 0
        self.cash = 1.0
        self.portfolio_value = 1.0
        return self.features[self.t - 20]

    def step(self, action):
        # action: 0 (sell), 1 (hold), 2 (buy)
        target_position = [0, 0.5, 1.0][action]

        old_value = self.position * self.prices[self.t]
        self.position = target_position
        new_value = self.position * self.prices[self.t]

        self.t += 1
        done = self.t >= len(self.prices) - 1

        reward = new_value - old_value
        return self.features[self.t - 20], reward, done

# 4. 训练
env = SimpleTradingEnv(prices, features)
state_dim = features.shape[1]
agent = DQNAgent(state_dim, action_dim=3)

for episode in range(100):
    state = env.reset()
    total_reward = 0

    while True:
        action = agent.act(state)
        next_state, reward, done = env.step(action)

        agent.replay_buffer.push((state, action, reward, next_state, done))
        agent.learn()

        state = next_state
        total_reward += reward

        if done:
            break

    print(f"Episode {episode}: Reward = {total_reward:.4f}")
```

```python
"""PPO 交易"""
import yfinance as yf
from stable_baselines3 import PPO
from stable_baselines3.common.vec_env import DummyVecEnv

# 1. 数据
prices = yf.download('AAPL', start='2010-01-01')['Close'].values

# 2. 环境
env = DummyVecEnv([lambda: TradingEnv(prices, features)])

# 3. PPO 训练
model = PPO('MlpPolicy', env, verbose=1)
model.learn(total_timesteps=100000)

# 4. 评估
obs = env.reset()
total_reward = 0
while True:
    action, _ = model.predict(obs)
    obs, reward, done, _ = env.step(action)
    total_reward += reward
    if done:
        break

print(f"Total reward: {total_reward:.4f}")
```
