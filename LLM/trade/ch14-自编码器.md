## 20.1 自编码器基础

### 思想

$$
\text{Encoder: } z = f(x) \\
\text{Decoder: } \hat{x} = g(z)
$$

最小化重构误差：

$$
\mathcal{L} = \|x - \hat{x}\|^2
$$

- $z$：隐表示（编码）
- $\hat{x}$：重构

### 应用

- 降维
- 去噪
- 异常检测
- 特征学习
- 资产定价

## 20.6 条件自编码器（CVAE）

### 思想

编码 / 解码条件于资产特征：

$$
\hat{R}_i = \text{CAE}(F_i | C_i)
$$

- $R_i$：资产 i 的收益
- $F_i$：因子
- $C_i$：资产特征（行业、市值等）

### 实现

```python
import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, Dense, Concatenate

def build_cvae(n_factors, n_conditions):
    # 输入
    factor_input = Input(shape=(n_factors,), name='factors')
    cond_input = Input(shape=(n_conditions,), name='conditions')

    # 合并
    merged = Concatenate()([factor_input, cond_input])

    # Encoder
    x = Dense(64, activation='relu')(merged)
    x = Dense(32, activation='relu')(x)
    z = Dense(10, activation='relu', name='encoded')(x)

    # Decoder（用条件）
    decoder_input = Concatenate()([z, cond_input])
    x = Dense(32, activation='relu')(decoder_input)
    x = Dense(64, activation='relu')(x)
    output = Dense(n_factors, activation='linear')(x)

    model = Model([factor_input, cond_input], output)
    return model

# 训练
cvae = build_cvae(n_factors, n_conditions)
cvae.compile(optimizer='adam', loss='mse')

cvae.fit(
    [factors_train, conditions_train],
    returns_train,
    epochs=50, batch_size=32,
)
```

## 20.8 变分自编码器（VAE）

### 思想

- 学隐变量的概率分布
- KL 正则

```python
import tensorflow as tf
from tensorflow.keras.layers import Input, Dense, Lambda
from tensorflow.keras.models import Model

def build_vae(input_dim, latent_dim=10):
    # Encoder
    inputs = Input(shape=(input_dim,))
    x = Dense(64, activation='relu')(inputs)
    x = Dense(32, activation='relu')(x)
    z_mean = Dense(latent_dim)(x)
    z_log_var = Dense(latent_dim)(x)

    # 采样
    def sampling(args):
        z_mean, z_log_var = args
        epsilon = tf.random.normal(shape=tf.shape(z_mean))
        return z_mean + tf.exp(0.5 * z_log_var) * epsilon

    z = Lambda(sampling)([z_mean, z_log_var])

    # Decoder
    decoder_input = Input(shape=(latent_dim,))
    x = Dense(32, activation='relu')(decoder_input)
    x = Dense(64, activation='relu')(x)
    outputs = Dense(input_dim, activation='sigmoid')(x)
    decoder = Model(decoder_input, outputs)

    # 完整 VAE
    outputs = decoder(z)
    vae = Model(inputs, outputs)

    # KL 损失
    kl_loss = -0.5 * tf.reduce_sum(z_log_var - tf.square(z_mean) - tf.exp(z_log_var) + 1)

    def vae_loss(y_true, y_pred):
        reconstruction_loss = tf.reduce_mean(tf.square(y_true - y_pred))
        return reconstruction_loss + kl_loss

    vae.add_loss(vae_loss)
    vae.compile(optimizer='adam')

    return vae

vae = build_vae(input_dim=100)
vae.fit(X_train, X_train, epochs=50)
```

## 20.9 实战：CVAE 横截面收益预测

```python
"""CVAE 横截面收益预测"""
import numpy as np
import pandas as pd
import yfinance as yf
from sklearn.preprocessing import StandardScaler
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, Dense, Dropout, Concatenate

# 1. 数据
tickers = ['AAPL', 'MSFT', 'GOOGL', 'AMZN', 'META', 'NVDA', 'JPM', 'JNJ', 'XOM', 'V']
prices = yf.download(tickers, start='2010-01-01')['Close']
returns = prices.pct_change().dropna()

# 2. 特征工程（动量、波动率、量价）
features = pd.DataFrame(index=returns.index)
for ticker in tickers:
    for w in [5, 20, 60]:
        features[f'{ticker}_mom_{w}'] = prices[ticker].pct_change(w)
        features[f'{ticker}_vol_{w}'] = returns[ticker].rolling(w).std()
    features[f'{ticker}_logvol'] = np.log(prices[ticker].pct_change().rolling(20).std())

features = features.dropna()

# 3. 横截面数据
def create_panel(features, returns, ticker_col):
    panels = []
    for ticker in ticker_col:
        feat_cols = [c for c in features.columns if c.startswith(ticker + '_')]
        X = features[feat_cols].values
        y = returns[ticker].loc[features.index].values

        for i in range(60, len(X)):
            panels.append({
                'factors': X[i-60:i].flatten(),
                'target': y[i],
                'ticker': ticker,
                'date': features.index[i],
            })
    return pd.DataFrame(panels)

panel = create_panel(features, returns, tickers)

# 4. 资产特征（行业）
asset_features = pd.DataFrame({
    'ticker': tickers,
    'sector_id': [0, 0, 0, 0, 0, 0, 1, 2, 3, 4],  # 5 sectors
})
sector_oh = pd.get_dummies(asset_features['sector_id']).values
n_sectors = sector_oh.shape[1]

# 5. 模型
factor_dim = len([c for c in features.columns if c.startswith(tickers[0] + '_')]) * 60
n_conditions = n_sectors

factor_input = Input(shape=(factor_dim,), name='factors')
cond_input = Input(shape=(n_conditions,), name='conditions')

merged = Concatenate()([factor_input, cond_input])
x = Dense(128, activation='relu')(merged)
x = Dropout(0.3)(x)
x = Dense(64, activation='relu')(x)
x = Dropout(0.3)(x)
output = Dense(1)(x)

cvae = Model([factor_input, cond_input], output)
cvae.compile(optimizer='adam', loss='mse')

# 6. 准备训练数据
def prepare_panel(panel, asset_features, n_sectors):
    factors = np.stack(panel['factors'].values)
    conditions = np.stack([
        asset_features[asset_features['ticker'] == t][[f's_{i}' for i in range(n_sectors)]].values[0]
        for t in panel['ticker']
    ])
    targets = panel['target'].values
    return factors, conditions, targets

factors, conditions, targets = prepare_panel(panel, asset_features, n_sectors)

# 7. 训练
split_idx = int(0.8 * len(factors))
cvae.fit(
    [factors[:split_idx], conditions[:split_idx]], targets[:split_idx],
    validation_data=([factors[split_idx:], conditions[split_idx:]], targets[split_idx:]),
    epochs=50, batch_size=32,
)

# 8. 预测
preds = cvae.predict([factors[split_idx:], conditions[split_idx:]])

# 9. 横截面 IC
predictions_df = pd.DataFrame({
    'date': panel['date'].iloc[split_idx:].values,
    'ticker': panel['ticker'].iloc[split_idx:].values,
    'pred': preds.flatten(),
    'target': targets[split_idx:],
})

ic = predictions_df.groupby('date').apply(
    lambda x: x['pred'].corr(x['target'])
)
print(f"Mean IC: {ic.mean():.4f}")
```
