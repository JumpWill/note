## 24.2 ReAct

### 思想

- **Re**ason + **Act**
- 推理和动作交替

```python
# 伪代码
def react_agent(question):
    scratchpad = []

    while not done:
        # 思考
        thought = llm(f"""
        Question: {question}
        Past steps: {scratchpad}
        What should I do next?
        """)

        # 动作
        action = llm(f"""
        Thought: {step[0]['thought']}
        Choose an action from: {available_tools}
        """)

        # 执行
        observation = execute(action)

        # 更新
        scratchpad.append({
            'thought': thought,
            'action': action,
            'observation': observation,
        })

        if action.type == 'finish':
            return observation
```

```python
def reflexion_agent(task):
    memory = []

    for attempt in range(max_attempts):
        # 行动
        result = react_agent(task, memory)

        # 反思
        reflection = llm(f"""
        Task: {task}
```

```python
def tree_of_thoughts(question, n_branches=3, depth=3):
    root = Node(question)

    for d in range(depth):
        # 扩展
        for node in current_level:
            for _ in range(n_branches):
                thought = llm(f"Continue reasoning: {node.content}")
                child = Node(thought)
                node.children.append(child)

        # 评估
        for node in current_level:
            score = llm_evaluate(node.content)
            node.score = score

        # 选择 top
        current_level = top_k(current_level, k=n_branches)

    return best(current_level).content
```

## 24.5 LangGraph

### 状态机代理

```python
from langgraph.graph import StateGraph, END
from typing import TypedDict

# 状态
class State(TypedDict):
    question: str
    research: str
    analysis: str
    report: str

# 节点
def researcher(state):
    response = llm(f"Research: {state['question']}")
    return {'research': response}

def analyst(state):
    response = llm(f"Analyze: {state['research']}")
    return {'analysis': response}

def writer(state):
    response = llm(f"Write report based on analysis: {state['analysis']}")
    return {'report': response}

# 图
workflow = StateGraph(State)
workflow.add_node('researcher', researcher)
workflow.add_node('analyst', analyst)
workflow.add_node('writer', writer)

workflow.add_edge('researcher', 'analyst')
workflow.add_edge('analyst', 'writer')
workflow.add_edge('writer', END)

workflow.set_entry_point('researcher')

app = workflow.compile()
result = app.invoke({'question': 'Apple vs Microsoft'})
```

```python
            'type': 'object',
            'properties': {
                'ticker': {'type': 'string'},
                'period': {'type': 'string', 'enum': ['annual', 'quarterly']},
            },
            'required': ['ticker'],
        },
    },
]

# 代理循环
def agent_loop(messages, tools):
    while True:
        response = client.messages.create(
            model='claude-3-5-sonnet',
            max_tokens=1024,
            tools=tools,
            messages=messages,
        )

        if response.stop_reason == 'tool_use':
            tool_use = next(b for b in response.content if b.type == 'tool_use')
            result = execute_tool(tool_use.name, tool_use.input)
            messages.append({'role': 'assistant', 'content': response.content})
            messages.append({
                'role': 'user',
                'content': [{
                    'type': 'tool_result',
                    'tool_use_id': tool_use.id,
                    'content': str(result),
                }],
            })
        else:
            return response.content
```

## 24.7 金融研究代理

### 完整实现

```python
"""金融研究代理"""
from langchain.agents import Tool, initialize_agent
from langchain.llms import OpenAI
from langchain.tools import tool
import yfinance as yf

# 工具集
@tool
def get_stock_price(ticker: str) -> str:
    """获取股票当前价格。"""
    stock = yf.Ticker(ticker)
    return f"{ticker}: ${stock.history(period='1d')['Close'].iloc[-1]:.2f}"

@tool
def get_financials(ticker: str) -> str:
    """获取公司财务摘要。"""
    stock = yf.Ticker(ticker)
    info = stock.info
    return f"""
    P/E: {info.get('trailingPE')}
    EPS: {info.get('trailingEps')}
    Revenue: {info.get('totalRevenue')}
    """

@tool
def get_news(ticker: str) -> str:
    """获取最新新闻。"""
    stock = yf.Ticker(ticker)
    news = stock.news[:5]
    return "\n".join([n['title'] for n in news])

tools = [get_stock_price, get_financials, get_news]

# 代理
llm = OpenAI(model='gpt-4', temperature=0)

agent = initialize_agent(
    tools, llm,
    agent='zero-shot-react-description',
    verbose=True,
    max_iterations=10,
)

# 运行
result = agent.run("""
1. Get Apple's current stock price and financials
2. Get Microsoft's current stock price and financials
3. Compare which is more attractive
""")
```

## 24.8 多代理对抗辩论

### Adversarial Debate

```python
"""多代理对抗预测"""
from langchain.llms import OpenAI

llm = OpenAI(model='gpt-4')

class BullAgent:
    """多头代理。"""
    def argue(self, ticker):
        return llm(f"""
        You are a bullish analyst. Argue why {ticker} will go up.
        Use data and logic. Be persuasive.
        """)

class BearAgent:
    """空头代理。"""
    def argue(self, ticker):
        return llm(f"""
        You are a bearish analyst. Argue why {ticker} will go down.
        Use data and logic. Be persuasive.
        """)

class JudgeAgent:
    """裁判代理。"""
    def decide(self, bull_args, bear_args):
        return llm(f"""
        You are an impartial judge. Based on the arguments, decide:
        Bull: {bull_args}
        Bear: {bear_args}

        Provide a final forecast with confidence.
        """)

# 辩论
ticker = 'AAPL'

bull = BullAgent().argue(ticker)
decision = JudgeAgent().decide(bull, BearAgent().argue(ticker))

print(decision)
```

## 24.10 实战：完整金融代理

```python
"""完整金融研究代理"""
from langchain.agents import initialize_agent, AgentType
from langchain.llms import OpenAI
from langchain.tools import tool
from langchain.memory import ConversationBufferMemory
import yfinance as yf

# 1. 工具
@tool
def stock_analysis(ticker: str) -> str:
    """完整股票分析。"""
    stock = yf.Ticker(ticker)
    info = stock.info
    hist = stock.history(period='1y')

    # 基本面
    pe = info.get('trailingPE', 'N/A')
    eps = info.get('trailingEps', 'N/A')

    # 技术面
    sma_50 = hist['Close'].rolling(50).mean().iloc[-1]
    sma_200 = hist['Close'].rolling(200).mean().iloc[-1]
    current = hist['Close'].iloc[-1]

    return f"""
    {ticker} Analysis:
    Price: ${current:.2f}
    P/E: {pe}
    EPS: {eps}
    SMA50: ${sma_50:.2f}
    SMA200: ${sma_200:.2f}
    Trend: {'Bullish' if current > sma_50 > sma_200 else 'Bearish'}
    """

@tool
def compare_stocks(tickers: str) -> str:
    """多股票比较。"""
    tickers_list = tickers.split(',')
    results = []
    for t in tickers_list:
        stock = yf.Ticker(t.strip())
        info = stock.info
        results.append(f"{t.strip()}: P/E={info.get('trailingPE', 'N/A')}")
    return "\n".join(results)

@tool
def calculate_position(portfolio_value: float, risk_pct: float) -> str:
    """根据 Kelly 公式建议仓位。"""
    kelly = risk_pct / 100 * 0.5  # 半 Kelly
    position = portfolio_value * kelly
    return f"Recommended position: ${position:.2f}"

tools = [stock_analysis, compare_stocks, calculate_position]

# 2. 内存
memory = ConversationBufferMemory(memory_key='chat_history')

# 3. 代理
llm = OpenAI(model='gpt-4', temperature=0)
agent = initialize_agent(
    tools, llm,
    agent=AgentType.CONVERSATIONAL_REACT_DESCRIPTION,
    memory=memory,
    verbose=True,
    max_iterations=15,
)

# 4. 多轮对话
queries = [
    "Analyze Apple stock",
    "Compare it with Microsoft",
    "Given a $100k portfolio, what position size do you recommend?",
]

for q in queries:
    print(f"\nUser: {q}")
    result = agent.run(q)
    print(f"Agent: {result}")
```
