## 8.2 信号 → 权重

### 等权（多空）

```python
def signal_to_weight_equal(signal):
    """信号值标准化后 → 权重"""
    # 信号标准化
    rank = signal.rank(pct=True)

    # top quintile 多头
    longs = rank[rank > 0.8].index
    shorts = rank[rank < 0.2].index

    # 等权
    weight = pd.Series(0, index=signal.index)
    weight[longs] = 1.0 / len(longs) if len(longs) > 0 else 0
    weight[shorts] = -1.0 / len(shorts) if len(shorts) > 0 else 0
    return weight
```

### 信号加权（按信号大小）

```python
def signal_to_weight_proportional(signal):
    """按信号大小分配仓位"""
    normalized = signal - signal.mean()  # 中心化
    weights = normalized / normalized.abs().sum()  # 归一化
    return weights
```

### 波动率倒数加权

```python
def inverse_vol_weights(returns):
    """低波资产获得更高权重"""
    vol = returns.std()
    inv_vol = 1 / vol
    weights = inv_vol / inv_vol.sum()
    return weights
```

```python
# 简化
def apply_trading_costs(weights, current_weights, commission=0.001, slippage=0.0005):
    delta = weights - current_weights
    cost = (commission + slippage) * delta.abs().sum()
    return cost
```

### 多空平衡

```python
def dollar_neutral(weights):
    """多空总市值相等。"""
    long_sum = weights[weights > 0].sum()
    short_sum = -weights[weights < 0].sum()
    target = (long_sum + short_sum) / 2
    weights[weights > 0] *= target / long_sum
    weights[weights < 0] *= target / short_sum
    return weights
```

```python
# strategy.py
from zipline.api import (
    order_target_percent, symbol, schedule_function,
    date_rules, time_rules, get_datetime,
    set_commission, set_slippage, set_max_leverage,
)
import pandas as pd

def initialize(context):
    # 股票 universe
    context.universe = [symbol(s) for s in ['AAPL', 'MSFT', 'GOOGL', 'AMZN']]

    # 调仓频率
    schedule_function(rebalance, date_rules.month_start(), time_rules.market_open())

    # 真实约束
    set_commission(commission.PerShare(cost=0.005, min_trade_cost=1.0))
    set_slippage(slippage.VolumeShareSlippage(volume_limit=0.025, price_impact=0.02))
    set_max_leverage(1.5)

def rebalance(context, data):
    # 计算因子
    factor = compute_alpha(context, data)

    # 选 top N 多头
    longs = factor[factor > factor.quantile(0.8)].index

    # 等权
    for stock in longs:
        order_target_percent(stock, 0.15)

    # 不在 longs → 卖出
    for stock in context.portfolio.positions:
        if stock not in longs:
            order_target_percent(stock, 0)

def compute_alpha(context, data):
    """用 20 日动量作为 alpha。"""
    prices = data.history(context.universe, 'price', 21, '1d')
    return (prices.iloc[-1] / prices.iloc[0]) - 1

def analyze(context, perf):
    import matplotlib.pyplot as plt
    fig = plt.figure(figsize=(12, 8))
    perf['portfolio_value'].plot()
    plt.title('Portfolio Value')
    plt.savefig('backtest.png')
```

## 8.5 zipline Pipeline

### 用 Pipeline 表达复杂策略

```python
from zipline.pipeline import Pipeline, CustomFactor
from zipline.pipeline.data import USEquityPricing
from zipline.api import attach_pipeline, pipeline_output

class Momentum(CustomFactor):
    """20 日动量。"""
    inputs = [USEquityPricing.close]
    window_length = 21

    def compute(self, today, assets, out, close):
        out[:] = (close[-1] / close[0]) - 1

def initialize(context):
    pipe = Pipeline()
    mom = Momentum()
    pipe.add(mom, 'momentum')
    pipe.set_screen(mom.top(50))
    attach_pipeline(pipe, 'my_pipe')

def rebalance(context, data):
    pipe_result = pipeline_output('my_pipe')
    longs = pipe_result.sort_values('momentum', ascending=False).head(20).index

    for stock in longs:
        order_target_percent(stock, 0.05)
```

## 8.9 实战：完整 ML4T 流水线

```python
"""完整 ML4T 流水线示例"""
import yfinance as yf
import pandas as pd
import numpy as np
from sklearn.linear_model import Ridge
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import TimeSeriesSplit
import pyfolio as pf

# 1. 数据
prices = yf.download(['AAPL', 'MSFT', 'GOOGL', 'AMZN', 'META'],
                     start='2015-01-01')['Close']
returns = prices.pct_change()

# 2. 特征
def build_features(prices):
    features = pd.DataFrame()
    for col in prices.columns:
        s = prices[col]
        features[f'{col}_mom'] = s.pct_change(20)
        features[f'{col}_vol'] = s.pct_change().rolling(20).std()
    return features.dropna()

X = build_features(prices).dropna()
y = returns.shift(-1).loc[X.index].dropna()
X, y = X.align(y, join='inner', axis=0)

# 3. 滚动预测
tscv = TimeSeriesSplit(n_splits=5, test_size=63)
all_pred = []
for train_idx, test_idx in tscv.split(X):
    X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
    y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]

    scaler = StandardScaler()
    X_train_s = scaler.fit_transform(X_train)
    X_test_s = scaler.transform(X_test)

    model = Ridge(alpha=1.0)
    model.fit(X_train_s, y_train)
    pred = pd.DataFrame(model.predict(X_test_s),
                        index=X_test.index, columns=y_test.columns)
    all_pred.append(pred)

predictions = pd.concat(all_pred)

# 4. 信号 → 权重（每天选 top）
def compute_weights(signal):
    """每天选 top 2 多头，bottom 1 空头，等权。"""
    weights = pd.Series(0.0, index=signal.index)
    if not signal.dropna().empty:
        top2 = signal.nlargest(2)
        bot1 = signal.nsmallest(1)
        weights[top2.index] = 0.5
        weights[bot1.index] = -0.5
    return weights

# 5. 转换为策略收益
daily_weights = predictions.apply(compute_weights, axis=1)
strategy_returns = (daily_weights.shift(1) * returns).sum(axis=1)
strategy_returns = strategy_returns.dropna()

# 6. 评估
pf.create_full_tear_sheet(
    strategy_returns,
    benchmark_rets=returns.mean(axis=1),  # 等权基准
    live_start_date='2022-01-01',
)
```
