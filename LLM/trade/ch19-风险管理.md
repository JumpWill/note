## 19.3 VaR（Value at Risk）

### 历史 VaR

```python
def var_historical(returns, confidence=0.95):
    """历史 VaR。"""
    return np.percentile(returns, (1 - confidence) * 100)

# 1 天 VaR (95%)
var_1d_95 = var_historical(daily_returns, 0.95)

```

### 参数 VaR

```python
from scipy.stats import norm

def var_parametric(returns, confidence=0.95):
    """参数 VaR（假设正态分布）。"""
    mu = returns.mean()
    sigma = returns.std()
    z = norm.ppf(1 - confidence)

    return -(mu + z * sigma)

# 1 天 VaR
var_p = var_parametric(daily_returns, 0.95)
```

### Monte Carlo VaR

```python
def var_monte_carlo(returns, n_simulations=10000, confidence=0.95):
    """Monte Carlo VaR。"""
    mu = returns.mean()
    sigma = returns.std()

    simulations = np.random.normal(mu, sigma, n_simulations)

    return np.percentile(simulations, (1 - confidence) * 100)
```

## 19.4 CVaR（Conditional VaR / Expected Shortfall）

### 概念

VaR 的尾部期望（比 VaR 更稳健）

```python
def cvar_historical(returns, confidence=0.95):
    """CVaR / Expected Shortfall。"""
    var = np.percentile(returns, (1 - confidence) * 100)

    # 超过 VaR 的平均损失
    tail = returns[returns <= var]
    return tail.mean()

# 1 天 CVaR (95%)
cvar_1d_95 = cvar_historical(daily_returns, 0.95)
print(f"1-day CVaR (95%): {cvar_1d_95:.4f}")
```

### CVaR 优化

```python
"""CVaR 组合优化"""
import cvxpy as cp

def cvar_portfolio(returns, confidence=0.95):
    """最小化 CVaR。"""
    n = returns.shape[1]
    T = returns.shape[0]

    # 决策变量
    w = cp.Variable(n)
    alpha = cp.Variable()  # VaR
    z = cp.Variable(T, nonneg=True)  # 辅助变量

    # 目标
    cvar = alpha + (1 / (T * (1 - confidence))) * cp.sum(z)

    # 约束
    constraints = [
        cp.sum(w) == 1,
        w >= 0,
        z >= returns @ w + alpha,
        z >= 0,
    ]

    # 求解
    prob = cp.Problem(cp.Minimize(cvar), constraints)
    prob.solve()

    return w.value
```

## 19.5 风险归因

### 因子模型归因

```python
def factor_risk_attribution(returns, factor_loadings, factor_cov):
    """因子风险归因。"""
    # 因子风险
    factor_risk = factor_loadings @ factor_cov @ factor_loadings.T

    # 特异性风险
    specific_risk = np.diag(np.var(returns - returns @ factor_loadings, axis=0))

    # 总风险
    total_risk = factor_risk + specific_risk

    return {
        'factor_risk': np.sqrt(np.diag(factor_risk)),
        'specific_risk': np.sqrt(np.diag(specific_risk)),
        'total_risk': np.sqrt(np.diag(total_risk)),
    }
```

```python
def hypothetical_stress(portfolio, shock_dict):
    """假设压力测试。"""
    total_value = sum(portfolio.values())

    loss = 0
    for ticker, value in portfolio.items():
        if ticker in shock_dict:
            loss += value * shock_dict[ticker]
```

## 19.7 Drawdown 控制

### 最大回撤

```python
def max_drawdown(returns):
    """最大回撤。"""
    cum = (1 + returns).cumprod()
    peak = cum.cummax()
    dd = (cum - peak) / peak
    return dd.min()

# Calmar ratio
def calmar_ratio(returns):
    annual_return = returns.mean() * 252
    mdd = max_drawdown(returns)
    return annual_return / abs(mdd)
```

### 动态止损

```python
class DrawdownControl:
    """回撤控制。"""
    def __init__(self, max_drawdown=0.15, recovery_threshold=0.05):
        self.max_drawdown = max_drawdown
        self.recovery_threshold = recovery_threshold
        self.peak = None
        self.in_drawdown = False

    def update(self, equity):
        # 更新峰值
        if self.peak is None or equity > self.peak:
            self.peak = equity
            self.in_drawdown = False

        # 计算回撤
        drawdown = (equity - self.peak) / self.peak

        # 触发
        if drawdown < -self.max_drawdown:
            self.in_drawdown = True
            return 'reduce'  # 减少仓位
        elif self.in_drawdown and drawdown > -self.recovery_threshold:
            self.in_drawdown = False
            return 'recover'  # 恢复仓位

        return 'normal'
```

## 19.8 风险预算

### 风险平价

```python
def risk_budgeting(returns, target_risk_contributions):
    """风险预算。"""
    n = len(returns.columns)
    cov = returns.cov()

    # 优化求解
    from scipy.optimize import minimize

    def risk_budget_obj(w, cov, target):
        risk_contrib = w * (cov @ w) / np.sqrt(w @ cov @ w)
        return np.sum((risk_contrib - target)**2)

    constraints = [{'type': 'eq', 'fun': lambda w: w.sum() - 1}]
    bounds = [(0, 1)] * n

    res = minimize(
        risk_budget_obj, np.ones(n) / n,
        args=(cov.values, target_risk_contributions),
        constraints=constraints, bounds=bounds,
    )

    return res.x
```

## 19.9 动态对冲

### Delta 对冲（期权）

```python
def delta_hedge(stock_price, option_delta, shares_held):
    """Delta 对冲。"""
    # 期权 delta 影响
    hedge_shares = -shares_held * option_delta

    # 需要买卖的股数
    trade = hedge_shares - 0  # 假设初始 0 股

    return trade
```

```python
def beta_hedge(portfolio_beta, portfolio_value, futures_price, futures_multiplier=50):
    """用期货 Beta 对冲。"""
    # S&P 500 期货合约数
    contract_value = futures_price * futures_multiplier
    n_contracts = portfolio_beta * portfolio_value / contract_value

    return int(-n_contracts)
```

## 19.10 Deep Hedging

### 思想

用神经网络学最优对冲策略

```python
import torch
import torch.nn as nn

class HedgeNet(nn.Module):
    """Deep Hedging 网络。"""
    def __init__(self, state_dim, action_dim):
        super().__init__()
        self.lstm = nn.LSTM(state_dim, 64, batch_first=True)
        self.fc = nn.Linear(64, action_dim)

    def forward(self, state):
        lstm_out, _ = self.lstm(state)
        hedge_action = self.fc(lstm_out)
        return hedge_action

# 训练
def deep_hedge_loss(hedge_actions, returns, transaction_cost=0.001):
    """Deep Hedging 损失（含成本）。"""
    # 组合 PnL
    pnl = hedge_actions * returns - transaction_cost * hedge_actions.diff().abs()
    pnl = pnl.sum(dim=1)

    # CVaR 目标
    var = torch.quantile(pnl, 0.05)
    cvar = pnl[pnl <= var].mean()

    # 风险调整
    return -pnl.mean() + 2 * (-cvar)
```

## 19.11 Kill Switch

```python
class KillSwitch:
    """紧急停止。"""
    def __init__(self, config):
        self.max_drawdown = config.get('max_drawdown', 0.20)
        self.max_daily_loss = config.get('max_daily_loss', 0.05)
        self.max_position_loss = config.get('max_position_loss', 0.10)
        self.peak_equity = None
        self.daily_start = None

    def check(self, equity, positions):
        """检查是否触发。"""
        if self.peak_equity is None or equity > self.peak_equity:
            self.peak_equity = equity
        if self.daily_start is None:
            self.daily_start = equity

        # 1. 总回撤
        drawdown = (equity - self.peak_equity) / self.peak_equity
        if drawdown < -self.max_drawdown:
            return True, f"Max drawdown {drawdown:.2%}"

        # 2. 日损失
        daily_pnl = (equity - self.daily_start) / self.daily_start
        if daily_pnl < -self.max_daily_loss:
            return True, f"Daily loss {daily_pnl:.2%}"

        # 3. 单股损失
        for ticker, position in positions.items():
            pos_loss = position.get('unrealized_pnl_pct', 0)
            if pos_loss < -self.max_position_loss:
                return True, f"{ticker} position loss {pos_loss:.2%}"

        return False, None

    def trigger(self, reason):
        """触发。"""
        # 平仓
        close_all_positions()

        # 告警
        send_alert(f"KILL SWITCH: {reason}")

        # 暂停
        pause_system()
```
