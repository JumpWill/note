### 关系抽取

```python
# 抽取 CEO↔Company
def extract_ceo_relations(text):
    doc = nlp(text)
    relations = []

    for token in doc:
        if token.text in ['CEO', 'chief', 'executive']:
            # 找主语（公司）和宾语（人）
            for child in token.children:
                if child.dep_ == 'nsubj':
                    company = child.text
                if child.dep_ == 'attr':
                    person = child.text

            relations.append((company, 'CEO', person))

    return relations
```

## 23.3 图数据库

### Neo4j

```python
# pip install neo4j
from neo4j import GraphDatabase

driver = GraphDatabase.driver(
    'bolt://localhost:7687',
    auth=('neo4j', 'password'),
)

def add_company(tx, name):
    tx.run('MERGE (c:Company {name: $name})', name=name)

def add_relationship(tx, source, relation, dest):
    tx.run(
        '''
        MATCH (a:Company {name: $source})
        MATCH (b:Company {name: $dest})
        MERGE (a)-[r:RELATES {type: $relation}]->(b)
        ''',
        source=source, dest=dest, relation=relation,
    )

with driver.session() as session:
    for triple in triples:
        session.write_transaction(add_company, triple[0])
        session.write_transaction(add_relationship, *triple)

# 查询
with driver.session() as session:
    result = session.run(
        'MATCH (a:Company)-[:CEO]->(p:Person) RETURN a.name, p.name'
    )
    for record in result:
        print(f"{record['a.name']} CEO is {record['p.name']}")
```

## 23.6 图神经网络（GNN）

### 基本概念

- 节点嵌入（graph embedding）
- 消息传递（message passing）
- 用于分类 / 回归

### PyTorch Geometric

```python
# pip install torch-geometric
import torch
from torch_geometric.nn import GCNConv
from torch_geometric.data import Data

# 图数据
edge_index = torch.tensor([[0, 1, 1, 2], [1, 0, 2, 1]], dtype=torch.long)
x = torch.tensor([[1], [2], [3]], dtype=torch.float)

data = Data(x=x, edge_index=edge_index)

# GCN
class GCN(torch.nn.Module):
    def __init__(self, in_features, hidden, out_features):
        super().__init__()
        self.conv1 = GCNConv(in_features, hidden)
        self.conv2 = GCNConv(hidden, out_features)

    def forward(self, x, edge_index):
        x = self.conv1(x, edge_index).relu()
        x = self.conv2(x, edge_index)
        return x

model = GCN(1, 16, 1)
output = model(data.x, data.edge_index)
```

### 应用

```python
"""公司关系预测"""
import torch
from torch_geometric.nn import GATConv

class CompanyEmbedding(torch.nn.Module):
    """公司嵌入 + 关系预测。"""
    def __init__(self, n_features, hidden_dim=32):
        super().__init__()
        self.conv1 = GATConv(n_features, hidden_dim, heads=4)
        self.conv2 = GATConv(hidden_dim * 4, hidden_dim)

    def encode(self, x, edge_index):
        x = self.conv1(x, edge_index).relu()
        x = self.conv2(x, edge_index)
        return x

    def predict(self, z, edge_label_index):
        # 边的两端节点嵌入
        src = z[edge_label_index[0]]
        dst = z[edge_label_index[1]]
        return (src * dst).sum(dim=-1)
```

## 23.7 金融网络

### 公司网络

```python
# 构建公司共现网络
# 节点：公司
# 边：在同一新闻中共同出现

import pandas as pd
from collections import defaultdict

news = pd.read_csv('financial_news.csv')  # date, text, tickers

# 共现
co_occurrence = defaultdict(int)

for _, row in news.iterrows():
    tickers = row['tickers'].split(',')
    for i, t1 in enumerate(tickers):
        for t2 in tickers[i+1:]:
            pair = tuple(sorted([t1, t2]))
            co_occurrence[pair] += 1

# 构图
gex = nx.Graph()
gex.add_nodes_from(set(t for pair in co_occurrence for t in pair))

for (t1, t2), count in co_occurrence.items():
    if count > 5:  # 阈值
        gex.add_edge(t1, t2, weight=count)

# 中心性
centrality = nx.degree_centrality(gex)
print(sorted(centrality.items(), key=lambda x: -x[1])[:10])
```

```python
import spacy

def query_graph(graph, cypher_query):

    graph,
    'MATCH (a:Company {id: "Apple"})-[:COMPETES_WITH]->(b) RETURN b.id',
```

## 23.9 时序图

### Temporal KG

```python
# 节点：实体 + 时间戳
# 边：关系 + 时间戳

edges = [
    ('AAPL', 'CEO', 'Tim Cook', '2011-08-24'),
    ('AAPL', 'CEO', 'Tim Cook', '2023-01-01'),  # 现在
]

# 时序查询
def temporal_query(graph, entity, date):
    """查询某日期的关系。"""
    cypher = f"""
    MATCH (a {{id: '{entity}'}})-[r]->(b)
    WHERE r.start_date <= '{date}' AND r.end_date >= '{date}'
    RETURN b.id, r.type
    """
    return graph.query(cypher)
```

## 23.10 防止时间泄漏

### 关键问题

训练 KG 时不能偷看未来

```python
# 时间泄漏：用了未来数据训练
# 解决：只用截止某日期的数据

def temporal_safe_split(data, cutoff_date):
    train = data[data['date'] < cutoff_date]
    test = data[data['date'] >= cutoff_date]
    return train, test
```
