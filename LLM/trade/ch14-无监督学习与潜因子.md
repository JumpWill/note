## 13.2 PCA (主成分分析)

### 数学

求解协方差矩阵的特征值问题：

$$
\Sigma v_i = \lambda_i v_i
$$

最大特征值对应最大方差方向。

### Sklearn

```python

# 标准化
```

## 13.5 HRP（层次风险平价）

### 思想

用层次聚类代替逆协方差矩阵。

```python
import numpy as np
import pandas as pd
from scipy.cluster.hierarchy import linkage
from scipy.spatial.distance import squareform

def get_quasi_diag(link):
    """从 linkage 中获取 quasi-diagonalization 排序。"""
    link = link.astype(int)
    sort_ix = pd.Series([link[-1, 0], link[-1, 1]])
    num_items = link[-1, 3]

    while sort_ix.max() >= num_items:
        sort_ix.index = range(0, sort_ix.shape[0] * 2, 2)
        df0 = sort_ix[sort_ix >= num_items]
        i = df0.index
        j = df0.values - num_items
        sort_ix[i] = link[j, 0]
        df0 = pd.Series(link[j, 1], index=i + 1)
        sort_ix = sort_ix.append(df0)
        sort_ix = sort_ix.sort_index()
        sort_ix.index = range(sort_ix.shape[0])

    return sort_ix.tolist()

def recursive_bisection(cov, sort_ix):
    """递归二分。"""
    n = len(sort_ix)
    weights = np.ones(n)
    clusters = [sort_ix]

    while clusters:
        # 分两组
        new_clusters = []
        for cluster in clusters:
            if len(cluster) > 1:
                mid = len(cluster) // 2
                c1 = cluster[:mid]
                c2 = cluster[mid:]
                new_clusters.extend([c1, c2])

                # 计算两组合并方差
                cov1 = cov[np.ix_(c1, c1)].sum()
                cov2 = cov[np.ix_(c2, c2)].sum()

                # 按反方差分配
                alpha = 1 - cov1 / (cov1 + cov2)
                weights[c1] *= alpha
                weights[c2] *= (1 - alpha)

        clusters = new_clusters

    return weights / weights.sum()

def hrp_portfolio(returns):
    """HRP 主函数。"""
    cov = returns.cov()
    corr = returns.corr()

    # 距离
    dist = np.sqrt(0.5 * (1 - corr))
    dist_condensed = squareform(dist, checks=False)

    # 层次聚类
    link = linkage(dist_condensed, method='single')

    # quasi-diag
    sort_ix = get_quasi_diag(link)
    cov_sorted = cov.iloc[sort_ix].iloc[:, sort_ix]

    # 递归二分
    weights = recursive_bisection(cov_sorted.values, sort_ix)

    # 输出权重
    weights_dict = dict(zip(cov.columns[sort_ix], weights))
    return pd.Series(weights_dict)
```

## 13.6 NCO (嵌套聚类优化)

```python
from sklearn.cluster import KMeans

def clustered_optimization(returns, n_clusters=5):
    """NCO 主函数。"""
    # 1. 协方差聚类
    corr = returns.corr()
    dist = np.sqrt(0.5 * (1 - corr))
    dist_condensed = squareform(dist, checks=False)

    link = linkage(dist_condensed, method='ward')
    clusters = fcluster(link, t=n_clusters, criterion='maxclust')

    # 2. 每个 cluster 内部等权
    weights = pd.Series(0, index=returns.columns)

    for c in range(1, n_clusters + 1):
        cluster_stocks = returns.columns[clusters == c]
        n = len(cluster_stocks)
        if n > 0:
            weights[cluster_stocks] = 1.0 / (n * n_clusters)

    return weights
```

## 13.7 风险平价

```python
def risk_parity(returns, n_iter=100):
    """风险平价：每个资产贡献相等风险。"""
    cov = returns.cov()
    n = len(cov)
    weights = np.ones(n) / n

    for _ in range(n_iter):
        # 计算风险贡献
        portfolio_var = weights @ cov @ weights
        marginal_contrib = cov @ weights
        risk_contrib = weights * marginal_contrib / np.sqrt(portfolio_var)

        # 调整权重
        target = portfolio_var / n
        for i in range(n):
            weights[i] *= target / risk_contrib[i]
        weights /= weights.sum()

    return pd.Series(weights, index=returns.columns)
```

## 13.9 隐因子模型

### 假设

收益 = k 个隐因子 + 特异性残差

$$
R = B F + \epsilon
$$

- $R$：股票收益（$T \times N$）
- $B$：因子载荷（$T \times k$）
- $F$：因子（$T \times k$）
- $\epsilon$：特异性

### 估计

## 13.11 实战：完整组合优化

```python
"""多种组合优化对比"""
import pandas as pd
import numpy as np
import yfinance as yf
import matplotlib.pyplot as plt

prices = yf.download(['AAPL', 'MSFT', 'GOOGL', 'AMZN', 'META', 'JPM', 'JNJ', 'XOM'],
                     start='2015-01-01')['Close']
returns = prices.pct_change().dropna()

# 1. 等权
weights_eq = np.ones(len(returns.columns)) / len(returns.columns)

# 2. 反波动率
vol = returns.std() * np.sqrt(252)
inv_vol = 1 / vol
weights_iv = inv_vol / inv_vol.sum()

# 3. 均值-方差 (shrinkage)
import sklearn.covariance
cov = sklearn.covariance.ledoit_wolf(returns)[0]
mu = returns.mean() * 252
from scipy.optimize import minimize

def mv_objective(w, mu, cov, ra=2.0):
    return -w @ mu + ra * 0.5 * w @ cov @ w

constraints = [{'type': 'eq', 'fun': lambda w: w.sum() - 1}]
bounds = [(0, 1)] * len(returns.columns)
w0 = np.ones(len(returns.columns)) / len(returns.columns)

res = minimize(mv_objective, w0, args=(mu.values, cov, 2.0),
               constraints=constraints, bounds=bounds)
weights_mv = pd.Series(res.x, index=returns.columns)

# 4. HRP
weights_hrp = hrp_portfolio(returns)

# 5. 风险平价
weights_rp = pd.Series(risk_parity(returns), index=returns.columns)

# 比较
all_weights = pd.DataFrame({
    'equal': weights_eq,
    'inv_vol': weights_iv,
    'mv': weights_mv,
    'hrp': weights_hrp,
    'risk_parity': weights_rp,
}, index=returns.columns)

print(all_weights)

# 评估
for col in all_weights.columns:
    w = all_weights[col]
    port_returns = (returns * w).sum(axis=1)
    sharpe = port_returns.mean() / port_returns.std() * np.sqrt(252)
    vol = port_returns.std() * np.sqrt(252)
    max_dd = (port_returns.cumsum() - port_returns.cumsum().cummax()).min()
    print(f"{col}: Sharpe={sharpe:.2f}, Vol={vol:.2%}, MaxDD={max_dd:.2%}")
```
