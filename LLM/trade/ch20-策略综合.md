## 20.1 Fundamental Law of Active Management

### 公式

$$
\text{IR} \cdot \sqrt{\text{BR}} = \text{TC}
$$

- IR：信息比率（IC × √(宽)）
- BR：广度（决策频率）
- TC：交易能力（换手成本）

```python
def fundamental_law(active_returns, residuals):
    """Fundamental Law of Active Management。"""
    ic = np.corrcoef(active_returns, residuals)[0, 1]
    br = len(active_returns)
    return ic * np.sqrt(br)
```

## 20.2 IC-Sharpe 解耦

### 概念

- IC（信息系数）衡量预测能力
- Sharpe 衡量策略整体表现
- 两者关系：$\text{Sharpe} \approx \text{IC} \times \sqrt{N}$

```python
def ic_sharpe_relationship(ic, n_assets):
    """IC 与 Sharpe 的关系。"""
    return ic * np.sqrt(n_assets)

# 例如：IC = 0.05, 1000 只股票 → Sharpe ≈ 1.58
```

## 20.3 模型族级联

### 思想

按复杂度级联多个模型：

```
Layer 1: 简单模型（线性 / 因子） → 基础预测
Layer 2: 树模型 → 残差学习
Layer 3: 深度学习 → 复杂模式
```

```python
class CascadedModels:
    """级联模型。"""
    def __init__(self):
        self.layer1 = LinearRegression()  # 因子模型
        self.layer2 = GradientBoosting()  # 残差学习
        self.layer3 = NeuralNetwork()      # 复杂模式

    def fit(self, X, y):
        # Layer 1
        self.layer1.fit(X, y)
        pred1 = self.layer1.predict(X)
        residual1 = y - pred1

        # Layer 2
        self.layer2.fit(X, residual1)
        pred2 = self.layer2.predict(X)
        residual2 = residual1 - pred2

        # Layer 3
        self.layer3.fit(X, residual2)

    def predict(self, X):
        pred1 = self.layer1.predict(X)
        pred2 = self.layer2.predict(X)
        pred3 = self.layer3.predict(X)
        return pred1 + pred2 + pred3
```

## 20.4 模型融合

### 加权平均

```python
def weighted_ensemble(predictions_dict, weights=None):
    """加权融合。"""
    if weights is None:
        weights = {k: 1.0 / len(predictions_dict) for k in predictions_dict}

    ensemble = sum(weights[k] * predictions_dict[k] for k in predictions_dict)
    return ensemble
```

### Stacking

```python
from sklearn.linear_model import Ridge
from sklearn.model_selection import KFold

def stacking(models, X, y, n_folds=5):
    """Stacking：用第一层预测作为第二层特征。"""
    # 第一层 OOF 预测
    oof_predictions = np.zeros((len(y), len(models)))

    for i, (name, model) in enumerate(models.items()):
        kf = KFold(n_splits=n_folds, shuffle=False)

        for train_idx, val_idx in kf.split(X):
            model_clone = clone(model)
            model_clone.fit(X[train_idx], y[train_idx])
            oof_predictions[val_idx, i] = model_clone.predict(X[val_idx])

    # 第二层
    meta_model = Ridge()
    meta_model.fit(oof_predictions, y)

    return meta_model, models
```

### 动态权重

```python
def dynamic_weights(performances, lookback=63):
    """根据最近表现动态调权重。"""
    weights = {}
    for name, perf_history in performances.items():
        # 倒数加权（差越多权重越小）
        recent_sharpe = perf_history.iloc[-lookback:].mean() / perf_history.iloc[-lookback:].std()
        weights[name] = max(0, recent_sharpe)

    # 归一化
    total = sum(weights.values())
    if total > 0:
        weights = {k: v / total for k, v in weights.items()}

    return weights
```

## 20.5 多策略组合

```python
class MultiStrategy:
    """多策略组合。"""
    def __init__(self, strategies, weights=None):
        self.strategies = strategies
        self.weights = weights or {k: 1.0 / len(strategies) for k in strategies}

    def signals(self, data):
        """融合多策略信号。"""
        all_signals = {k: s.generate_signal(data) for k, s in self.strategies.items()}
        return weighted_ensemble(all_signals, self.weights)

    def update_weights(self, performances):
        """动态更新权重。"""
        self.weights = dynamic_weights(performances)
```

## 20.6 跨模型 decorrelation

### 思想

- 相似模型贡献不大
- 选择相关性低的模型

```python
def select_uncorrelated_models(predictions, threshold=0.7):
    """选择相关性低的模型。"""
    selected = []
    corr_matrix = predictions.corr()

    for model in predictions.columns:
        # 已选模型相关性
        if not selected:
            selected.append(model)
            continue

        # 检查相关性
        max_corr = max([abs(corr_matrix.loc[model, s]) for s in selected])
        if max_corr < threshold:
            selected.append(model)

    return selected
```

## 20.7 成本存活分析

### 思想

策略在不同成本水平下还能跑赢？

```python
def cost_survival_analysis(strategy_returns, benchmark_returns, cost_range=range(0, 50, 5)):
    """不同成本下表现。"""
    results = []

    for cost_bps in cost_range:
        # 假设成本
        net_returns = strategy_returns - (cost_bps / 10000) * 252  # 年化假设

        sharpe = net_returns.mean() / net_returns.std() * np.sqrt(252)
        excess = (net_returns - benchmark_returns).mean() * 252

        results.append({
            'cost_bps': cost_bps,
            'sharpe': sharpe,
            'excess_return': excess,
        })

    return pd.DataFrame(results)
```

## 20.8 Holdout Failure Modes

### 失败模式

```python
def holdout_failure_modes(returns):
    """识别回测失败模式。"""
    failures = {}

    # 1. Sharpe 衰减
    rolling_sharpe = returns.rolling(63).mean() / returns.rolling(63).std() * np.sqrt(252)
    failures['sharpe_decay'] = (rolling_sharpe.iloc[-1] < 0.5 * rolling_sharpe.max())

    # 2. 偏度变化
    failures['skew_change'] = (abs(returns.iloc[-252:].skew()) < 0.5)

    # 3. 极端损失频率
    extreme_loss_freq = (returns < returns.quantile(0.05)).sum() / len(returns)
    failures['extreme_loss'] = extreme_loss_freq > 0.10

    return failures
```

## 20.9 完整策略综合流水线

```python
"""完整策略综合"""
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression, Ridge
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.model_selection import TimeSeriesSplit

class StrategyAggregator:
    """策略聚合器。"""
    def __init__(self):
        self.models = {}
        self.weights = {}

    def add_model(self, name, model):
        self.models[name] = model

    def fit(self, X, y, time_series=True):
        """训练所有模型。"""
        for name, model in self.models.items():
            if time_series:
                # 时序训练
                model.fit(X, y)
            else:
                model.fit(X, y)

        # 评估（用 OOF）
        oof_preds = self._oof_predict(X, y)
        self._update_weights(oof_preds, y)

    def _oof_predict(self, X, y):
        """OOF 预测。"""
        tscv = TimeSeriesSplit(n_splits=5)
        oof = pd.DataFrame(index=range(len(y)), columns=self.models.keys())

        for train_idx, val_idx in tscv.split(X):
            X_train, X_val = X.iloc[train_idx], X.iloc[val_idx]
            y_train = y.iloc[train_idx]

            for name, model in self.models.items():
                from sklearn.base import clone
                m = clone(model)
                m.fit(X_train, y_train)
                oof.loc[val_idx, name] = m.predict(X_val)

        return oof.dropna()

    def _update_weights(self, oof_preds, y):
        """根据 IC 更新权重。"""
        ics = oof_preds.apply(lambda p: p.corr(y.loc[p.index]))

        # Sharpe-like 权重
        weights = ics.clip(lower=0)
        if weights.sum() > 0:
            weights = weights / weights.sum()
        else:
            weights = pd.Series(1.0 / len(ics), index=ics.index)

        self.weights = weights.to_dict()

    def predict(self, X):
        """融合预测。"""
        preds = {name: m.predict(X) for name, m in self.models.items()}
        return weighted_ensemble(preds, self.weights)
```

## 20.10 实战：模型族级联策略

```python
"""级联策略实战"""
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.neural_network import MLPRegressor

# 1. 准备
prices = yf.download(['AAPL', 'MSFT', 'GOOGL'], start='2015-01-01')['Close']
returns = prices.pct_change()

# 特征
features = pd.DataFrame()
for ticker in prices.columns:
    for w in [5, 20]:
        features[f'{ticker}_mom_{w}'] = prices[ticker].pct_change(w)

target = returns.mean(axis=1).shift(-5)

X = features.dropna()
y = target.loc[X.index]

# 2. 三个模型
splitter = TimeSeriesSplit(n_splits=5)
predictions = {}

for name, model in [('linear', LinearRegression()),
                     ('gbm', GradientBoostingRegressor(n_estimators=100)),
                     ('mlp', MLPRegressor(hidden_layer_sizes=(64, 32)))]:
    pred_list = []
    for train_idx, val_idx in splitter.split(X):
        m = clone(model)
        m.fit(X.iloc[train_idx], y.iloc[train_idx])
        pred_list.append(pd.Series(m.predict(X.iloc[val_idx]), index=y.iloc[val_idx].index))

    predictions[name] = pd.concat(pred_list)

# 3. IC-Sharpe 关系
for name, pred in predictions.items():
    ic = pred.corr(y.loc[pred.index])
    n_assets = 3
    expected_sharpe = ic * np.sqrt(n_assets)
    print(f"{name}: IC={ic:.4f}, Expected Sharpe={expected_sharpe:.2f}")

# 4. 加权
ics = {name: pred.corr(y.loc[pred.index]) for name, pred in predictions.items()}
weights = {name: max(0, ic) for name, ic in ics.items()}
total = sum(weights.values())
weights = {k: v / total for k, v in weights.items()}

ensemble_pred = sum(weights[k] * predictions[k] for k in predictions)
ic_ensemble = ensemble_pred.corr(y.loc[ensemble_pred.index])
print(f"Ensemble: IC={ic_ensemble:.4f}")
```

## 20.12 实战陷阱

### 过度集成

- 模型太多过拟合
- 选 3-5 个最好

### 相关模型

- 加权相似模型无意义
- decorrelation 关键

### 动态权重

- 表现变 → 调权重
- 用最近 N 期表现
