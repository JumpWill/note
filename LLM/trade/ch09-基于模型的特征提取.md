## 9.4 波动率特征

### 历史波动率

```python
def historical_volatility(returns, window=20):
    return returns.rolling(window).std() * np.sqrt(252)
```

### Parkinson 估计（用高低价）

```python
def parkinson_volatility(high, low, window=20):
    """Parkinson 波动率。"""
    ratio = np.log(high / low)
    return ratio.rolling(window).std() * np.sqrt(252 / (4 * np.log(2)))
```

### Garman-Klass 估计

```python
def garman_klass(open, high, low, close, window=20):
    """Garman-Klass 波动率。"""
    log_hl = np.log(high / low)
    log_co = np.log(close / open)

    gk = 0.5 * log_hl**2 - (2 * np.log(2) - 1) * log_co**2

    return gk.rolling(window).mean().apply(np.sqrt) * np.sqrt(252)
```

### Yang-Zhang 估计

```python
def yang_zhang(open, high, low, close, window=20):
    """Yang-Zhang 波动率（最准确）。"""
    log_ho = np.log(high / open)
    log_lo = np.log(low / open)
    log_co = np.log(close / open)

    log_oc = np.log(open / close.shift(1))
    log_oc_sq = log_oc**2

    # 三个分量
    var_overnight = log_oc_sq.rolling(window).mean()
    var_open = (log_ho * (log_ho - log_co) +
                log_lo * (log_lo - log_co)).rolling(window).mean()
    var_rs = log_co**2

    yz = (var_overnight +
          0.5 * var_open +
          (1 - 1 / 3) * var_rs)

    return yz.apply(np.sqrt) * np.sqrt(252)
```

## 9.5 GARCH 模型特征

```python
from arch import arch_model

def garch_volatility(returns, p=1, q=1):
    """GARCH(1,1) 波动率。"""
    model = arch_model(returns, vol='Garch', p=p, q=q)
    fit = model.fit(disp='off')

    # 条件波动率
    conditional_vol = fit.conditional_volatility
    return conditional_vol
```

### GARCH 残差

```python
def garch_residuals(returns):
    """GARCH 标准化残差。"""
    model = arch_model(returns, vol='Garch', p=1, q=1)
    fit = model.fit(disp='off')
    return fit.std_resid  # 标准化残差
```

## 9.6 滚动相关性特征

```python
def rolling_correlation(series1, series2, window=20):
    """滚动相关性。"""
    return series1.rolling(window).corr(series2)
```

### 滚动 Beta

```python
def rolling_beta(stock_returns, market_returns, window=60):
    """滚动 Beta。"""
    cov = stock_returns.rolling(window).cov(market_returns)
    var = market_returns.rolling(window).var()
    return cov / var
```

## 9.7 不确定性特征

### 预测区间

```python
def prediction_interval(model, X, confidence=0.95):
    """预测区间。"""
    from sklearn.utils import resample

    # Bootstrap
    predictions = []
    n_bootstrap = 100

    for _ in range(n_bootstrap):
        X_boot = resample(X)
        pred = model.predict(X_boot)
        predictions.append(pred)

    predictions = np.array(predictions)

    # 区间
    lower = np.percentile(predictions, (1 - confidence) / 2 * 100, axis=0)
    upper = np.percentile(predictions, (1 + confidence) / 2 * 100, axis=0)

    return lower, upper, predictions.mean(axis=0)
```

### 模型置信度

```python
def model_confidence(predictions, lower, upper):
    """模型置信度（区间宽度倒数）。"""
    interval_width = upper - lower
    confidence = 1 / (interval_width + 1e-10)
    return confidence
```

## 9.8 残差特征

### 异常检测

```python
def outlier_detection(returns, threshold=3):
    """Z-score 异常检测。"""
    z = (returns - returns.mean()) / returns.std()
    outliers = returns[abs(z) > threshold]
    return outliers
```

### 去噪

```python
def wavelet_denoise(series, wavelet='db5', level=3):
    """小波去噪。"""
    import pywt

    coeffs = pywt.wavedec(series, wavelet, level=level)

    # 阈值
    sigma = np.median(np.abs(coeffs[-1])) / 0.6745
    threshold = sigma * np.sqrt(2 * np.log(len(series)))

    # 软阈值
    coeffs[1:] = [pywt.threshold(c, threshold, mode='soft') for c in coeffs[1:]]

    # 重构
    return pywt.waverec(coeffs, wavelet)
```

## 9.9 卡尔曼滤波

```python
from pykalman import KalmanFilter

def kalman_filter(series):
    """卡尔曼滤波（隐状态估计）。"""
    kf = KalmanFilter(
        transition_matrices=[1],
        observation_matrices=[1],
        initial_state_mean=series.iloc[0],
        initial_state_covariance=1,
        observation_covariance=1,
        transition_covariance=0.01,
    )

    state_means, _ = kf.filter(series.values)
    return pd.Series(state_means.flatten(), index=series.index)
```

### 应用：动态对冲比率

```python
"""Kalman 动态 hedge ratio"""
def dynamic_hedge_ratio(price_a, price_b):
    """动态 hedge ratio（pair trading）。"""
    kf = KalmanFilter(
        transition_matrices=np.array([[1, 0], [0, 1]]),
        observation_matrices=np.array([[1, -1]]),
        initial_state_mean=np.array([0, 0]),
        initial_state_covariance=np.eye(2),
        observation_covariance=np.array([[1]]),
        transition_covariance=np.eye(2) * 0.0001,
    )

    state, _ = kf.filter(price_a.values - price_b.values)
    hedge_ratios = -state[:, 1]  # beta

    return pd.Series(hedge_ratios, index=price_a.index)
```

"""完整特征工程流水线"""
import pandas as pd
import numpy as np

def build_features(prices, returns):
    features = pd.DataFrame(index=returns.index)

    # 1. 动量
    for w in [5, 20, 60]:
        features[f'mom_{w}'] = prices.pct_change(w)

    # 2. 波动率
    features['vol_20'] = historical_volatility(returns, 20)
    features['vol_60'] = historical_volatility(returns, 60)

    # 3. 比率
    features['vol_ratio'] = features['vol_20'] / features['vol_60']

    # 4. GARCH 残差
    garch_vol = garch_volatility(returns)
    features['garch_vol'] = garch_vol
    features['garch_resid'] = garch_residuals(returns)

    market_returns = yf.download('^GSPC')['Close'].pct_change()
