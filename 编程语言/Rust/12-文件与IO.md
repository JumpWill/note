## 二、路径处理 (`Path` / `PathBuf`)

```rust
use std::path::{Path, PathBuf};

fn main() {
    let p = Path::new("/usr/local/bin");

    // 拆分
    println!("parent: {:?}", p.parent());    // /usr/local
    println!("file: {:?}", p.file_name());   // bin
    println!("stem: {:?}", p.file_stem());   // bin
    println!("ext: {:?}", p.extension());    // None

    // 拼接
    let mut p2 = PathBuf::from("/tmp");
    p2.push("data.txt");
    println!("{}", p2.display()); // /tmp/data.txt

    // 检查
    println!("exists: {}", p.exists());
    println!("is_file: {}", p.is_file());
    println!("is_dir: {}", p.is_dir());
}
```

### 2. 分块读取

```rust
use std::fs::File;
use std::io::{Read, BufReader};

fn main() -> std::io::Result<()> {
    let mut file = File::open("hello.txt")?;

    // 读到 String
    let mut s = String::new();
    file.read_to_string(&mut s)?;

    // 读到缓冲区
    let mut buf = [0u8; 1024];
    let n = file.read(&mut buf)?;

    // BufReader(推荐,减少系统调用)
    let reader = BufReader::new(File::open("hello.txt")?);
    for line in reader.lines() {
        println!("{}", line?);
    }

    Ok(())
}
```

### 3. 逐行读取

```rust
use std::fs::File;
use std::io::{BufRead, BufReader};

fn read_lines(path: &str) -> std::io::Result<Vec<String>> {
    let file = File::open(path)?;
    let reader = BufReader::new(file);
    reader.lines().collect()
}

fn main() -> std::io::Result<()> {
    let lines = read_lines("hello.txt")?;
    for (i, line) in lines.iter().enumerate() {
        println!("{}: {}", i, line);
    }
    Ok(())
}
```

### 1. 一次性写入

```rust
use std::fs;

fn main() -> std::io::Result<()> {
    fs::write("output.txt", "Hello, World!")?;
    fs::write("data.bin", vec![0x01, 0x02, 0x03])?;
```

### 2. 使用 File

```rust
use std::fs::File;
use std::io::Write;

fn main() -> std::io::Result<()> {
    // 创建(覆盖)
    let mut file = File::create("output.txt")?;
    file.write_all(b"Hello\n")?;
    writeln!(file, "Number: {}", 42)?;

    // BufWriter(推荐,减少系统调用)
    use std::io::BufWriter;
    let mut bw = BufWriter::new(File::create("output.txt")?);
    bw.write_all(b"buffered\n")?;
    bw.flush()?;  // 显式刷新

    Ok(())
}
```

### 3. OpenOptions 精细控制

```rust
use std::fs::OpenOptions;

fn main() -> std::io::Result<()> {
    let file = OpenOptions::new()
        .read(true)
        .write(true)
        .create(true)
        .append(true)       // 追加
        .truncate(false)    // 不截断
        .open("log.txt")?;

    Ok(())
}
```

| 选项 | 作用 |
| --- | --- |
| `read` | 可读 |
| `write` | 可写 |
| `append` | 追加 |
| `truncate` | 截断已有内容 |
| `create` | 不存在则创建 |
| `create_new` | 仅新建(已存在则失败) |

## 五、文件元数据

```rust
use std::fs;
use std::os::unix::fs::PermissionsExt;

fn main() -> std::io::Result<()> {
    let meta = fs::metadata("file.txt")?;

    println!("size: {} bytes", meta.len());
    println!("readonly: {}", meta.permissions().readonly());
    println!("is_file: {}", meta.is_file());
    println!("is_dir: {}", meta.is_dir());
    println!("is_symlink: {}", meta.file_type().is_symlink());

    if let Ok(m) = meta.modified() {
        println!("modified: {:?}", m);
    }

    // Unix 权限
    #[cfg(unix)]
    {
        let mode = meta.permissions().mode();
        println!("mode: {:o}", mode);

        // 修改权限
        let mut perms = meta.permissions();
        perms.set_mode(0o755);
        fs::set_permissions("file.txt", perms)?;
    }

    Ok(())
}
```

## 十一、错误处理模式

### 1. 用 thiserror 包装 IO 错误

```rust
// Cargo.toml: thiserror = "1"
use thiserror::Error;
use std::io;

#[derive(Error, Debug)]
pub enum FileError {
    #[error("文件不存在: {0}")]
    NotFound(String),
    #[error("权限不足: {0}")]
    PermissionDenied(String),
    #[error("IO 错误: {0}")]
    Io(#[from] io::Error),
}

pub fn read_file_safe(path: &str) -> Result<String, FileError> {
    let s = std::fs::read_to_string(path).map_err(|e| match e.kind() {
        io::ErrorKind::NotFound => FileError::NotFound(path.to_string()),
        io::ErrorKind::PermissionDenied => FileError::PermissionDenied(path.to_string()),
        _ => FileError::Io(e),
    })?;
    Ok(s)
}
```

### 2. 用 anyhow 加上下文

```rust
use anyhow::{Context, Result};
use std::fs;

fn load_config(path: &str) -> Result<String> {
    fs::read_to_string(path)
        .with_context(|| format!("无法读取配置: {}", path))
}
```

## 十三、要点速记

- **`Path` 是不可变视图,`PathBuf` 是可拥有**
- **`fs::read_to_string` / `fs::write` 一行搞定**
- **大文件用 `BufReader` / `BufWriter` 减少 syscall**
- **`OpenOptions` 精细控制 read/write/append/truncate**
- **跨平台优先 `Path`,不要拼字符串路径**
- **`?` + `From` 自动传播 IO 错误**
- **`memmap` 零拷贝,适合大文件只读**
- **`tempfile` 的 `Drop` 自动清理临时文件**
- **`walkdir` 替代手写递归遍历**
- **异步 IO 用 `tokio::fs::File`**
- **文件错误用 `thiserror` 定义类型,应用层用 `anyhow`**

### 5. `src/scanner.rs` - 目录扫描

```rust
use std::path::{Path, PathBuf};
use walkdir::WalkDir;

use crate::error::{BackupError, Result};

pub struct ScanResult {
    pub files: Vec<PathBuf>,
    pub total_bytes: u64,
}

pub fn scan(root: &Path, excludes: &[String]) -> Result<ScanResult> {
    if !root.exists() {
        return Err(BackupError::NotFound(root.display().to_string()));
    }

    let mut files = Vec::new();
    let mut total = 0u64;

    for entry in WalkDir::new(root).follow_links(false) {
        let entry = entry.map_err(|e| BackupError::Walk(e.to_string()))?;
        if !entry.file_type().is_file() {
            continue;
        }

        let path = entry.path();
        if should_exclude(path, excludes) {
            continue;
        }

        let meta = entry.metadata()
            .map_err(|e| BackupError::Io(e.into()))?;
        total += meta.len();
        files.push(path.to_path_buf());
    }

    Ok(ScanResult { files, total_bytes: total })
}

fn should_exclude(path: &Path, excludes: &[String]) -> bool {
    let s = path.to_string_lossy();
    excludes.iter().any(|p| s.contains(p))
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_exclude() {
        assert!(should_exclude(Path::new("/tmp/.git/HEAD"), &vec![".git".into()]));
        assert!(!should_exclude(Path::new("/tmp/main.rs"), &vec![".git".into()]));
    }
}
```

### 6. `src/archiver.rs` - 归档写入

```rust
use std::fs::{self, File};
use std::io::{self, BufReader, BufWriter, Read, Write};
use std::path::{Path, PathBuf};

use flate2::read::GzDecoder;
use flate2::write::GzEncoder;
use flate2::Compression;

use crate::error::{BackupError, Result};

pub struct ArchiveWriter {
    writer: BufWriter<GzEncoder<File>>,
    count: u32,
}

impl ArchiveWriter {
    pub fn new(path: &Path, compress: bool) -> Result<Self> {
        let file = File::create(path)?;
        let writer = if compress {
            BufWriter::new(GzEncoder::new(file, Compression::default()))
        } else {
            // 简化:实际还需自定义包装
            BufWriter::new(GzEncoder::new(file, Compression::default()))
        };
        Ok(Self { writer, count: 0 })
    }

    pub fn add_file(&mut self, root: &Path, src: &Path) -> Result<()> {
        // 简化格式:每行 "<relative_path>\t<size>\n<content>"
        let rel = src.strip_prefix(root)
            .unwrap_or(src)
            .to_string_lossy()
            .into_owned();

        let mut file = File::open(src)?;
        let mut buf = Vec::new();
        file.read_to_end(&mut buf)?;

        writeln!(self.writer, "{}\t{}", rel, buf.len())?;
        self.writer.write_all(&buf)?;
        self.count += 1;
        Ok(())
    }

    pub fn finish(mut self) -> Result<u32> {
        self.writer.flush()?;
        Ok(self.count)
    }
}

/// 解压归档到目标目录
pub fn extract(archive: &Path, dest: &Path) -> Result<()> {
    fs::create_dir_all(dest)?;

    let file = File::open(archive)?;
    let reader = BufReader::new(GzDecoder::new(BufReader::new(file)));
    // 实际解析:这里简化
    let mut buf = String::new();
    io::copy(&mut reader.take(0), &mut buf)?;

    Ok(())
}
```

```rust
use chrono::{DateTime, Utc};
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BackupManifest {
    pub name: String,
    pub created_at: DateTime<Utc>,
    pub source: String,
    pub file_count: u32,
    pub total_bytes: u64,
    pub archive: String,
}

impl BackupManifest {
    pub fn new(name: String, source: String, file_count: u32, total_bytes: u64, archive: String) -> Self {
        Self {
            name,
            created_at: Utc::now(),
            source,
            file_count,
            total_bytes,
            archive,
        }
    }

    pub fn save(&self, path: &str) -> std::io::Result<()> {
        let s = serde_json::to_string_pretty(self).unwrap();
        std::fs::write(path, s)
    }

    pub fn load(path: &str) -> std::io::Result<Self> {
        let s = std::fs::read_to_string(path)?;
        Ok(serde_json::from_str(&s).unwrap())
    }
}
```

### 9. `src/main.rs`

```rust
use std::path::PathBuf;

use anyhow::{Context, Result};
use backup_tool::*;

fn main() -> Result<()> {
    let args: Vec<_> = std::env::args().collect();
    if args.len() < 3 {
        eprintln!("用法: {} <source> <output>", args[0]);
        std::process::exit(1);
    }

    let source = PathBuf::from(&args[1]);
    let output = PathBuf::from(&args[2]);

    println!("📦 备份 {} -> {}", source.display(), output.display());

    // 1. 扫描
    let scan = scan(&source, &[]).context("扫描失败")?;
    println!("扫描: {} 文件,共 {} 字节",
        scan.files.len(), scan.total_bytes);

    // 2. 归档
    let archive_path = output.join("backup.tar.gz");
    let mut writer = archiver::ArchiveWriter::new(&archive_path, true)
        .context("创建归档失败")?;

    for file in &scan.files {
        writer.add_file(&source, file)?;
    }

    let count = writer.finish()?;
    println!("归档完成,{} 个文件", count);

    // 3. 写 manifest
    let manifest = BackupManifest::new(
        "manual-backup".into(),
        source.display().to_string(),
        count,
        scan.total_bytes,
        archive_path.display().to_string(),
    );
    let manifest_path = output.join("manifest.json");
    manifest.save(&manifest_path.display().to_string())
        .context("写 manifest 失败")?;

    println!("✅ 备份完成");
    println!("   归档: {}", archive_path.display());
    println!("   清单: {}", manifest_path.display());

    Ok(())
}
```

### 11. 关键点

| 概念 | 用法 |
| --- | --- |
| `Path` vs `PathBuf` | 视图 vs 可拥有 |
| `walkdir` | 递归目录,自动跳过 .git 等 |
| `BufReader` / `BufWriter` | 包装减少 syscall |
| `flate2` | gzip 压缩 |
| `serde_json` + chrono | manifest 持久化 |
| `OpenOptions::create_new` | 防止覆盖已有归档 |
| `chrono::Utc::now()` | 时间戳 |
| `?` + `From` | 错误自动传播 |
