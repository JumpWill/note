# 11 - 错误处理

Lua 错误分两类:**语法错误**(编译期)和**运行时错误**(`error` / 异常)。运行时错误可用 `pcall` 捕获。

## 一、error 与 assert

### 1. error(msg [, level])

```lua
-- 抛出错误
error("something went wrong")

-- 带级别(level 指明错误位置)
// level 1:错误指向调用 error() 的地方
// level 2:错误指向调用 error() 的函数的调用者
// 默认 1
local function inner() error("inner error", 1) end
local function outer() inner() end
outer()
-- inner error: ...
-- stack traceback:
--     inner: ...
--     outer: ...
```

### 2. assert(v [, msg])

```lua
-- assert(v) 等价: v and v or error(...)
-- 当 v 为假时,触发错误

local function divide(a, b)
    assert(type(a) == "number", "a must be number")
    assert(type(b) == "number", "b must be number")
    assert(b ~= 0, "division by zero")
    return a / b
end

print(divide(10, 2))   -- 5
print(divide(10, 0))   -- 触发错误
-- .../main.lua:8: assertion failed!

-- assert 常用于参数校验
local function set_user(name)
    assert(name and #name > 0, "name required")
    _G.user = name
end
```

## 二、pcall - 受保护调用

```lua
-- pcall(f, ...) 调用函数 f,捕获错误
-- 返回 true/false + 返回值/错误信息

local ok, result = pcall(function() error("oops") end)

if ok then
    print("成功:", result)
else
    print("错误:", result)   -- "oops"
end

-- 包装函数参数
local function safe_divide(a, b)
    local ok, result = pcall(function() return divide(a, b) end)
    if not ok then return nil, result end
    return result
end

print(safe_divide(10, 0))  -- nil  "assertion failed!"
```

### 实战:安全执行脚本

```lua
local function safe_dofile(path)
    local f, err = loadfile(path)
    if not f then return false, "load failed: " .. err end

    local ok, result = pcall(f)
    if not ok then return false, "run failed: " .. result end

    return true, result
end

local ok, err = safe_dofile("user_script.lua")
if not ok then
    print("脚本错误:", err)
end
```

## 三、xpcall - 自定义错误处理

```lua
-- xpcall(f, err_handler, ...):可指定错误处理函数

local function err_handler(err)
    print("Caught:", err)
    print(debug.traceback("stack:", 2))
    return err  -- 必须返回字符串
end

local ok, err = xpcall(function()
    error("my error")
end, err_handler)

-- Caught: my error
-- stack: ...
```

### 实战:详细错误日志

```lua
local function with_log(fn, name)
    return function(...)
        local ok, err = xpcall(fn, function(e)
            io.stderr:write(string.format(
                "[%s] %s\n%s\n",
                os.date("%Y-%m-%d %H:%M:%S"),
                name, debug.traceback(e, 2)
            ))
            return e
        end, ...)
        if not ok then return nil, err end
        return ...
    end
end

local logged = with_log(function() error("oops") end, "myFunc")
logged()
-- [2024-01-15 ...] myFunc
-- stack traceback:
--     ...
```

## 四、错误传播

```lua
local function inner()
    -- 重新抛出当前错误
    error("inner failed", 2)
end

local function middle()
    inner()  -- 错误自动传播
end

local function outer()
    middle()
end

local ok, err = pcall(outer)
if not ok then
    print("错误:", err)
end
```

## 五、错误级别(level)

```lua
-- level 参数控制错误的调用栈位置

local function helper()
    error("from helper", 2)  -- level 2:错误指向 helper 的调用者
end

local function caller()
    helper()
end

caller()
-- 错误信息指向 caller (line in caller)
```

## 六、综合实战:Result 模式

```lua
-- result.lua
-- 用 pcall 模拟 Result<T, E>

local M = {}

-- 包装:成功时返回 (true, value),失败返回 (false, err)
function M.try(fn, ...)
    return pcall(fn, ...)
end

-- 创建成功结果
function M.ok(value)
    return true, value
end

-- 创建失败结果
function M.err(error)
    return false, error
end

-- 链式处理
function M.chain(result, fn)
    local ok, val = result
    if not ok then return result end
    return fn(val)
end

-- 使用
local function parse_number(s)
    local n = tonumber(s)
    if not n then return M.err("not a number") end
    return M.ok(n)
end

local result = parse_number("42")
if result == false then
    print("错误")
else
    print("成功:", result)
end

-- 链式
local r1 = parse_number("100")
local r2 = M.chain(r1, function(n)
    if n > 50 then return M.ok("big") end
    return M.err("too small")
end)
```

## 七、调试技巧

### 1. `debug.traceback(msg [, level])`

```lua
local function deep() error("deep error") end
local function middle() deep() end
local function top() middle() end

print(debug.traceback("call from top:"))
-- call from top:
-- stack traceback:
--     [C]: in function 'deep'
--     .../file.lua:8: in function 'middle'
--     .../file.lua:10: in function 'top'
--     .../file.lua:13: in main chunk
```

### 2. `debug.getinfo(fn [, what])`

```lua
local info = debug.getinfo(1, "Sl")
print(info.short_src, info.linedefined, info.what)
```

### 3. `debug.sethook(fn, mask [, count])`

```lua
local function trace(event, line)
    local info = debug.getinfo(2, "Sl")
    print(info.short_src .. ":" .. line .. " " .. event)
end

debug.sethook(trace, "l")  -- 每行调用
-- 你的代码...
debug.sethook(nil)  -- 取消
```

## 八、要点速记

- **运行时错误用 `error(msg, level)` 抛出**
- **`assert(v, msg)`** 当 v 为假时触发错误,常用于参数校验
- **`pcall(f, ...)`** 安全调用,返回 `(ok, result)`
- **`xpcall(f, err_handler, ...)`** 自定义错误处理
- **`debug.traceback`** 输出调用栈
- **`debug.getinfo(2, "Sl")`** 获取调用信息
- **错误级别 level:1 = 当前,2 = 调用者**
- **错误传播:不处理就向上冒泡**
- **没有 try/catch,用 `pcall` 模拟**
