# 07 - 元表(metatable)与面向对象

Lua 没有 class,通过 **metatable**(元表)实现面向对象和运算符重载。

## 一、元表基础

### 1. 设置与获取

```lua
local t = {}

-- 设置元表
setmetatable(t, {__index = {x = 1, y = 2}})

-- 获取元表
local mt = getmetatable(t)

-- 移除元表
setmetatable(t, nil)
```

### 2. `__index` - 字段查找 fallback

```lua
local prototype = {name = "alice", age = 30}

local t = setmetatable({}, {__index = prototype})

print(t.name)    -- "alice"(查不到,在元表 __index 找)
print(t.age)     -- 30
t.x = 10          -- 设置新字段(不影响 prototype)
print(prototype.x)  -- nil
```

### 3. `__newindex` - 字段赋值

```lua
local t = setmetatable({}, {
    __newindex = function(t, key, value)
        print(string.format("设置 %s = %s", key, value))
        rawset(t, key, value)  -- 用 rawset 真正写入,避免递归
    end,
})

t.x = 10  -- 打印:设置 x = 10
```

## 二、运算符重载

### 1. 算术运算

```lua
local set1 = {10, 20, 30}
local set2 = {20, 40, 50}

local mt = {
    __add = function(a, b)
        local result = {}
        local seen = {}
        for _, v in ipairs(a) do seen[v] = true end
        for _, v in ipairs(b) do seen[v] = true end
        for k in pairs(seen) do table.insert(result, k) end
        return result
    end,
}

setmetatable(set1, mt)
local set3 = set1 + set2
-- {10, 20, 30, 40, 50}(集合并集)
```

### 2. 常用元方法

```lua
local mt = {
    __add      = function(a, b) return a + b end,        -- +
    __sub      = function(a, b) return a - b end,        -- -
    __mul      = function(a, b) return a * b end,        -- *
    __div      = function(a, b) return a / b end,        -- /
    __mod      = function(a, b) return a % b end,        -- %
    __pow      = function(a, b) return a ^ b end,        -- ^
    __unm      = function(a) return -a end,             -- 一元 -
    __concat   = function(a, b) return tostring(a) .. tostring(b) end,  -- ..
    __len      = function(a) return #a end,             -- #
    __eq       = function(a, b) return a == b end,       -- ==
    __lt       = function(a, b) return a < b end,        -- <
    __le       = function(a, b) return a <= b end,       -- <=
    __index    = function(t, k) ... end,                -- 读
    __newindex = function(t, k, v) ... end,              -- 写
    __call     = function(t, ...) ... end,               -- 当函数调用
    __tostring = function(t) return "..." end,          -- tostring
    __metatable = ...,                                  -- 保护元表
}
```

### 3. 实战:Vector

```lua
local Vector = {}
Vector.__index = Vector

function Vector.new(x, y)
    return setmetatable({x = x, y = y}, Vector)
end

Vector.__add = function(a, b)
    return Vector.new(a.x + b.x, a.y + b.y)
end

Vector.__sub = function(a, b)
    return Vector.new(a.x - b.x, a.y - b.y)
end

Vector.__mul = function(a, k)
    return Vector.new(a.x * k, a.y * k)
end

Vector.__tostring = function(v)
    return string.format("Vector(%g, %g)", v.x, v.y)
end

local v1 = Vector.new(1, 2)
local v2 = Vector.new(3, 4)
print(v1 + v2)        -- "Vector(4, 6)"
print(v1 * 3)         -- "Vector(3, 6)"
```

## 三、面向对象编程

### 1. 经典模式

```lua
-- Animal.lua
local Animal = {}
Animal.__index = Animal

function Animal.new(name)
    local self = setmetatable({name = name}, Animal)
    return self
end

function Animal:speak()
    return self.name .. " makes a sound"
end

return Animal
```

```lua
-- Dog.lua
local Animal = require("Animal")

local Dog = setmetatable({}, {__index = Animal})
Dog.__index = Dog
setmetatable(Dog, {__index = Animal})  -- 继承

function Dog.new(name, breed)
    local self = setmetatable({}, Dog)
    self.name = name
    self.breed = breed
    return self
end

function Dog:speak()
    return self.name .. " barks"
end

return Dog
```

```lua
-- main.lua
local Animal = require("Animal")
local Dog = require("Dog")

local d = Dog.new("Buddy", "Golden")
print(d:speak())      -- "Buddy barks"(Dog 覆盖)
print(d.name)         -- "Buddy"
```

### 2. 类继承

```lua
-- 设置 Dog 继承 Animal
local Animal = require("Animal")
local Dog = require("Dog")

-- 关键:setmetatable(Dog, {__index = Animal})
-- 查找顺序:Dog 实例 → Dog 类 → Animal 类
```

### 3. 多继承(Mixin)

```lua
-- 用 __index 为函数,组合多个父类
local function search(k, ...)
    for _, parent in ipairs({...}) do
        local v = parent[k]
        if v then return v end
    end
end

local M = {}
M.__index = function(t, k) return search(k, Parent1, Parent2) end
```

## 四、综合实战:简单 OOP 项目

### 1. 项目结构

```text
oop_demo/
├── main.lua
└── src/
    ├── class.lua      # class 系统
    ├── shape.lua      # 父类
    ├── circle.lua     # 子类
    └── rect.lua       # 子类
```

### 2. `src/class.lua` - 简易 class 系统

```lua
local M = {}

function M.new(cls)
    return setmetatable({}, {__index = cls})
end

function M.extends(parent)
    local cls = {}
    for k, v in pairs(parent) do cls[k] = v end

    cls.__index = cls
    setmetatable(cls, {__index = parent})

    function cls:new(...)
        local instance = setmetatable({}, {__index = self})
        if instance.init then instance:init(...) end
        return instance
    end

    return cls
end

function M.isInstance(obj, cls)
    return getmetatable(getmetatable(obj)) == cls
end

return M
```

### 3. `src/shape.lua`

```lua
local class = require("class")

local Shape = class.new()

function Shape:init(name)
    self.name = name
end

function Shape:area()
    return 0
end

function Shape:describe()
    return self.name .. ": area=" .. self:area()
end

return Shape
```

### 4. `src/circle.lua`

```lua
local class = require("class")
local Shape = require("shape")

local Circle = class.extends(Shape)

function Circle:init(name, r)
    self.name = name
    self.r = r
end

function Circle:area()
    return math.pi * self.r * self.r
end

return Circle
```

### 5. `src/rect.lua`

```lua
local class = require("class")
local Shape = require("shape")

local Rect = class.extends(Shape)

function Rect:init(name, w, h)
    self.name = name
    self.w = w
    self.h = h
end

function Rect:area()
    return self.w * self.h
end

return Rect
```

### 6. `main.lua`

```lua
package.path = "./src/?.lua;" .. package.path

local Circle = require("circle")
local Rect = require("rect")

local c = Circle:new("my circle", 2)
print(c:describe())  -- "my circle: area=12.566..."

local r = Rect:new("my rect", 3, 4)
print(r:describe())  -- "my rect: area=12"
```

## 五、要点速记

- **元表是普通 table,定义行为规则**
- **`__index`** 是字段查找 fallback,OOP 核心
- **`__newindex`** 拦截字段赋值
- **`__tostring`** 自定义 `tostring()` / `print()` 输出
- **`__call`** 让 table 像函数一样调用
- **`__concat`** 自定义 `..` 拼接
- **面向对象:`setmetatable + __index`**
- **继承:子类 `__index = 子类`,`子类的 metatable.__index = 父类`**
- **冒号语法 `obj:method()` = `obj.method(obj)` 自动传 self**
- **`rawget` / `rawset` 绕过元方法直接读 / 写**
