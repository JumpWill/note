# 08 - 常用标准库

Lua 标准库覆盖字符串、表、数学、操作系统、IO 等场景。

## 一、string 库

```lua
local s = "Hello, World"

-- 长度 / 子串
print(string.len(s))                  -- 12
print(string.sub(s, 1, 5))           -- "Hello"
print(string.sub(s, -5))             -- "World"(负数从末尾数)

-- 大小写
print(string.upper(s))                -- "HELLO, WORLD"
print(string.lower(s))                -- "hello, world"

-- 查找
print(string.find(s, "World"))        -- 8  12  (起始、结束)
print(string.find(s, "world"))        -- nil
print(string.find(s, "world", 1, true))  -- 忽略大小写

-- 反向查找
print(string.match(s, "(%w+)"))       -- "Hello"(第一个匹配)
print(string.match(s, "Hello, (%w+)")) -- "Hello", "World"(捕获)

-- 全局替换
print(string.gsub(s, "l", "L"))       -- "HeLLo, World"  2

-- 重复 / 反转
print(string.rep("ab", 3))            -- "ababab"
print(string.reverse("hello"))         -- "olleh"

-- 字节 / 字符
print(string.byte("A"))               -- 65
print(string.char(65))                -- "A"

-- 格式化
print(string.format("pi=%.2f", math.pi))    -- "pi=3.14"
print(string.format("%05d", 42))             -- "00042"

-- 模式字符(常用)
-- %a 字母   %d 数字   %s 空白
-- %w 字母数字下划线
-- %u 大写   %l 小写   %p 标点
-- %c 控制字符
-- %.+  1+ 个字符(贪婪)
-- %-*  0+ 个字符(非贪婪)
```

### 实战:字符串工具

```lua
local M = {}

function M.split(s, sep)
    local result = {}
    sep = sep or "%s+"
    for part in string.gmatch(s, "([^" .. sep .. "]+)") do
        table.insert(result, part)
    end
    return result
end

function M.starts_with(s, prefix)
    return s:sub(1, #prefix) == prefix
end

function M.ends_with(s, suffix)
    return suffix == "" or s:sub(-#suffix) == suffix
end

function M.trim(s)
    return s:match("^%s*(.-)%s*$") or s
end

function M.replace_all(s, from, to)
    local result = s:gsub(from, to)
    -- gsub 返回 (string, count)
    return result
end

return M
```

## 二、table 库

```lua
local t = {3, 1, 4, 1, 5, 9, 2, 6}

-- 增删改
table.insert(t, 7)              -- 末尾追加
table.insert(t, 1, 99)          -- 位置 1 插入
table.remove(t)                  -- 末尾删除
table.remove(t, 1)               -- 位置 1 删除

-- 排序
table.sort(t)
table.sort(t, function(a, b) return a > b end)

-- 连接
print(table.concat(t, ","))     -- "1,1,2,3,4,5,6,7"

-- 解包
local a, b, c = table.unpack(t)

-- 移动(Lua 5.3+)
local src = {1, 2, 3}
local dst = {}
table.move(src, 1, 3, 1, dst)

-- 最大值
print(math.max(table.unpack(t)))
```

## 三、math 库

```lua
-- 常量
print(math.pi)        -- 3.1415926535898
print(math.huge)      -- inf
print(math.mininteger)
print(math.maxinteger)

-- 基础
print(math.abs(-5))                -- 5
print(math.max(1, 2, 3))           -- 3
print(math.min(1, 2, 3))           -- 1
print(math.ceil(3.14))             -- 4
print(math.floor(3.99))            -- 3
print(math.modf(3.14))             -- 3.0  0.14

-- 取整 / 取余
print(math.floor(7 / 2))            -- 3
print(7 % 2)                         -- 1

-- 三角
print(math.sin(math.pi / 2))        -- 1.0
print(math.cos(0))                  -- 1.0
print(math.tan(math.pi / 4))        -- 0.99999...
print(math.rad(180))                -- π
print(math.deg(math.pi))            -- 180

-- 指数 / 对数
print(math.exp(1))                  -- 2.71828...
print(math.log(100, 10))            -- 2
print(math.sqrt(2))                 -- 1.41421...

-- 随机
math.randomseed(os.time())
print(math.random())                -- [0, 1)
print(math.random(100))             -- [1, 100]
print(math.random(50, 100))         -- [50, 100]
```

## 四、os 库

```lua
-- 时间
print(os.time())                    -- 当前时间戳(秒)
print(os.time({year = 2024, month = 1, day = 15}))   -- 自定义时间

-- 日期格式化
print(os.date())                    -- "2024-01-15 10:30:00"
print(os.date("%Y-%m-%d"))          -- "2024-01-15"
print(os.date("%H:%M:%S"))          -- "10:30:00"

-- 计时
local start = os.clock()
-- ... 耗时操作
print(os.clock() - start)

-- 环境
print(os.getenv("HOME"))
os.execute("ls")                    -- 执行 shell(危险)
os.exit(0)                          -- 退出
os.rename("a", "b")                 -- 重命名/移动
os.remove("a")                      -- 删除
os.tmpname()                        -- 临时文件名

-- 时区
print(os.difftime(os.time(), os.time()))  -- 时间差(秒)
```

### 4. table vs string 性能

```lua
-- 字符串拼接
-- ❌ 循环中 .. 拼接
local s = ""
for i = 1, 10000 do s = s .. "x" end     -- O(n²)

-- ✅ 用 table.concat
local t = {}
for i = 1, 10000 do t[i] = "x" end
local s = table.concat(t)                  -- O(n)
```

## 七、综合实战:小工具集

```lua
-- utils.lua
local M = {}

-- 字符串
function M.starts_with(s, prefix)
    return s:sub(1, #prefix) == prefix
end

function M.contains(s, substr)
    return s:find(substr, 1, true) ~= nil
end

function M.uuid()
    local hex = "0123456789abcdef"
    local uuid = ""
    for i = 1, 32 do
        local r = math.random(1, 16)
        uuid = uuid .. hex:sub(r, r)
        if i == 8 or i == 12 or i == 16 or i == 20 then
            uuid = uuid .. "-"
        end
    end
    return uuid
end

-- 数学
function M.clamp(v, lo, hi)
    if v < lo then return lo end
    if v > hi then return hi end
    return v
end

function M.round(v, n)
    local p = 10 ^ (n or 0)
    return math.floor(v * p + 0.5) / p
end

-- 表
function M.deep_copy(t)
    if type(t) ~= "table" then return t end
    local out = {}
    for k, v in pairs(t) do
        out[k] = M.deep_copy(v)
    end
    return out
end

function M.deep_equal(a, b)
    if type(a) ~= type(b) then return false end
    if type(a) ~= "table" then return a == b end
    for k, v in pairs(a) do
        if not M.deep_equal(v, b[k]) then return false end
    end
    for k in pairs(b) do
        if a[k] == nil then return false end
    end
    return true
end

return M
```
