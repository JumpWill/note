# 04 - 函数

Lua 函数是**第一类值**(first-class),可以赋值给变量、作为参数、作为返回值。

## 一、函数定义

### 1. 基本语法

```lua
-- 命名函数
function greet(name)
    return "Hello, " .. name
end

print(greet("Lua"))

-- 匿名函数(赋给变量)
local greet = function(name)
    return "Hi, " .. name
end

-- 简写(Lua 推荐)
local function greet(name)
    return "Hi, " .. name
end
```

### 2. 立即执行(IIFE)

```lua
local result = (function(x, y)
    return x + y
end)(10, 20)

print(result)  -- 30
```

## 二、参数

### 1. 普通参数

```lua
local function add(a, b)
    return a + b
end

print(add(1, 2))  -- 3
```

### 2. 参数不足/过多

```lua
local function f(a, b, c)
    return a, b, c
end

-- 不足:补 nil
print(f(1))            -- 1  nil  nil

-- 过多:丢弃
print(f(1, 2, 3, 4))  -- 1  2  3
```

### 3. 可变长参数 `...`

```lua
local function add(...)
    local sum = 0
    for _, v in ipairs({...}) do  -- {...} 转 table
        sum = sum + v
    end
    return sum
end

print(add(1, 2, 3, 4))  -- 10

-- select 访问特定位置
local function f(...)
    print(select("#", ...))     -- 参数个数(含 nil)
    print(select(2, ...))      -- 从第 2 个开始
end

f("a", "b", "c")  -- 3  b  c

-- 最后一个表达式
local function last(...)
    return select("#", ...) > 0 and select("#", ...) or nil
end
```

### 4. 传入 table(引用传递)

```lua
-- table 是引用类型,函数内修改会影响外部
local function modify(t)
    t.name = "modified"   -- 会改外部
end

local data = {name = "original"}
modify(data)
print(data.name)  -- "modified"

-- 防御:浅拷贝
local function safe_modify(t)
    local copy = {}
    for k, v in pairs(t) do copy[k] = v end
    copy.name = "modified"
    return copy
end

local data = {name = "original"}
local new_data = safe_modify(data)
print(data.name)     -- "original"
print(new_data.name)  -- "modified"
```

## 三、返回值

### 1. 多返回值

```lua
local function divmod(a, b)
    return a // b, a % b
end

local q, r = divmod(10, 3)
print(q, r)  -- 3  1
```

### 2. return 后无值:返回 nil

```lua
local function f()
    return
end

print(f())  -- nil
```

### 3. 多返回值边界

```lua
-- 当表达式中出现多个值,只取第一个(除非在最后)
local function f() return 1, 2, 3 end

print(f())       -- 1  2  3(单独调用取全部)
print((f()))     -- 1(被括号包裹,只取一个)
local x = f()    -- 1(赋值只取一个)
local a, b, c = f()  -- 全部

-- unpack
print(table.unpack({10, 20, 30}))  -- 10  20  30
```

## 四、闭包

```lua
-- 闭包 = 函数 + 引用环境
local function make_counter(start)
    local count = start or 0
    return function()
        count = count + 1
        return count
    end
end

local c1 = make_counter(10)
print(c1())  -- 11
print(c1())  -- 12

local c2 = make_counter(100)
print(c2())  -- 101(独立的 count)
```

### 应用:装饰器

```lua
local function with_logging(fn, name)
    name = name or fn.__name or "fn"
    return function(...)
        print(name .. " called")
        local result = fn(...)
        print(name .. " returned")
        return result
    end
end

local function add(a, b) return a + b end
local logged_add = with_logging(add, "add")
print(logged_add(1, 2))
```

## 五、高阶函数

### 1. 函数作为参数

```lua
local function map(t, fn)
    local out = {}
    for i, v in ipairs(t) do
        out[i] = fn(v)
    end
    return out
end

local squares = map({1, 2, 3}, function(x) return x * x end)
-- {1, 4, 9}
```

### 2. 函数作为返回值(柯里化)

```lua
local function add(a)
    return function(b)
        return a + b
    end
end

local add5 = add(5)
print(add5(3))   -- 8
```

## 六、函数调用

```lua
local function f(a, b, c) return a + b + c end

-- 标准调用
f(1, 2, 3)

-- 表调用(冒号语法,自动传 self)
local obj = {x = 10}
function obj:method(y)
    return self.x + y
end

-- obj:method(5) 等价 obj.method(obj, 5)
print(obj:method(5))  -- 15

-- 函数也是表,可以存到字段
local M = {}
function M.hello() print("hi") end
M.hello()
```

## 七、综合实战:工具函数库

```lua
-- utils.lua
local M = {}

-- 函数组合
function M.compose(...)
    local fns = {...}
    return function(x)
        for i = #fns, 1, -1 do
            x = fns[i](x)
        end
        return x
    end
end

-- 用法
local f = M.compose(
    function(x) return x * 2 end,
    function(x) return x + 1 end,
    function(x) return x .. "!" end
)
print(f(3))   -- "7!"   (3*2+1=7,然后 "7!")

-- 记忆化
function M.memoize(fn)
    local cache = {}
    return function(x)
        if cache[x] == nil then
            cache[x] = fn(x)
        end
        return cache[x]
    end
end

local slow = function(n)
    -- 模拟耗时计算
    return n * n
end
local fast = M.memoize(slow)

print(fast(5))  -- 计算 25
print(fast(5))  -- 缓存命中

-- 节流(throttle)
function M.throttle(fn, ms)
    local last_call = 0
    return function(...)
        local now = os.clock() * 1000
        if now - last_call >= ms then
            last_call = now
            return fn(...)
        end
    end
end

-- once
function M.once(fn)
    local called = false
    local result
    return function(...)
        if not called then
            called = true
            result = fn(...)
        end
        return result
    end
end

return M
```

## 八、要点速记

- **函数是第一类值:可赋值、传参、返回、存表**
- **定义:`function name() ... end` 或 `local f = function() end`**
- **参数不足补 nil,过多丢弃**
- **可变参数 `...`,`select("#", ...)` 取个数,`{...}` 转表**
- **多返回值:`return a, b, c`**
- **闭包:函数 + 引用环境,变量独立**
- **冒号语法 `obj:method()` 自动传 `self`**
- **`table.unpack` 解包 table 为多值**
- **`goto` 标签 + `::label::` 实现跳转**
- **没有运算符重载(用 metatable)**
