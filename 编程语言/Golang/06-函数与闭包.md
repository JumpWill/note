# 06 - 函数与闭包

## 一、函数基础

### 1. 定义与调用

```go
// 无参无返回值
func SayHello() {
    fmt.Println("Hello")
}

// 带参数
func Add(a, b int) int {
    return a + b
}

// 多返回值
func Divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("division by zero")
    }
    return a / b, nil
}

// 命名返回值
func Split(sum int) (x, y int) {
    x = sum * 4 / 9
    y = sum - x
    return  // 裸 return,返回命名返回值
}
```

### 2. 函数是第一类公民

```go
// 赋值给变量
f := Add
fmt.Println(f(1, 2))

// 作为参数
func Apply(nums []int, f func(int) int) []int {
    out := make([]int, len(nums))
    for i, v := range nums {
        out[i] = f(v)
    }
    return out
}

doubled := Apply([]int{1, 2, 3}, func(x int) int { return x * 2 })

// 作为返回值
func MakeAdder(n int) func(int) int {
    return func(x int) int {
        return x + n
    }
}
add5 := MakeAdder(5)
fmt.Println(add5(10))  // 15
```

### 3. 函数类型

```go
type Handler func(http.ResponseWriter, *http.Request)

func Middleware(h Handler) Handler {
    return func(w http.ResponseWriter, r *http.Request) {
        log.Println("before")
        h(w, r)
        log.Println("after")
    }
}
```

### 4. 变参函数

```go
func Sum(nums ...int) int {
    total := 0
    for _, n := range nums {
        total += n
    }
    return total
}

Sum(1, 2, 3)            // 6
Sum([]int{1, 2, 3}...)  // 6(展开 slice)
```

## 二、匿名函数与闭包

### 1. 匿名函数

```go
// 立即执行
func() {
    fmt.Println("anon")
}()

// 赋值给变量
greet := func(name string) {
    fmt.Println("Hi,", name)
}
greet("Alice")
```

### 2. 闭包

闭包 = 函数 + 引用环境

```go
func Counter() func() int {
    n := 0
    return func() int {
        n++
        return n
    }
}

c := Counter()
fmt.Println(c())  // 1
fmt.Println(c())  // 2
fmt.Println(c())  // 3
```

### 3. 闭包陷阱:循环变量捕获

```go
// ❌ Go 1.21 之前:所有闭包共享同一个 i(经典 bug)
funcs := []func(){}
for i := 0; i < 3; i++ {
    funcs = append(funcs, func() {
        fmt.Println(i)
    })
}
for _, f := range funcs {
    f()                  // 3 3 3(都引用同一个 i,循环结束后 i=3)
}

// ✅ 修复:循环里重新赋值
for i := 0; i < 3; i++ {
    i := i              // 新变量(Go 1.21 之前)
    funcs = append(funcs, func() {
        fmt.Println(i)
    })
}

// ✅ Go 1.22+:循环变量每轮独立
for i := 0; i < 3; i++ {
    funcs = append(funcs, func() {
        fmt.Println(i)  // 0 1 2
    })
}
```

### 4. 闭包应用

```go
// 累加器
func MakeAccumulator(init int) func(int) int {
    sum := init
    return func(x int) int {
        sum += x
        return sum
    }
}

// 装饰器模式
func WithLogging(fn func()) func() {
    return func() {
        start := time.Now()
        defer func() {
            log.Printf("took %v", time.Since(start))
        }()
        fn()
    }
}

// 函数柯里化
func Curry(f func(int, int) int) func(int) func(int) int {
    return func(a int) func(int) int {
        return func(b int) int {
            return f(a, b)
        }
    }
}
```

## 三、defer

`defer` 在**函数返回前**执行,无论正常返回还是 panic。常用于资源清理。

### 1. 基本用法

```go
func readFile(path string) ([]byte, error) {
    f, err := os.Open(path)
    if err != nil {
        return nil, err
    }
    defer f.Close()       // 函数返回前关闭

    return io.ReadAll(f)
}
```

### 2. defer 顺序:LIFO

```go
func main() {
    defer fmt.Println(1)
    defer fmt.Println(2)
    defer fmt.Println(3)
}
// 输出:3 2 1(后进先出)
```

### 3. defer 闭包参数

```go
// ❌ 闭包捕获变量(推迟到执行时求值)
for i := 0; i < 3; i++ {
    defer func() {
        fmt.Println(i)
    }()
}
// 输出:3 3 3

// ✅ 立即求值:作为参数传入
for i := 0; i < 3; i++ {
    defer func(n int) {
        fmt.Println(n)
    }(i)
}
// 输出:2 1 0
```

### 4. defer 修改返回值

```go
func Compute() (result int) {
    defer func() {
        result *= 2       // 可以修改命名返回值
    }()
    return 10              // 实际返回 20
}
```

### 5. defer 用法汇总

```go
// 资源清理(最常用)
defer file.Close()
defer rows.Close()
defer tx.Rollback()

// 解锁
mu.Lock()
defer mu.Unlock()

// 计时
start := time.Now()
defer func() {
    log.Printf("took %v", time.Since(start))
}()

// recover panic(见下文)
defer func() {
    if r := recover(); r != nil {
        log.Println("recovered:", r)
    }
}()
```

## 四、错误处理

### 1. error 接口

```go
type error interface {
    Error() string
}
```

### 2. 自定义错误

```go
// errors.New 创建
var ErrNotFound = errors.New("not found")

// fmt.Errorf 格式化
err := fmt.Errorf("user %d not found", id)

// 自定义类型
type ValidationError struct {
    Field string
    Msg   string
}

func (e *ValidationError) Error() string {
    return fmt.Sprintf("invalid %s: %s", e.Field, e.Msg)
}
```

### 3. 错误传播

```go
// 显式 return
data, err := readFile("data.txt")
if err != nil {
    return fmt.Errorf("read failed: %w", err)   // %w 包装,保留原 error
}

// 检查错误链
if errors.Is(err, ErrNotFound) {
    // ...
}

// 取特定类型
var ve *ValidationError
if errors.As(err, &ve) {
    fmt.Println("field:", ve.Field)
}
```

### 4. panic 与 recover

```go
// panic 立即终止当前 goroutine
func main() {
    defer func() {
        if r := recover(); r != nil {
            fmt.Println("recovered:", r)
        }
    }()
    panic("oops")
}
```

### 5. 错误处理风格

```go
// 1. 命令式风格(Go 习惯)
if err := do(); err != nil {
    return err
}

// 2. 早期返回(guard clause)
func process(data []byte) error {
    if len(data) == 0 {
        return errors.New("empty data")
    }
    // ...
}

// 3. 哨兵错误(可比较)
var ErrEmpty = errors.New("empty")

if err == ErrEmpty { ... }
```

## 六、综合实战:配置加载器

```go
package config

import (
    "errors"
    "fmt"
    "os"
    "strconv"
    "time"
)

type Config struct {
    Host    string
    Port    int
    Timeout time.Duration
    Debug   bool
}

var (
    ErrMissing   = errors.New("missing required field")
    ErrInvalid   = errors.New("invalid value")
)

func Load() (*Config, error) {
    cfg := &Config{
        Host:    "localhost",
        Port:    8080,
        Timeout: 30 * time.Second,
        Debug:   false,
    }

    if h := os.Getenv("APP_HOST"); h != "" {
        cfg.Host = h
    }

    if p := os.Getenv("APP_PORT"); p != "" {
        port, err := strconv.Atoi(p)
        if err != nil {
            return nil, fmt.Errorf("%w: APP_PORT=%q: %v", ErrInvalid, p, err)
        }
        if port < 1 || port > 65535 {
            return nil, fmt.Errorf("%w: port %d out of range", ErrInvalid, port)
        }
        cfg.Port = port
    }

    if d := os.Getenv("APP_DEBUG"); d != "" {
        cfg.Debug = d == "true"
    }

    return cfg, nil
}

func (c *Config) Addr() string {
    return fmt.Sprintf("%s:%d", c.Host, c.Port)
}
```

## 七、要点速记

- **多返回值:`(result, error)` 是惯例**
- **命名返回值 + 裸 `return`,可省略**
- **`func(...) func(...)` 高阶函数**
- **`defer` LIFO 顺序,用于资源清理**
- **闭包捕获外部变量;Go 1.22+ 循环变量每轮独立**
- **错误是值,显式传播,`%w` 包装保留链**
- **`errors.Is` / `errors.As` 检查错误链**
- **`panic` 仅用于不可恢复错误,`recover` 仅在 defer 中有效**
- **哨兵错误 `var ErrXxx = errors.New(...)` 用于相等比较**
- **`min`/`max` Go 1.21+ 内建**
- **`new(T)` 返回零值指针,`make` 用于 slice/map/channel**

### 4. `retry.go` - 闭包 + 错误传播

```go
package fnutil

import (
    "context"
    "errors"
    "time"
)

// Retry 在 fn 失败时重试,指数退避
func Retry(ctx context.Context, max int, base time.Duration, fn func() error) error {
    var err error
    for i := 0; i < max; i++ {
        if err = fn(); err == nil {
            return nil
        }

        // 检查 context
        if ctx.Err() != nil {
            return ctx.Err()
        }

        // 最后一次不等
        if i == max-1 {
            break
        }

        // 指数退避:base, 2*base, 4*base, ...
        delay := base * time.Duration(1<<i)

        select {
        case <-time.After(delay):
        case <-ctx.Done():
            return ctx.Err()
        }
    }

    return errors.Join(ErrTooManyRetries, err)
}

// 示例
func RetryExample() error {
    return Retry(context.Background(), 3, 100*time.Millisecond, func() error {
        // 实际 IO 调用
        return nil
    })
}
```

### 5. `once.go` - 带超时的 Once

```go
package fnutil

import (
    "sync"
    "sync/atomic"
    "time"
)

// TimeoutOnce 只执行一次,但带超时保护
type TimeoutOnce struct {
    done    atomic.Bool
    mu      sync.Mutex
    timeout time.Duration
}

func NewTimeoutOnce(timeout time.Duration) *TimeoutOnce {
    return &TimeoutOnce{timeout: timeout}
}

func (o *TimeoutOnce) Do(f func()) {
    if o.done.Load() {
        return
    }
    o.mu.Lock()
    defer o.mu.Unlock()
    if o.done.Load() {
        return
    }
    done := make(chan struct{})
    go func() {
        f()
        close(done)
    }()
    select {
    case <-done:
        o.done.Store(true)
    case <-time.After(o.timeout):
        // 超时,下次重试
    }
}
```

### 6. `memoize.go` - 记忆化

```go
package fnutil

import "sync"

// Memoize 缓存函数结果
type Memoize[T, U any] struct {
    fn   func(T) (U, error)
    mu   sync.Mutex
    cache map[T]result[U]
}

type result[U any] struct {
    val U
    err error
}

func NewMemoize[T comparable, U any](fn func(T) (U, error)) *Memoize[T, U] {
    return &Memoize[T, U]{
        fn:    fn,
        cache: make(map[T]result[U]),
    }
}

func (m *Memoize[T, U]) Get(key T) (U, error) {
    m.mu.Lock()
    defer m.mu.Unlock()

    if r, ok := m.cache[key]; ok {
        return r.val, r.err
    }

    val, err := m.fn(key)
    m.cache[key] = result[U]{val, err}
    return val, err
}
```

### 8. `fnutil_test.go` - 测试

package fnutil

import (
    "context"
    "errors"
    "testing"
    "time"
)

var ErrTransient = errors.New("transient")

func TestRetry(t *testing.T) {
    var attempts int
    err := Retry(context.Background(), 3, time.Millisecond, func() error {
        attempts++
        if attempts < 2 {
            return ErrTransient
        }
        return nil
    })
    if err != nil {
        t.Errorf("want nil, got %v", err)
    }
    if attempts != 2 {
        t.Errorf("want 2 attempts, got %d", attempts)
    }
}

func TestRetryContextCancel(t *testing.T) {
    ctx, cancel := context.WithCancel(context.Background())
    cancel()    // 立即取消

    err := Retry(ctx, 5, time.Second, func() error {
        return ErrTransient
    })
    if !errors.Is(err, context.Canceled) {
        t.Errorf("want context.Canceled, got %v", err)
    }
}

func TestMemoize(t *testing.T) {
    calls := 0
    m := NewMemoize(func(x int) (int, error) {
        calls++
        return x * 2, nil
    })

    for i := 0; i < 3; i++ {
        _, _ = m.Get(5)
    }
    if calls != 1 {
        t.Errorf("want 1 call, got %d", calls)
    }
}

func TestTrace(t *testing.T) {
    defer Trace("test")()
    time.Sleep(time.Millisecond)
    // 日志会显示耗时
}

### 9. 关键点

| 概念 | 用法 |
| --- | --- |
| 闭包 + 错误 | `Retry` 接受返回 error 的函数 |
| 指数退避 | `1<<i` 翻倍 |
| 上下文取消 | `select` 监听 `ctx.Done()` |
| 错误包装 | `errors.Join(ErrTooManyRetries, err)` |
| 泛型 | `Memoize[T, U any]` |
| atomic + mutex | `TimeoutOnce` 双检查 |
| defer 计时 | `Trace()` 返回闭包 |
| 记忆化 | 函数结果缓存 |
