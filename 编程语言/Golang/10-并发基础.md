# 10 - 并发基础:goroutine 与 channel

Go 的并发核心是 **goroutine**(轻量协程)+ **channel**(通信管道)。

## 一、goroutine

### 1. 创建

```go
go func() {
    fmt.Println("hello from goroutine")
}()

// 带参数
go func(msg string) {
    fmt.Println(msg)
}("hi")

// 启动多个
for i := 0; i < 5; i++ {
    go func(n int) {
        fmt.Println(n)
    }(i)
}
```

### 2. 主协程退出 = 全部退出

```go
func main() {
    go func() {
        time.Sleep(time.Second)
        fmt.Println("子协程")  // 可能来不及打印
    }()
    // main 立即返回,程序结束
}
```

### 3. 同步:WaitGroup

```go
var wg sync.WaitGroup

for i := 0; i < 5; i++ {
    wg.Add(1)
    go func(n int) {
        defer wg.Done()
        fmt.Println(n)
    }(i)
}

wg.Wait()        // 等待所有计数器归零
```

## 二、channel

### 1. 创建

```go
ch := make(chan int)       // 无缓冲
ch := make(chan int, 5)    // 缓冲 5 个元素
```

### 2. 发送与接收

```go
ch <- 42       // 发送
v := <-ch      // 接收
v, ok := <-ch  // 检查 channel 是否关闭
```

### 3. 关闭

```go
close(ch)

// 关闭后:
// - 接收:剩余元素可继续读,读完返回零值 + ok=false
// - 发送:panic: send on closed channel
// - range:自动退出
```

### 4. 无缓冲 channel(同步)

```go
ch := make(chan int)

// 发送会阻塞,直到有接收方
go func() {
    ch <- 42       // 等下面那行准备好
}()

v := <-ch          // 收到 42
fmt.Println(v)
```

### 5. 有缓冲 channel(异步)

```go
ch := make(chan int, 3)

ch <- 1        // 不阻塞(缓冲未满)
ch <- 2
ch <- 3
// ch <- 4     // 阻塞(缓冲已满,直到有人读)

// len() 当前元素数
// cap() 容量
fmt.Println(len(ch), cap(ch))  // 3 3
```

### 6. 单向 channel

```go
// 只写
func producer(out chan<- int) {
    for i := 0; i < 10; i++ {
        out <- i
    }
    close(out)
}

// 只读
func consumer(in <-chan int) {
    for v := range in {
        fmt.Println(v)
    }
}

ch := make(chan int, 5)
go producer(ch)
consumer(ch)
```

### 7. range channel

```go
ch := make(chan int, 3)
ch <- 1
ch <- 2
ch <- 3
close(ch)

for v := range ch {
    fmt.Println(v)   // 1 2 3(channel 关闭后退出)
}
```

## 三、select

`select` 同时等待多个 channel,谁先就绪就用谁。

### 1. 基本用法

```go
select {
case v := <-ch1:
    fmt.Println("ch1:", v)
case v := <-ch2:
    fmt.Println("ch2:", v)
case ch3 <- 42:
    fmt.Println("sent to ch3")
case <-time.After(time.Second):
    fmt.Println("timeout")
default:
    fmt.Println("no channel ready")
}
```

### 2. 超时模式

```go
select {
case res := <-ch:
    return res, nil
case <-time.After(5 * time.Second):
    return nil, errors.New("timeout")
}
```

### 3. 非阻塞模式(default)

```go
select {
case v := <-ch:
    fmt.Println("got:", v)
default:
    fmt.Println("not ready")
}
```

### 4. 退出信号(经典模式)

```go
func worker(done <-chan struct{}, jobs <-chan int, results chan<- int) {
    for {
        select {
        case <-done:
            return
        case j := <-jobs:
            results <- j * 2
        }
    }
}

done := make(chan struct{})
go func() {
    time.Sleep(time.Second)
    close(done)   // 通知退出
}()
```

## 四、常用并发模式

### 1. 超时控制

```go
func fetch(ctx context.Context) (string, error) {
    ch := make(chan string, 1)
    go func() {
        // 模拟 IO
        time.Sleep(500 * time.Millisecond)
        ch <- "result"
    }()

    select {
    case r := <-ch:
        return r, nil
    case <-ctx.Done():
        return "", ctx.Err()
    }
}

ctx, cancel := context.WithTimeout(context.Background(), 100*time.Millisecond)
defer cancel()

result, err := fetch(ctx)
fmt.Println(result, err)  // "" context deadline exceeded
```

### 2. 工作池(worker pool)

```go
func worker(id int, jobs <-chan int, results chan<- int) {
    for j := range jobs {
        fmt.Printf("worker %d processing job %d\n", id, j)
        time.Sleep(time.Second)   // 模拟工作
        results <- j * 2
    }
}

func main() {
    jobs := make(chan int, 100)
    results := make(chan int, 100)

    // 启动 3 个 worker
    for w := 1; w <= 3; w++ {
        go worker(w, jobs, results)
    }

    // 投递 9 个任务
    for j := 1; j <= 9; j++ {
        jobs <- j
    }
    close(jobs)

    // 收集结果
    for i := 0; i < 9; i++ {
        <-results
    }
}
```

### 3. 并行计算

```go
func process(n int) int {
    time.Sleep(time.Duration(n) * 100 * time.Millisecond)
    return n * n
}

func main() {
    nums := []int{1, 2, 3, 4, 5}
    results := make([]int, len(nums))

    var wg sync.WaitGroup
    for i, n := range nums {
        wg.Add(1)
        go func(i, n int) {
            defer wg.Done()
            results[i] = process(n)
        }(i, n)
    }
    wg.Wait()

    fmt.Println(results)
}
```

### 4. 扇出扇入(fan-out / fan-in)

```go
func gen(nums ...int) <-chan int {
    out := make(chan int)
    go func() {
        defer close(out)
        for _, n := range nums {
            out <- n
        }
    }()
    return out
}

func sq(in <-chan int) <-chan int {
    out := make(chan int)
    go func() {
        defer close(out)
        for n := range in {
            out <- n * n
        }
    }()
    return out
}

func main() {
    in := gen(1, 2, 3, 4, 5)
    ch1 := sq(in)
    ch2 := sq(in)

    // 合并 ch1, ch2
    for n := range merge(ch1, ch2) {
        fmt.Println(n)   // 1 4 9 16 25(乱序)
    }
}

func merge(cs ...<-chan int) <-chan int {
    out := make(chan int)
    var wg sync.WaitGroup
    for _, c := range cs {
        wg.Add(1)
        go func(c <-chan int) {
            defer wg.Done()
            for n := range c {
                out <- n
            }
        }(c)
    }
    go func() {
        wg.Wait()
        close(out)
    }()
    return out
}
```

## 五、context

### 1. 基础

```go
ctx := context.Background()                  // 根 context
ctx, cancel := context.WithCancel(ctx)        // 可取消
defer cancel()

ctx, cancel = context.WithTimeout(ctx, 5*time.Second)  // 超时
ctx, cancel = context.WithDeadline(ctx, time.Now().Add(time.Hour))
ctx = context.WithValue(ctx, "key", "value")   // 携带值
```

### 2. 跨 goroutine 取消

```go
func main() {
    ctx, cancel := context.WithCancel(context.Background())

    go func() {
        for {
            select {
            case <-ctx.Done():
                fmt.Println("cancelled")
                return
            case <-time.After(time.Second):
                fmt.Println("tick")
            }
        }
    }()

    time.Sleep(3 * time.Second)
    cancel()    // 取消
    time.Sleep(time.Second)
}
```

### 3. 带值的 context

```go
type ctxKey string

const userIDKey ctxKey = "userID"

ctx := context.WithValue(ctx, userIDKey, 42)

// 取值
func GetUser(ctx context.Context) (int, bool) {
    id, ok := ctx.Value(userIDKey).(int)
    return id, ok
}
```

## 六、综合实战:并发下载器

```go
package main

import (
    "fmt"
    "io"
    "net/http"
    "os"
    "sync"
)

type Result struct {
    URL  string
    Size int64
    Err  error
}

func download(url string) Result {
    resp, err := http.Get(url)
    if err != nil {
        return Result{URL: url, Err: err}
    }
    defer resp.Body.Close()

    n, err := io.Copy(io.Discard, resp.Body)
    return Result{URL: url, Size: n, Err: err}
}

func main() {
    urls := []string{
        "https://golang.org",
        "https://github.com",
        "https://play.golang.org",
        "https://godoc.org",
    }

    results := make(chan Result, len(urls))
    var wg sync.WaitGroup

    for _, url := range urls {
        wg.Add(1)
        go func(u string) {
            defer wg.Done()
            results <- download(u)
        }(url)
    }

    go func() {
        wg.Wait()
        close(results)
    }()

    for r := range results {
        if r.Err != nil {
            fmt.Printf("%s: error %v\n", r.URL, r.Err)
        } else {
            fmt.Printf("%s: %d bytes\n", r.URL, r.Size)
        }
    }
}
```

## 七、要点速记

- **`go func()` 启动 goroutine,主 goroutine 退出则程序退出**
- **`sync.WaitGroup` 等待多个 goroutine**
- **channel 默认无缓冲(同步),`make(ch, n)` 有缓冲**
- **关闭后发送会 panic,接收返回零值 + ok=false**
- **`range channel` 直到 channel 关闭**
- **单向 channel:`chan<-` 只写 / `<-chan` 只读,用于约束接口**
- **`select` 多 channel 等待,`default` 实现非阻塞**
- **超时模式:`select { case <-time.After(): ... }`**
- **`context` 跨 goroutine 传递取消信号和值**
- **关闭原则:发送方关闭,接收方用 `v, ok := <-ch` 或 `range` 检测**
