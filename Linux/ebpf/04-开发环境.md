# 04 - eBPF 开发环境

## 工具链全景

```
                你的 eBPF 程序
                       │
                       ▼
        ┌──────────────┴──────────────┐
        │                             │
       C 语言                    bpftrace 脚本
        │                             │
        ▼                             │
  LLVM/Clang 编译                   bpftrace
        │                             │
        ▼                             │
   eBPF ELF (.o)                  内部 JIT
        │                             │
        ▼                             │
   ┌────┴─────┐                       │
   │ libbpf   │                       │
   │ bpftool  │───────────────────────┘
   └────┬─────┘
        │
        ▼
    内核加载
```

## 选型决策

| 需求 | 推荐 |
|------|------|
| 临时一行命令排查 | bpftrace |
| 一键性能工具 | bcc 工具集 |

| 容器/k8s 网络 | Cilium |

## 1. libbpf（推荐生产路径）

### 安装

```bash
# Ubuntu/Debian
sudo apt install libbpf-dev linux-headers-$(uname -r) clang llvm

# CentOS/RHEL
sudo yum install libbpf-devel kernel-devel clang llvm

# Arch
sudo pacman -S libbpf clang llvm linux-headers

# 源码安装最新
git clone https://github.com/libbpf/libbpf
cd libbpf && make && sudo make install
```

### 最小依赖检查

```bash
# 必备
clang --version       # ≥ 10
llvm-strip --version  # ≥ 10
uname -r              # 内核 ≥ 5.4 推荐
ls /sys/kernel/btf/   # 内核是否有 BTF
```

### 项目骨架

```bash
mkdir myebpf && cd myebpf
cat > .gitignore <<EOF
*.o
*.bpf.o
*.skel.h
vmlinux.h
EOF
mkdir src
```

`src/minimal.bpf.c`：

```c
#include "vmlinux.h"
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_tracing.h>

char LICENSE[] SEC("license") = "GPL";

SEC("tracepoint/syscalls/sys_enter_openat")
int handle_openat(struct trace_event_raw_sys_enter *ctx) {
    char filename[256];
    bpf_probe_read_user_str(filename, sizeof(filename), (char *)ctx->args[0]);
    bpf_printk("openat: %s\n", filename);
    return 0;
}
```

`src/minimal.c`：

```c
#include <bpf/bpf.h>
#include <bpf/libbpf.h>

int main() {
    struct bpf_object *obj = bpf_object__open_file("minimal.bpf.o", NULL);
    if (libbpf_get_error(obj)) {
        fprintf(stderr, "open failed\n");
        return 1;
    }
    if (bpf_object__load(obj)) {
        fprintf(stderr, "load failed\n");
        return 1;
    }

    struct bpf_program *prog = bpf_object__find_program_by_name(obj, "handle_openat");
    struct bpf_link *link = bpf_program__attach(prog);
    if (libbpf_get_error(link)) {
        fprintf(stderr, "attach failed\n");
        return 1;
    }

    printf("running... press Ctrl-C to stop\n");
    while (1) sleep(3600);

    bpf_link__destroy(link);
    bpf_object__close(obj);
    return 0;
}
```

`Makefile`：

```makefile
CLANG     ?= clang
BPFTOOL  ?= bpftool
CFLAGS   := -O2 -g -Wall -target bpf -D__TARGET_ARCH_x86
LDFLAGS  := -lbpf -lelf -lz -static

# 自动生成 vmlinux.h（内核 BTF 头）
VMLINUX := vmlinux.h
$(VMLINUX):
	$(BPFTOOL) btf dump file /sys/kernel/btf/vmlinux format c > $@

# 编译 .bpf.c → .bpf.o
%.bpf.o: %.bpf.c $(VMLINUX)
	$(CLANG) $(CFLAGS) -c $< -o $@

# 生成 .skel.h（用户态绑定）
%.skel.h: %.bpf.o
	$(BPFTOOL) gen skeleton $< > $@

# 用户态编译
minimal: minimal.c minimal.skel.h minimal.bpf.o
	$(CLANG) -O2 -g -Wall minimal.c -o $@ $(LDFLAGS)

.PHONY: clean
clean:
	rm -f *.o *.skel.h vmlinux.h minimal
```

### 什么是 `.skel.h`（BPF skeleton）

`bpftool gen skeleton` 从 `.bpf.o` 自动生成一个 C 头文件，**把所有 map / program / object 的访问绑成强类型 API**：

```c
// 自动生成的 skel.h 暴露：
struct hello_bpf;                         // 主骨架
struct hello_bpf *hello_bpf__open(void);  // 打开
int hello_bpf__load(struct hello_bpf *);   // 加载（触发 verifier）
int hello_bpf__attach(struct hello_bpf *); // 自动挂载
void hello_bpf__destroy(struct hello_bpf *);

// 内嵌子结构，强类型访问 map / program
skel->maps.cfg_map                       // struct bpf_map *
skel->progs.handle_openat                // struct bpf_program *
skel->rodata->my_const                   // 只读常量
skel->bss->my_var                        // 私有变量
```

> 对比「裸 libbpf API」：不用 skeleton 时你得手动 `bpf_object__open_file` + `bpf_object__find_program_by_name` + `bpf_object__find_map_fd_by_name`，字符串易错。Skeleton 是工业级标准，所有新项目都该用。

构建并运行：

```bash
make
sudo ./minimal
# 另一窗口
sudo cat /sys/kernel/debug/tracing/trace_pipe
```

## 2. bpftool

> eBPF 工具的瑞士军刀，由内核源码维护。

### 安装

```bash
# Ubuntu/Debian
sudo apt install linux-tools-common linux-tools-$(uname -r)

# 源码（推荐，最新）
git clone https://github.com/libbpf/bpftool
cd bpftool/src && make && sudo make install
```

### 常用命令

```bash
# 列出已加载的程序
sudo bpftool prog show

# 详细看一个程序（看 verifier 用了哪些 helper、map）
sudo bpftool prog show id 42 --pretty

# 看某程序的字节码/JIT 后的汇编
sudo bpftool prog dump xlated id 42     # 伪 eBPF 字节码
sudo bpftool prog dump jited id 42      # JIT 后的 x86 汇编

# 列 map
sudo bpftool map show

# 看 map 数据（key/value 类型）
sudo bpftool map dump id 10

# 更新 map 数据
sudo bpftool map update id 10 key 0x01 0x00 0x00 0x00 value 0x42 0x00 0x00 0x00 0x00 0x00 0x00 0x00

# 加载 / 卸载程序
sudo bpftool prog load xxx.bpf.o /sys/fs/bpf/xxx
sudo bpftool prog pin /sys/fs/bpf/xxx /sys/fs/bpf/xxx_pinned
sudo bpftool prog detach id 42

# 生成 skeleton 头
bpftool gen skeleton xxx.bpf.o > xxx.skel.h

# 用 BTF 验证结构体偏移
bpftool btf dump file /sys/kernel/btf/vmlinux format c | less
```

## 3. bcc（BPF Compiler Collection）

> Python + C 的运行时编译器。**适合快速开发**，对运行时要求高。

### 安装

```bash
# Ubuntu
sudo apt install bpfcc-tools python3-bpfcc linux-headers-$(uname -r) libbpfcc-dev

# CentOS
sudo yum install bcc-tools python3-bcc

# 验证
sudo /usr/share/bcc/tools/execsnoop
```

### 典型工具

```bash
# 进程 exec 追踪
sudo execsnoop

# 系统调用统计
sudo syscount

# TCP 连接追踪
sudo tcpconnect

# 块设备延迟直方图
sudo biolatency

# 函数调用次数
sudo funccount 'vfs_read'

# 进程可执行文件读写追踪
sudo filelife

# 用户态函数追踪（自定义）
sudo /usr/share/bcc/tools/trace \
    'p::open():arg0' \
    'printf("open: %s\n", arg0)'
```

### Python 嵌入示例

```python
#!/usr/bin/env python3
from bcc import BPF

prog = r"""
#include <uapi/linux/ptrace.h>

BPF_HASH(start, u32, u64);

int trace_start(struct pt_regs *ctx) {
    u32 pid = bpf_get_current_pid_tgid() >> 32;
    u64 ts = bpf_ktime_get_ns();
    start.update(&pid, &ts);
    return 0;
}

int trace_end(struct pt_regs *ctx) {
    u32 pid = bpf_get_current_pid_tgid() >> 32;
    u64 *tsp = start.lookup(&pid);
    if (!tsp) return 0;
    u64 delta = bpf_ktime_get_ns() - *tsp;
    bpf_trace_printk("read() took %llu ns\\n", delta);
    start.delete(&pid);
    return 0;
}
"""

b = BPF(text=prog)
b.attach_kprobe(event="vfs_read", fn_name="trace_start")
b.attach_kretprobe(event="vfs_read", fn_name="trace_end")
b.trace_print()
```

## 4. bpftrace（推荐新手）

> 高级追踪 DSL，单行命令就能完成复杂分析。

### 安装

```bash
# Ubuntu
sudo apt install bpftrace

# 验证
bpftrace --version  # ≥ 0.16 推荐
```

### 一行例子

```bash
# 哪些进程在调用 connect()
sudo bpftrace -e 't:syscalls:sys_enter_connect { printf("%s [%d] -> %s\n", comm, pid, ustarg(0)); }'

# 每个进程 1 秒内的 read 字节数
sudo bpftrace -e 'kprobe:vfs_read { @[comm] = count(); }'

# 函数耗时直方图
sudo bpftrace -e 'kprobe:vfs_read { @start[tid] = nsecs; } 
                  kr:vfs_read / @start[tid] / { @ns = hist(nsecs - @start[tid]); delete(@start[tid]); }'

# 谁打开了 /etc/passwd
sudo bpftrace -e 't:syscalls:sys_enter_openat /str(args->filename) == "/etc/passwd"/ { printf("%s\n", comm); }'

# 块 IO 延迟分布
sudo biolatency.bt  # = bcc biolatency 的 bpftrace 版本
```

### 进阶 - 写脚本

```c
#!/usr/bin/env bpftrace
/* trace_open.bt */

#include <linux/sched.h>

BEGIN {
    printf("Tracing open... Ctrl-C to stop.\n");
}

t:syscalls:sys_enter_openat
{
    @[comm] = count();
    printf("pid=%d comm=%s filename=%s\n", pid, comm, str(args->filename));
}

END {
    printf("\nTop by frequency:\n");
    printa(@);
}
```

```bash
sudo chmod +x trace_open.bt
sudo ./trace_open.bt
```

## 5. LLVM/Clang 注意事项

### 版本要求

| 特性 | 最低 Clang |
|------|-----------|
| 基础 eBPF | 3.4 |
| CO-RE | 9 |
| BTF debug info | 9 |
| ringbuf | 10 |
| 用户态 BTF | 11 |

### 编译选项（推荐）

```bash
clang -O2 -g \
      -target bpf \
      -D__TARGET_ARCH_x86 \     # 或 arm64
      -fno-stack-protector \
      -fno-jump-tables \         # 4.18+ verifier 接受
      -I/usr/include/x86_64-linux-gnu \
      minimal.bpf.c -c -o minimal.bpf.o
```

`-fno-jump-tables` 很重要：某些内核版本对 switch jump table 处理有 verifier bug。

### 多架构编译

```bash
# x86_64
clang -target bpf -D__TARGET_ARCH_x86 -c prog.bpf.c -o prog_x86.bpf.o

# arm64
clang -target bpf -D__TARGET_ARCH_arm64 -c prog.bpf.c -o prog_arm64.bpf.o
```

## 6. 内核配置要求

```bash
# 查看是否启用
zgrep -E 'CONFIG_BPF(_SYSCALL)?=|CONFIG_XDP=|CONFIG_DEBUG_INFO_BTF=' /proc/config.gz
# 或
cat /boot/config-$(uname -r) | grep -E 'CONFIG_BPF|CTF|DEBUG_INFO_BTF'

# 必要项
CONFIG_BPF=y
CONFIG_BPF_SYSCALL=y
CONFIG_BPF_JIT=y
CONFIG_HAVE_EBPF_JIT=y
CONFIG_DEBUG_INFO_BTF=y    # CO-RE 必须
CONFIG_DEBUG_INFO_BTF_MODULES=y
CONFIG_NET_CLS_BPF=y       # TC
CONFIG_NET_ACT_BPF=y       # TC action
CONFIG_BPF_EVENTS=y        # kprobe/tracepoint
CONFIG_BPF_STREAM_PARSER=y # socket
CONFIG_XDP_SOCKETS=y       # AF_XDP
```

## 7. Docker 内开发 eBPF

```dockerfile
FROM ubuntu:22.04
RUN apt-get update && apt-get install -y \
    clang llvm \
    libbpf-dev \
    linux-headers-generic \
    bpftrace bpfcc-tools \
    linux-tools-generic \
    linux-cloud-tools-generic \
    && rm -rf /var/lib/apt/lists/*
```

特权模式启动（必需）：

```bash
docker run --rm -it \
    --privileged \
    --pid host \
    -v /sys:/sys:ro \
    -v /sys/kernel/debug:/sys/kernel/debug:rw \
    your-ebpf-dev /bin/bash
```

或者用 [eunomia-bpf 的镜像](https://github.com/eunomia-bpf/eBPF-hello-world)：

```bash
docker run --rm -it --privileged \
    -v $(pwd):/work \
    ghcr.io/eunomia-bpf/ecli-cc /bin/bash
```
