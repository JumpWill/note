## 一、网络加速与 LB

### 1.1 XDP L4 负载均衡（Katran 设计）

```
客户端 ──→ 网卡 ──→ XDP ──→ 真实服务器
                          │
                          ├─ DSR：原路返回
                          └─ TPR：经 LB 服务器返回
```

**优势**：
- 早于协议栈执行（**30-50ns**）
- 单机可处理 **10M+ PPS**
- 绕过 iptables/NAT 表查询

**代码骨架**：

```c
SEC("xdp")
int xdp_lb(struct xdp_md *ctx) {
    // 1. 解析包
    struct pkt_info pkt;
    if (parse_packet(ctx, &pkt) < 0) return XDP_PASS;

    // 2. 查一致性哈希
    u32 key = pkt.five_tuple_hash;
    struct real *dst = bpf_map_lookup_elem(&server_pool, &key);
    if (!dst) return XDP_PASS;

    // 3. 重写目的 MAC/IP
    swap_mac(ctx);
    rewrite_ip(ctx, dst);
    update_csum(ctx);

    // 4. TX 回原 NIC
    return XDP_TX;
}
```

### 1.2 内核 Socket LB（sock_ops）

服务在连接建立阶段就让 `connect()` 直接连到真实后端，**绕过 LB**。

```c
SEC("sockops")
int bpf_sockops(struct bpf_sock_ops *skops) {
    if (skops->family != AF_INET) return BPF_OK;

    // 当 client 调 connect() 时
    if (skops->op == BPF_SOCK_OPS_TCP_CONNECT_CB) {
        struct bpf_sock_tuple *tuple = skops->remote_ip4;
        u32 *real = bpf_map_lookup_elem(&sk_ops_map, tuple);
        if (real) {
            // 直接重定向到真实后端
            bpf_sk_redirect(skops, real[0], BPF_F_INGRESS);
            return BPF_OK;
        }
    }

    // LB health check 失败时回退
    if (skops->op == BPF_SOCK_OPS_TCP_CONNECT_CB &&
        skops->remote_port == LB_PORT) {
        // ... 跳过
    }

    return BPF_OK;
}
```

**生产意义**：长连接（MySQL/Redis）场景能彻底绕过 LB，延迟下降 30%+。

### 1.3 DDoS 防护

```c
SEC("xdp")
int ddos_filter(struct xdp_md *ctx) {
    struct iphdr *ip = ...;

    // 简单阈值：每秒单 IP 不超过 10000 个 SYN
    u32 src_ip = ip->saddr;
    u64 *cnt = bpf_map_lookup_elem(&syn_count, &src_ip);
    if (cnt && *cnt > 10000) {
        return XDP_DROP;  // 攻击包直接丢
    }

    // 复合规则：TCP 指纹异常
    if (is_tcp_odd(ip)) {
        return XDP_DROP;
    }

    return XDP_PASS;
}
```

**Cloudflare**：基于 XDP 的 DDoS 缓解，每秒处理 **亿级** 包。

### 1.4 Service Mesh：Cilium 替代 iptables

传统 iptables 痛点：

```
Service A → Pod 1000 个 → 10000 条 iptables 规则
每次包经过都要遍历规则 → O(n) 延迟
```

Cilium 方案：

```
Service A → Pod 1000 个 → eBPF map O(1) 查询
```

**性能对比**：

| 节点规模 | iptables | Cilium eBPF |
|---------|----------|-------------|
| 100 pod | 5 ms | 0.1 ms |
| 1000 pod | 50 ms | 0.5 ms |
| 5000 pod | 250 ms (timeout) | 0.8 ms |

### 2.1 运行时威胁检测（Falco）

底层 eBPF：

```c
SEC("tracepoint/syscalls/sys_enter_connect")
int trace_connect(struct trace_event_raw_sys_enter *ctx) {
    struct sockaddr_in *sa = (struct sockaddr_in *)ctx->args[1];

    if (sa->sin_port != bpf_htons(22)) return 0;

    // 计数
    u32 key = sa->sin_addr.s_addr;
    u64 *count = bpf_map_lookup_elem(&ssh_brute, &key);
    if (!count) {
        u64 one = 1;
        bpf_map_update_elem(&ssh_brute, &key, &one, BPF_ANY);
    } else {
        *count += 1;
    }

    return 0;
}
```

### 2.2 LSM 阻断（Tetragon）

```c
// 禁止 nginx 读 /etc/shadow
SEC("lsm/file_open")
int BPF_PROG(restrict_open, struct file *file, int ret) {
    char comm[16];
    bpf_get_current_comm(comm, sizeof(comm));

    // 进程是 nginx
    if (bpf_strncmp(comm, 14, "nginx") != 0) return 0;

    // 路径是 /etc/shadow
    char path[256];
    bpf_d_path(&file->f_path, path, sizeof(path));
    if (bpf_strncmp(path, 12, "/etc/shadow") == 0) {
        bpf_printk("BLOCK: nginx trying to read /etc/shadow\n");
        return -EPERM;  // 阻断
    }
    return 0;
}
```

### 2.3 进程行为监控

```c
// 监控所有进程启动并记录「完整命令行」
SEC("tracepoint/sched/sched_process_exec")
int on_exec(struct trace_event_raw_sched_process_exec *ctx) {
    struct event_t e = {};
    u64 pid_tgid = bpf_get_current_pid_tgid();
    e.pid = pid_tgid >> 32;
    e.tid = pid_tgid & 0xffffffff;

    bpf_get_current_comm(e.comm, sizeof(e.comm));

    // cmdline 从 arg1（data ptr） 读
    bpf_probe_read_str(e.cmdline, sizeof(e.cmdline), (char *)ctx->filename);

    // 写 ringbuf
    struct event_t *pe = bpf_ringbuf_reserve(&rb, sizeof(*pe), 0);
    if (pe) {
        *pe = e;
        bpf_ringbuf_submit(pe, 0);
    }
    return 0;
}
```

### 2.4 凭证审计

```c
// 检测 setuid 调用
SEC("tracepoint/syscalls/sys_enter_setuid")
int on_setuid(struct trace_event_raw_sys_enter *ctx) {
    u32 uid = (u32)ctx->args[0];

    // 仅审计非 root 调用
    if (uid == 0) return 0;

    // 记录
    struct audit_event *e = bpf_ringbuf_reserve(&rb, sizeof(*e), 0);
    if (e) {
        e->event_type = AUDIT_SETUID;
        e->target_uid = uid;
        bpf_ringbuf_submit(e, 0);
    }
    return 0;
}
```

## 三、性能分析

### 3.1 函数耗时统计

```c
// 测量 vfs_read 耗时
SEC("kprobe/vfs_read")
int enter(struct pt_regs *ctx) {
    u64 pid_tgid = bpf_get_current_pid_tgid();
    u64 ts = bpf_ktime_get_ns();
    bpf_map_update_elem(&start_map, &pid_tgid, &ts, BPF_ANY);
    return 0;
}

SEC("kretprobe/vfs_read")
int exit(struct pt_regs *ctx) {
    u64 pid_tgid = bpf_get_current_pid_tgid();
    u64 *start = bpf_map_lookup_elem(&start_map, &pid_tgid);
    if (!start) return 0;

    u64 delta = bpf_ktime_get_ns() - *start;
    bpf_map_delete_elem(&start_map, &pid_tgid);

    // 分类直方图
    u64 slot = log2(delta / 1000);  // 转换为 us，log2 slot
    u32 k = slot;
    u64 *val = bpf_map_lookup_elem(&latency_hist, &k);
    if (val) (*val)++;
    else {
        u64 one = 1;
        bpf_map_update_elem(&latency_hist, &k, &one, BPF_ANY);
    }
    return 0;
}
```

### 3.2 Off-CPU 分析

int on_sched(struct trace_event_raw_sched_switch *ctx) {
    struct task_struct *next = (struct task_struct *)ctx->next_pid;
    u32 pid = bpf_get_current_pid_tgid() >> 32;

    // 简化：保存上一个任务切换时间
    u64 now = bpf_ktime_get_ns();

    if (prev_ts) {
        u64 off_cpu = now - *prev_ts;

}

## 八、企业落地建议

1. **从可观测开始**：用 Pixie/bcc 收集数据，不动业务
2. **从非生产开始**：dev/staging 环境先跑
3. **保持 CO-RE**：5.4+ 内核、libbpf 而非 bcc
4. **监控你的 eBPF**：自身有 bug 也会影响业务
5. **依赖图谱化**：用 [Beyla](https://github.com/grafana/beyla) / [Pixie](https://github.com/pixie-io/pixie) 自动生成
6. **培训**：团队至少 2-3 人深入理解 verifier 限制
7. **备份方案**：eBPF 不可用时降级为传统方案
